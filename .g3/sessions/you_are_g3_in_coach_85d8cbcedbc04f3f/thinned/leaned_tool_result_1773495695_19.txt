"""
QuestionIntentClassifier - classifies screening questions by intent.

Categories:
- critical: Disqualifying questions (visa sponsorship, criminal history)
- important: Significant but not disqualifying (relocation, salary)
- normal: Other questions
"""
import json
import re
from pathlib import Path
from dataclasses import dataclass
from typing import Optional


@dataclass
class QuestionIntent:
    """Represents the classified intent of a screening question."""
    name: str           # "work_authorization", "visa_sponsorship", etc.
    category: str       # "critical", "important", "normal"
    confidence: float   # 0.0 – 1.0


class QuestionIntentClassifier:
    """
    Classifies screening questions into intent categories.
    
    Uses pattern matching against a configurable set of intents.
    
    Usage:
        clf = QuestionIntentClassifier()
        intent = clf.classify("Will you require visa sponsorship?")
        # → QuestionIntent(name="visa_sponsorship", category="critical", confidence=0.9)
    """
    
    def __init__(self, config_path: str = "config/critical_intents.json"):
        """
        Initialize classifier with config file.
        
        Args:
            config_path: Path to JSON config with critical/important intents and patterns
        """
        config_path = Path(config_path)
        if not config_path.exists():
            # Fallback to defaults if config missing
            self.critical: set[str] = {"work_authorization", "visa_sponsorship", 
                                        "security_clearance", "criminal_history"}
            self.important: set[str] = {"relocation", "drivers_license", 
                                         "salary_expectations", "start_date"}
            self.patterns: dict[str, list[re.Pattern]] = {}
        else:
            config = json.loads(config_path.read_text())
            self.critical: set[str] = set(config.get("critical", []))
            self.important: set[str] = set(config.get("important", []))
            self.patterns: dict[str, list[re.Pattern]] = {
                intent: [re.compile(p, re.IGNORECASE) for p in patterns]
                for intent, patterns in config.get("patterns", {}).items()
            }

    def classify(self, question_text: str) -> Optional[QuestionIntent]:
        """
        Classify a screening question by its intent.
        
        Args:
            question_text: The screening question to classify
            
        Returns:
            QuestionIntent if matched, None otherwise
        """
        q = question_text.lower().strip()
        for intent, compiled_patterns in self.patterns.items():
            for pattern in compiled_patterns:
                if pattern.search(q):
                    category = (
                        "critical" if intent in self.critical
                        else "important" if intent in self.important
                        else "normal"
                    )
                    return QuestionIntent(
                        name=intent, 
                        category=category, 
                        confidence=0.9
                    )
        return None

    def is_critical(self, question_text: str) -> bool:
        """
        Check if a question is critical (disqualifying).
        
        Args:
            question_text: The screening question to check
            
        Returns:
            True if the question is classified as critical
        """
        intent = self.classify(question_text)
        return intent is not None and intent.category == "critical"

    def is_important(self, question_text: str) -> bool:
        """
        Check if a question is important but not critical.
        
        Args:
            question_text: The screening question to check
            
        Returns:
            True if the question is classified as important
        """
        intent = self.classify(question_text)
        return intent is not None and intent.category == "important"

    def get_intent_name(self, question_text: str) -> Optional[str]:
        """
        Get just the intent name for a question.
        
        Args:
            question_text: The screening question
            
        Returns:
            Intent name string or None
        """
        intent = self.classify(question_text)
        return intent.name if intent else None
