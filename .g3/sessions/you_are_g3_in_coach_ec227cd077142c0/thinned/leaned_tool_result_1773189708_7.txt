from dataclasses import dataclass
import logging

# Import ErrorType from models to avoid duplication
from src.models.job import ErrorType

logger = logging.getLogger(__name__)


@dataclass
class RetryConfig:
    """Configuration for a retry attempt."""
    attempt: int
    delay_seconds: int
    error_type: ErrorType
    should_retry: bool
    message: str


class RetryPolicy:
    """
    Centralized retry policy for all job application operations.
    
    Key principle: "Never give up" - transient errors should be retried
    with exponential backoff.
    
    Usage:
        policy = RetryPolicy()
        if policy.should_retry(error_type, attempt):
            await asyncio.sleep(policy.delay_seconds(attempt))
            # retry the operation
    """
    
    MAX_RETRIES = 3
    BACKOFF_SECONDS = [60, 300, 900]  # 1 min, 5 min, 15 min
    
    # Errors that are transient and worth retrying
    RETRYABLE_ERRORS = {
        ErrorType.NETWORK_ERROR,
        ErrorType.TIMEOUT,
        ErrorType.STALE_ELEMENT,
        ErrorType.RATE_LIMITED,
        ErrorType.CAPTCHA,
        ErrorType.PAGE_CRASH,
        ErrorType.UNKNOWN_ERROR,
    }
    
    # Errors that are permanent - no point retrying
    NON_RETRYABLE_ERRORS = {
        ErrorType.ALREADY_APPLIED,
        ErrorType.JOB_EXPIRED,
        ErrorType.NOT_ELIGIBLE,
        ErrorType.MISSING_REQUIRED_FIELD,
        ErrorType.BUDGET_EXHAUSTED,
        ErrorType.SITE_BROKEN,
    }
    
    # Site-down specific backoff (longer delays)
    SITE_DOWN_BACKOFF = [3600, 7200, 21600]  # 1hr, 2hr, 6hr
    
    @classmethod
    def should_retry(cls, error_type: ErrorType, attempt: int) -> bool:
        """
        Determine if an operation should be retried.
        
        Args:
            error_type: Classification of the error
            attempt: Current attempt number (0-indexed)
            
        Returns:
            True if retry should be attempted, False otherwise
        """
        if error_type in cls.NON_RETRYABLE_ERRORS:
            logger.info(f"Non-retryable error: {error_type.value}")
            return False
        
        if error_type not in cls.RETRYABLE_ERRORS:
            logger.warning(f"Unknown error type: {error_type.value}, treating as non-retryable")
            return False
        
        if attempt >= cls.MAX_RETRIES:
            logger.info(f"Max retries ({cls.MAX_RETRIES}) reached for {error_type.value}")
            return False
        
        return True
    
    @classmethod
    def delay_seconds(cls, attempt: int, is_site_down: bool = False) -> int:
        """
        Get delay before next retry attempt.
        
        Args:
            attempt: Current attempt number (0-indexed)
            is_site_down: If True, use longer delays for site-down scenarios
            
        Returns:
            Number of seconds to wait before retry
        """
        backoff = cls.SITE_DOWN_BACKOFF if is_site_down else cls.BACKOFF_SECONDS
        idx = min(attempt, len(backoff) - 1)
        return backoff[idx]
    
    @classmethod
    def get_retry_config(cls, error_type: ErrorType, attempt: int, is_site_down: bool = False) -> RetryConfig:
        """
        Get complete retry configuration for an error.
        
        Args:
            error_type: Classification of the error
            attempt: Current attempt number (0-indexed)
            is_site_down: Whether the site appears to be down
            
        Returns:
            RetryConfig with all retry details
        """
        should_retry = cls.should_retry(error_type, attempt)
        delay = cls.delay_seconds(attempt, is_site_down) if should_retry else 0
        
        if error_type in cls.NON_RETRYABLE_ERRORS:
            message = f"Permanent error: {error_type.value}. No retry."
        elif attempt >= cls.MAX_RETRIES:
            message = f"Max retries ({cls.MAX_RETRIES}) exceeded for {error_type.value}"
        else:
            message = f"Will retry {error_type.value} in {delay}s (attempt {attempt + 1}/{cls.MAX_RETRIES})"
        
        return RetryConfig(
            attempt=attempt,
            delay_seconds=delay,
            error_type=error_type,
            should_retry=should_retry,
            message=message
        )
    
    @classmethod
    def classify_error(cls, error: Exception, page_state: str = "") -> ErrorType:
        """
        Classify an exception into an ErrorType.
        
        Args:
            error: The exception that occurred
            page_state: Optional description of page state for context
            
        Returns:
            Appropriate ErrorType
        """
        error_name = type(error).__name__.lower()
        error_message = str(error).lower()
        
        # Network/timeout errors
        if "timeout" in error_name or "timeout" in error_message:
            return ErrorType.TIMEOUT
        
        if "network" in error_name or "connection" in error_message:
            return ErrorType.NETWORK_ERROR
        
        # Playwright specific
        if "stale" in error_message or "detached" in error_message:
            return ErrorType.STALE_ELEMENT
        
        if "target closed" in error_message:
            return ErrorType.PAGE_CRASH
        
        # Rate limiting
        if "rate" in error_message or "429" in error_message or "too many" in error_message:
            return ErrorType.RATE_LIMITED
        
        # CAPTCHA detection
        if "captcha" in error_message or "recaptcha" in error_message or "verify you" in page_state:
            return ErrorType.CAPTCHA
        
        # Already applied detection
        if "already applied" in error_message or "already_submitted" in error_message:
            return ErrorType.ALREADY_APPLIED
        
        # Job expired
        if "expired" in error_message or "no longer available" in error_message or "closed" in error_message:
            return ErrorType.JOB_EXPIRED
        
        # Not eligible
        if "not eligible" in error_message or "requirements not met" in error_message:
            return ErrorType.NOT_ELIGIBLE
        
        return ErrorType.UNKNOWN_ERROR


# Convenience function for common usage
def should_retry_error(error: Exception, attempt: int) -> tuple:
    """
    Quick check if an error should be retried.
    
    Args:
        error: The exception
        attempt: Current attempt number (0-indexed)
        
    Returns:
        Tuple of (should_retry, delay_seconds, error_type)
    """
    error_type = RetryPolicy.classify_error(error)
    config = RetryPolicy.get_retry_config(error_type, attempt)
    return config.should_retry, config.delay_seconds, config.error_type
