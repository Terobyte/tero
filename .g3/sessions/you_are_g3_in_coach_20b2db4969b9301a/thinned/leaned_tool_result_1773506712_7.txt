Tool result: """
AI Cost Tracking — отслеживание расходов на LLM API.

Паттерн: Singleton + ContextVar для job_id без изменения сигнатур клиентов.
Thread-safe через Lock. Персистенция в SQLite и JSON.
"""
from __future__ import annotations

import contextvars
import json
import threading
import time
from dataclasses import dataclass, field, asdict
from datetime import datetime, timezone, date
from pathlib import Path
from typing import Optional

try:
    from loguru import logger
except ImportError:
    import logging
    logger = logging.getLogger(__name__)


# ── Context var: текущий job_id для автоматической привязки ───────────
_current_job_id: contextvars.ContextVar[str | None] = contextvars.ContextVar(
    "_current_job_id", default=None
)


class cost_context:
    """
    Context manager — привязать все LLM-вызовы к job_id.
    Не требует менять сигнатуры клиентов.

    Usage:
        with cost_context("job_abc123"):
            answer = await ai_answerer.ask(question)
        # Все вызовы внутри автоматически записаны под job_abc123
    """
    def __init__(self, job_id: str):
        self.job_id = job_id
        self._token = None

    def __enter__(self):
        self._token = _current_job_id.set(self.job_id)
        return self

    def __exit__(self, *_):
        if self._token:
            _current_job_id.reset(self._token)


def get_current_job_id() -> str | None:
    """Получить текущий job_id из контекста (используется в клиентах)."""
    return _current_job_id.get()


# ── Data models ────────────────────────────────────────────────────────

@dataclass
class CostRecord:
    """Одна запись о расходах на LLM-вызов."""
    provider: str          # "kimi", "glm5", "gemini-flash", etc.
    model: str             # полное имя модели
    input_tokens: int
    output_tokens: int
    cost_usd: float
    job_id: Optional[str] = None
    module: str = ""       # "ai_answerer", "blind_recon", "batch_validator"
    timestamp: float = field(default_factory=time.time)


@dataclass
class BudgetStatus:
    """Результат проверки бюджетных лимитов."""
    within_budget: bool
    spent_today: float
    limit_today: float
    spent_this_app: float
    limit_per_app: float
    warning: Optional[str] = None


# ── Singleton ──────────────────────────────────────────────────────────

class CostTracker:
    """
    Singleton cost tracker для всего процесса.

    Usage:
        tracker = CostTracker.instance()
        cost = tracker.track("kimi", inp=400, out=150, module="ai_answerer")
        print(tracker.get_session_total())
    """

    # Цены за 1K tokens (USD). Обновлять при смене моделей!
    PRICING: dict[str, dict[str, float]] = {
        "kimi":          {"input": 0.0015,   "output": 0.002},
        "glm5":          {"input": 0.0005,   "output": 0.001},
        "gemini-flash":  {"input": 0.000075, "output": 0.0003},
        "gemini-pro":    {"input": 0.00125,  "output": 0.005},
        "claude-sonnet": {"input": 0.003,    "output": 0.015},
        "claude-haiku":  {"input": 0.00025,  "output": 0.00125},
        "kimi-vision":   {"input": 0.002,    "output": 0.003},
    }

    # model_name → pricing key (auto-resolve)
    MODEL_MAP: dict[str, str] = {
        "blackboxai/moonshotai/kimi-k2.5": "kimi",
        "z-ai/glm-5":                       "glm5",
        "gemini-3-flash-preview":           "gemini-flash",
        "gemini-1.5-pro":                   "gemini-pro",
        "claude-sonnet-4-20250514":         "claude-sonnet",
        "claude-haiku-3.5":                 "claude-haiku",
        "kimi-k2.5-vision":                 "kimi-vision",
    }

    _instance: Optional["CostTracker"] = None
    _cls_lock = threading.Lock()

    def __init__(self):
        self._records: list[CostRecord] = []
        self._lock = threading.Lock()
        self._budget = self._load_budget()
        self._persist_path = Path("data/session_costs.json")

    @classmethod
    def instance(cls) -> "CostTracker":
        """Singleton access — потокобезопасный."""
        if cls._instance is None:
            with cls._cls_lock:
                if cls._instance is None:
                    cls._instance = cls()
        return cls._instance

    # ── Tracking ──────────────────────────────────────────────────────

    def track(
        self,
        provider: str,
        input_tokens: int,
        output_tokens: int,
        job_id: Optional[str] = None,
        model: Optional[str] = None,
        module: str = "",
    ) -> float:
        """
        Записать стоимость вызова. Возвращает cost_usd.

        job_id автоматически берётся из cost_context если не передан.
        """
        # Автоматически подхватить job_id из контекста
        effective_job_id = job_id or get_current_job_id()

        # Resolve provider key
        pricing_key = self.MODEL_MAP.get(model or provider, provider)
        prices = self.PRICING.get(pricing_key, {"input": 0.001, "output": 0.002})
        cost = (input_tokens / 1000) * prices["input"] + \
               (output_tokens / 1000) * prices["output"]
        cost = round(cost, 6)

        record = CostRecord(
            provider=pricing_key,
            model=model or provider,
            input_tokens=input_tokens,
            output_tokens=output_tokens,
            cost_usd=cost,
            job_id=effective_job_id,
            module=module,
        )
        with self._lock:
            self._records.append(record)

        logger.debug(
            f"💰 ${cost:.4f} | {pricing_key} in={input_tokens} out={output_tokens}"
            + (f" | job={effective_job_id}" if effective_job_id else "")
            + (f" | [{module}]" if module else "")
        )
        return cost

    # ── Queries ───────────────────────────────────────────────────────

    def get_session_total(self) -> float:
        return round(sum(r.cost_usd for r in self._records), 4)

    def get_job_total(self, job_id: str) -> float:
        return round(sum(r.cost_usd for r in self._records if r.job_id == job_id), 4)

    def get_today_total(self) -> float:
        today = date.today()
        return round(sum(
            r.cost_usd for r in self._records
            if date.fromtimestamp(r.timestamp) == today
        ), 4)

    def get_breakdown_by_provider(self, job_id: Optional[str] = None) -> dict[str, float]:
        records = [r for r in self._records if job_id is None or r.job_id == job_id]
        result: dict[str, float] = {}
        for r in records:
            result[r.provider] = result.get(r.provider, 0.0) + r.cost_usd
        return {k: round(v, 4) for k, v in result.items()}

    def get_job_breakdown(self, job_id: str) -> list[CostRecord]:
        return [r for r in self._records if r.job_id == job_id]

    # ── Budget ────────────────────────────────────────────────────────

    def check_budget(self, job_id: Optional[str] = None) -> BudgetStatus:
        """Проверить лимиты. Вернуть BudgetStatus с предупреждением."""
        spent_today = self.get_today_total()
        spent_app = self.get_job_total(job_id) if job_id else 0.0
        limit_day = self._budget.get("max_per_day", 5.0)
        limit_app = self._budget.get("max_per_application", 0.10)
        warn_pct = self._budget.get("warn_at_percent", 80) / 100

        warning = None
        within = True

        if spent_today >= limit_day:
            within = False
            warning = f"❌ Daily budget exceeded: ${spent_today:.2f} / ${limit_day:.2f}"
        elif spent_today >= limit_day * warn_pct:
            warning = f"⚠️ Daily budget at {spent_today/limit_day:.0%}: ${spent_today:.2f} / ${limit_day:.2f}"

        if job_id and spent_app >= limit_app:
            within = False
            warning = f"❌ Per-app budget exceeded: ${spent_app:.4f} / ${limit_app:.2f}"

        return BudgetStatus(
            within_budget=within,
            spent_today=spent_today,
            limit_today=limit_day,
            spent_this_app=spent_app,
            limit_per_app=limit_app,
            warning=warning,
        )

    # ── Persistence ───────────────────────────────────────────────────

    def save_session(self):
        """Сохранить сессию в data/session_costs.json."""
        self._persist_path.parent.mkdir(parents=True, exist_ok=True)
        data = {
            "session_total_usd": self.get_session_total(),
            "today_total_usd": self.get_today_total(),
            "by_provider": self.get_breakdown_by_provider(),
            "record_count": len(self._records),
        }
        self._persist_path.write_text(json.dumps(data, indent=2))
        logger.info(f"💾 Cost session saved: ${data['session_total_usd']:.4f}")

    async def save_to_db(self, job_id: str):
        """Persist per-job costs to SQLite."""
        try:
            from src.utils.database import async_session
            from sqlalchemy import text
            total = self.get_job_total(job_id)
            breakdown = json.dumps(self.get_breakdown_by_provider(job_id))
            async with async_session() as session:
                await session.execute(
                    text("""INSERT OR REPLACE INTO ai_costs
                            (job_id, total_usd, breakdown, created_at)
                            VALUES (:job_id, :total, :breakdown, :ts)"""),
                    {"job_id": job_id, "total": total, "breakdown": breakdown,
                     "ts": datetime.now(timezone.utc).isoformat()},
                )
                await session.commit()
        except Exception as e:
            logger.warning(f"⚠️ CostTracker.save_to_db failed: {e}")

    def _load_budget(self) -> dict:
        try:
            cfg = json.loads(Path("config/settings.json").read_text())
            return cfg.get("cost_limits", {})
        except Exception:
            return {}

    def reset(self):
        """Сброс для тестов."""
        with self._lock:
            self._records.clear()
