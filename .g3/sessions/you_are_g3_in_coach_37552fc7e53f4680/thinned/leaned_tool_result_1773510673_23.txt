def setup_logging():
    """Configure loguru logging."""
    logger.remove()
    logger.add(
        sys.stderr,
        format="<green>{time:HH:mm:ss}</green> | <level>{level: <8}</level> | <cyan>{name}</cyan>:<cyan>{function}</cyan>:<cyan>{line}</cyan> - <level>{message}</level>",
        level="INFO",
    )
    logger.add(
        "logs/careerbot.log",
        rotation="10 MB",
        retention="7 days",
        level="DEBUG",
    )


def load_config() -> dict:
    """Load configuration from config/settings.json."""
    config_path = Path("config/settings.json")
    if config_path.exists():
        return json.loads(config_path.read_text())
    return {}


async def reindex_knowledge_base():
    """Reindex PKB in ChromaDB."""
    logger.info("🔄 Reindexing knowledge base...")
    
    try:
        from src.ai.knowledge_base import VectorKnowledgeBase
        from src.pkb.indexer import PKBIndexer
        
        kb = VectorKnowledgeBase()
        indexer = PKBIndexer()
        
        chunks = indexer.get_chunks()
        count = kb.index_pkb(chunks, force=True)
        
        # Also index Q&A pairs
        qa_path = Path("data/unknown_questions.json")
        if qa_path.exists():
            qa_entries = json.loads(qa_path.read_text(encoding="utf-8"))
            qa_count = kb.index_qa_cache(qa_entries)
            logger.info(f"✅ Indexed {qa_count} Q&A pairs")
        
        stats = kb.stats()
        logger.info(f"✅ Knowledge base reindexed: {stats}")
        
    except ImportError as e:
        logger.error(f"Failed to import required modules: {e}")
        logger.info("Make sure chromadb is installed: pip install chromadb>=0.5.0")
        sys.exit(1)
    except Exception as e:
        logger.error(f"Failed to reindex knowledge base: {e}")
        sys.exit(1)

def main():
    """Main entry point."""
    parser = argparse.ArgumentParser(description="CareerBot - AI-powered job application assistant")
    
    parser.add_argument("--apply", action="store_true", help="Apply to a job")
    parser.add_argument("--tailor", action="store_true", help="Tailor application for a job")
    parser.add_argument("--reindex-kb", action="store_true", help="Reindex knowledge base")
    parser.add_argument("--report-costs", action="store_true", help="Show cost report")
    parser.add_argument("--job-id", type=str, help="Job ID for apply/tailor")
    
    args = parser.parse_args()
    
    setup_logging()
    config = load_config()
    
    if args.reindex_kb:
        asyncio.run(reindex_knowledge_base())
    elif args.report_costs:
        asyncio.run(report_costs())
    elif args.tailor and args.job_id:
        asyncio.run(tailor_job(args.job_id))
    elif args.apply and args.job_id:
        asyncio.run(apply_job(args.job_id))
    else:
        parser.print_help()
