import os
from pathlib import Path


# Resume file path - used by both GreenhouseApplier and IndeedApplier
RESUME_PATH = Path(os.getenv("RESUME_PATH", "data/resume.pdf"))

# Cover letter file path - used by GreenhouseApplier

# Screenshots directory - used for debugging and error capture
SCREENSHOTS_DIR = Path(os.getenv("SCREENSHOTS_DIR", "data/screenshots"))


def ensure_dirs() -> None:
    """Ensure all required directories exist."""
    SCREENSHOTS_DIR.mkdir(parents=True, exist_ok=True)
