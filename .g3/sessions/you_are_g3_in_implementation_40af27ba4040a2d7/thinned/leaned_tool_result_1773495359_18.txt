"""
SessionContext — holds all Q&A for a screening session.

Used by BatchStepValidator to validate answers as a coherent whole
before submitting the application.

Usage:
    ctx = SessionContext(job_url="https://...")
    ctx.add(
        question_text="Are you authorized to work in the US?",
        field_type="radio",
        options=["Yes", "No"],
        answer="Yes",
        intent="work_authorization",
        is_critical=True,
    )
    # Later: validate all answers together
    validator = BatchStepValidator(ai_client, profile)
    corrections = await validator.validate(ctx)
"""
from dataclasses import dataclass, field
from typing import Optional


@dataclass
class AnsweredQuestion:
    """A single answered screening question."""
    text: str
    field_type: str  # "radio", "select", "text", "checkbox", etc.
    options: list[str]
    given_answer: str
    intent: Optional[str] = None  # "work_authorization", "visa_sponsorship", etc.
    is_critical: bool = False
    corrected_answer: Optional[str] = None


@dataclass
class SessionContext:
    """
    Holds all screening Q&A for a single job application session.
    
    Tracks which questions were answered, their intents, and whether
    any were flagged as critical (disqualifying if answered incorrectly).
    """
    job_url: str = ""
    step: int = 0
    answered: list[AnsweredQuestion] = field(default_factory=list)

    def add(
        self,
        question_text: str,
        field_type: str,
        options: list[str],
        answer: str,
        intent: Optional[str] = None,
        is_critical: bool = False,
    ) -> None:
        """Add an answered question to the session."""
        self.answered.append(
            AnsweredQuestion(
                text=question_text,
                field_type=field_type,
                options=options,
                given_answer=answer,
                intent=intent,
                is_critical=is_critical,
            )
        )

    def critical_answers(self) -> list[AnsweredQuestion]:
        """Return only questions marked as critical."""
        return [a for a in self.answered if a.is_critical]

    def to_dict(self) -> dict:
        """Serialize to dict for logging/debugging."""
        return {
            "job_url": self.job_url,
            "step": self.step,
            "answers": [
                {
                    "question": a.text,
                    "field_type": a.field_type,
                    "options": a.options,
                    "answer": a.given_answer,
                    "intent": a.intent,
                    "critical": a.is_critical,
                    "corrected": a.corrected_answer,
                }
                for a in self.answered
            ],
        }

    def apply_correction(self, question_text: str, corrected_answer: str) -> bool:
        """
        Apply a correction to a previously answered question.
        
        Returns True if the question was found and updated.
        """
        for a in self.answered:
            if a.text == question_text:
                a.corrected_answer = corrected_answer
                return True
        return False

    def get_final_answer(self, question_text: str) -> Optional[str]:
        """
        Get the final answer for a question.
        
        Returns corrected_answer if set, otherwise given_answer.
        """
        for a in self.answered:
            if a.text == question_text:
                return a.corrected_answer or a.given_answer
        return None

    def __len__(self) -> int:
        return len(self.answered)

    def __bool__(self) -> bool:
        return bool(self.answered)
