"""
BatchStepValidator - validates all screening answers in one AI call.

Detects:
- Disqualifying answers (e.g., "Yes" to visa sponsorship)
- Inconsistencies between answers
- Corrections needed based on candidate profile
"""
import json
from dataclasses import dataclass
from typing import Optional, Protocol
from loguru import logger

from src.applier.universal_screening.session_context import SessionContext


@dataclass
class Correction:
    """Represents a correction needed for a screening answer."""
    question_text: str
    current_answer: str
    corrected_answer: str
    reason: str
    severity: str   # "critical", "important", "minor"


class AIClientProtocol(Protocol):
    """Protocol for AI client interface."""
    async def generate(self, prompt: str, **kwargs) -> str: ...


class BatchStepValidator:
    """
    Validates screening answers in a single AI call.
    
    Checks for:
    - Disqualifying answers that should be corrected
    - Inconsistencies between related answers
    - Answers that contradict the candidate profile
    
    Usage:
        validator = BatchStepValidator(ai_client, candidate_profile)
        corrections = await validator.validate(session_context)
        critical = [c for c in corrections if c.severity == "critical"]
    """
    
    _SYSTEM = """You are a job application screening validator.
Review screening answers for inconsistencies or disqualifying responses.

Candidate profile:
- Work authorization: {work_auth}
- Needs visa sponsorship: {needs_sponsorship}

Rules:
1. "Yes" to needing visa sponsorship → CRITICAL error (correct to "No")
2. "No" to work authorization → CRITICAL error (correct to "Yes")
3. Flag inconsistencies between answers

Return JSON only:
{{"corrections": [{{"question":"...","current_answer":"...","corrected_answer":"...","reason":"...","severity":"critical|important|minor"}}], "has_critical_errors": false}}"""

    def __init__(self, ai_client: AIClientProtocol, profile: dict):
        """
        Initialize validator.
        
        Args:
            ai_client: AI client with generate() method
            profile: Candidate profile with work_auth settings
        """
        self._ai = ai_client
        self._profile = profile

    async def validate(self, ctx: SessionContext) -> list[Correction]:
        """
        Validate all answers in the session context.
        
        Args:
            ctx: SessionContext with answered questions
            
        Returns:
            List of Correction objects for answers needing changes
        """
        if not ctx.answered:
            return []
        
        prompt = self._build_prompt(ctx)
        
        try:
            raw = await self._ai.generate(prompt, max_tokens=1000)
            data = json.loads(raw)
            corrections = [
                Correction(
                    question_text=c["question"],
                    current_answer=c["current_answer"],
                    corrected_answer=c["corrected_answer"],
                    reason=c["reason"],
                    severity=c["severity"],
                )
                for c in data.get("corrections", [])
            ]
            if corrections:
                logger.warning(f"BatchValidator: {len(corrections)} corrections needed")
            return corrections
        except json.JSONDecodeError as e:
            logger.error(f"BatchValidator JSON parse error: {e}")
            return []
        except KeyError as e:
            logger.error(f"BatchValidator missing key: {e}")
            return []
        except Exception as e:
            logger.error(f"BatchValidator unexpected error: {e}")
            return []

    def _build_prompt(self, ctx: SessionContext) -> str:
        """
        Build the validation prompt from session context.
        
        Args:
            ctx: Session context with answered questions
            
        Returns:
            Formatted prompt string for AI
        """
        wa = self._profile.get("work_auth", {})
        system = self._SYSTEM.format(
            work_auth=wa.get("authorized", "yes"),
            needs_sponsorship=wa.get("needs_sponsorship", "no"),
        )
        
        qa_block = "\n".join(
            f"Q: {a.text}\nA: {a.given_answer}\nOptions: {a.options}\n"
            for a in ctx.answered
        )
        
        return f"{system}\n\nAnswers:\n{qa_block}"

    def has_critical_errors(self, corrections: list[Correction]) -> bool:
        """
        Check if any corrections are critical severity.
        
        Args:
            corrections: List of corrections
            
        Returns:
            True if any correction has severity "critical"
        """
        return any(c.severity == "critical" for c in corrections)
