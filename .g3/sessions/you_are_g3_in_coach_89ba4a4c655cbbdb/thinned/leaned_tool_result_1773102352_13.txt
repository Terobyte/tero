Tool result: """
Question discovery module — finds all form questions on a page.

5 detection strategies:
1. fieldset > legend (standard + 4 sources for legend)
2. [role='radiogroup'] / [role='group'] (ARIA groups)
3. input[type=radio] grouped by name (Indeed consent)
4. <select> elements
5. input[type=text/number], input[type=checkbox], <textarea>
"""

from typing import List, Optional, Any
from loguru import logger

from .models import Question


async def find_all_questions(page) -> List[Question]:
    """
    Find all questions on the current form step.
    
    Indeed uses fieldset/legend for radio-questions
    and label/input pairs for others.

    Args:
        page: Playwright page object

    Returns:
        List of Question objects
    """
    questions = []

    try:
        # 1a. Radio groups (fieldset > legend)
        # Indeed may put the question text in:
        #   a) <legend> inside the fieldset (standard)
        #   b) aria-label / aria-labelledby on the <fieldset> itself
        #   c) A preceding sibling element (p/div/label) OUTSIDE the fieldset
        # We try all four sources so we never miss a radio question.
        fieldsets = await page.locator("fieldset").all()
        for fs in fieldsets:
            try:
                legend: str = ""

                # Source 1: <legend> inside fieldset
                try:
                    legend_el = fs.locator("legend").first
                    if await legend_el.count() > 0:
                        legend = (await legend_el.text_content(timeout=1500) or "").strip()
                except Exception:
                    pass

                # Source 2: aria-label directly on <fieldset>
                if not legend:
                    try:
                        legend = (await fs.evaluate(
                            "el => el.getAttribute('aria-label') || ''"
                        )).strip()
                    except Exception:
                        pass

                # Source 3: aria-labelledby → referenced element text
                if not legend:
                    try:
                        legend = (await fs.evaluate("""el => {
                            const id = el.getAttribute('aria-labelledby');
                            if (id) {
                                const ref = document.getElementById(id);
                                if (ref) return ref.textContent || '';
                            }
                            return '';
                        }""")).strip()
                    except Exception:
                        pass

                # Source 4: preceding sibling / parent's preceding sibling
                # (Indeed screening forms often render the question label as a
                #  <p> or <div> BEFORE the <fieldset>, with no <legend> at all)
                if not legend:
                    try:
                        legend = (await fs.evaluate("""el => {
                            const prev = el.previousElementSibling;
                            if (prev && prev.textContent.trim())
                                return prev.textContent.trim();
                            const parent = el.parentElement;
                            if (parent) {
                                const pp = parent.previousElementSibling;
                                if (pp && pp.textContent.trim())
                                    return pp.textContent.trim();
                            }
                            return '';
                        }""")).strip()
                    except Exception:
                        pass

                if not legend:
                    continue

                # Strip trailing asterisk / required marker
                legend = legend.rstrip(" *\u00a0").strip()

                radios = await fs.locator("input[type='radio']").all()
                options = []

                for radio in radios:
                    try:
                        label = await radio.evaluate(
                            "el => el.labels?.[0]?.textContent "
                            "|| document.querySelector(`label[for='${el.id}']`)?.textContent "
                            "|| el.getAttribute('aria-label') || ''"
                        )
                        if label:
                            options.append(label.strip())
                    except Exception:
                        pass

                if legend and options:
                    questions.append(Question(
                        text=legend,
                        field_type="radio",
                        options=options,
                        element=fs,
                        required=True
                    ))
                    logger.debug(f"📋 Found radio (fieldset): '{legend[:50]}' options={options}")

            except Exception as e:
                logger.debug(f"⚠️ Failed to parse fieldset: {e}")

        # 1b. ARIA-based radio groups — Indeed consent questions use
        # div[role='radiogroup'] with div[role='radio'] children instead
        # of <fieldset><legend> + <input type='radio'>.
        # Deduplication: skip ARIA groups already found via fieldset
        seen_question_texts = {q.text for q in questions if q.field_type == "radio"}
        seen_groups: set = set()
        aria_groups = await page.locator("[role='radiogroup'], [role='group']").all()
        for grp in aria_groups:
            try:
                # get a stable identity key to avoid duplicates
                grp_id = await grp.evaluate("el => el.id || el.getAttribute('aria-labelledby') || el.outerHTML.slice(0,80)")
                if grp_id in seen_groups:
                    continue
                seen_groups.add(grp_id)

                # resolve the group label
                legend = await grp.evaluate("""
                    el => {
                        // aria-labelledby → referenced element text
                        const lbId = el.getAttribute('aria-labelledby');
                        if (lbId) {
                            const lb = document.getElementById(lbId);
                            if (lb) return lb.textContent;
                        }
                        // aria-label directly on group
                        const al = el.getAttribute('aria-label');
                        if (al) return al;
                        // preceding sibling or parent with legend/label role
                        const prev = el.previousElementSibling;
                        if (prev) return prev.textContent;
                        return '';
                    }
                """)
                if not legend or not legend.strip():
                    continue
                legend = legend.strip()

                # collect child radio options — both input[type=radio] and role=radio
                aria_radios = await grp.locator("[role='radio'], input[type='radio']").all()
                options = []
                for r in aria_radios:
                    try:
                        opt_label = await r.evaluate("""
                            el => el.getAttribute('aria-label')
                                || el.labels?.[0]?.textContent
                                || document.querySelector(`label[for='${el.id}']`)?.textContent
                                || el.textContent
                                || ''
                        """)
                        if opt_label and opt_label.strip():
                            options.append(opt_label.strip())
                    except Exception:
                        pass

                # Skip if this question already found via fieldset
                if legend in seen_question_texts:
                    logger.debug(f"📋 Skip ARIA duplicate: '{legend[:50]}'")
                    continue

                if legend and options:
                    questions.append(Question(
                        text=legend,
                        field_type="radio",
                        options=options,
                        element=grp,
                        required=True
                    ))
                    seen_question_texts.add(legend)
                    logger.debug(f"📋 Found radio (aria): '{legend[:50]}' options={options}")

            except Exception as e:
                logger.debug(f"⚠️ Failed to parse aria radiogroup: {e}")

        # 1c. Indeed custom questions — plain input[type=radio] grouped by name
        # Structure: <label data-testid="input-{name}-label">Question text</label>
        #            <label for="id"><input type="radio" name="q_xxx" value="1">Strong No</label>
        # NO fieldset, NO role=radiogroup. Detected by grouping radios by name attr.
        try:
            already_detected_texts = {q.text for q in questions if q.field_type == "radio"}
            radio_groups = await page.evaluate("""
                () => {
                    const groups = new Map();
                    for (const inp of document.querySelectorAll('input[type="radio"]')) {
                        const name = inp.name;
                        if (!name || name.length < 5) continue;
                        if (!groups.has(name)) {
                            // Question text is in an element with data-testid="input-{name}-label"
                            let qText = '';
                            const qlEl = document.querySelector(
                                `[data-testid="input-${name}-label"]`
                            );
                            if (qlEl) {
                                // strip trailing asterisk / whitespace
                                qText = qlEl.textContent.replace(/[\\s\\u00a0\\*]+$/, '').trim();
                            }
                            groups.set(name, { name, text: qText, options: [] });
                        }
                        // Option text from the wrapping <label> element
                        const wrappingLabel =
                            inp.closest('label') ||
                            (inp.id && document.querySelector(`label[for="${inp.id}"]`));
                        const optText = wrappingLabel
                            ? wrappingLabel.textContent.trim()
                            : (inp.value || '');
                        if (optText) groups.get(name).options.push(optText);
                    }
                    return Array.from(groups.values());
                }
            """)

            for grp in radio_groups:
                grp_text = (grp.get("text") or "").strip()
                grp_name = grp.get("name", "")
                grp_options = grp.get("options", [])
                if not grp_text or not grp_options:
                    continue
                # Skip if already found via fieldset/ARIA
                if grp_text in already_detected_texts:
                    continue
                # Store radio group name as a special marker string
                questions.append(Question(
                    text=grp_text,
                    field_type="radio",
                    options=grp_options,
                    element=f"__name__{grp_name}",  # handled in _fill_radio
                    required=True
                ))
                logger.debug(
                    f"📋 Found radio (name-group): '{grp_text[:50]}' "
                    f"options={grp_options}"
                )

        except Exception as e:
            logger.debug(f"⚠️ Failed to parse name-group radios: {e}")

        # 2. Select dropdowns
        selects = await page.locator("select").all()
        for sel in selects:
            try:
                label_text = await get_label_for_element(sel, page)
                if not label_text:
                    label_text = await sel.evaluate("el => el.getAttribute('aria-label') || el.getAttribute('name') || ''")

                if not label_text:
                    continue

                options = await sel.locator("option").all_text_contents()
                options = [o.strip() for o in options if o.strip() and o.strip().lower() not in ["select...", "select an option", "choose..."]]

                if label_text and options:
                    questions.append(Question(
                        text=label_text.strip(),
                        field_type="select",
                        options=options,
                        element=sel,
                        required=True
                    ))
                    logger.debug(f"📋 Found select: '{label_text[:50]}...' options={len(options)}")

            except Exception as e:
                logger.debug(f"⚠️ Failed to parse select: {e}")

        # 3. Text/number inputs with labels
        inputs = await page.locator("input[type='text'], input[type='number']").all()
        for inp in inputs:
            try:
                label_text = await get_label_for_element(inp, page)
                if not label_text:
                    continue

                input_type = await inp.evaluate("el => el.type")
                
                questions.append(Question(
                    text=label_text.strip(),
                    field_type=input_type,
                    options=[],
                    element=inp,
                    required=True
                ))
                logger.debug(f"📋 Found input: '{label_text[:50]}...'")
                
            except Exception as e:
                logger.debug(f"⚠️ Failed to parse input: {e}")

        # 4. Checkboxes with labels
        checkboxes = await page.locator("input[type='checkbox']").all()
        for cb in checkboxes:
            try:
                label_text = await get_label_for_element(cb, page)
                if not label_text:
                    continue
                
                questions.append(Question(
                    text=label_text.strip(),
                    field_type="checkbox",
                    options=["Yes", "No"],
                    element=cb,
                    required=False
                ))
                logger.debug(f"📋 Found checkbox: '{label_text[:50]}...'")
                
            except Exception as e:
                logger.debug(f"⚠️ Failed to parse checkbox: {e}")

        # 5. Textareas (cover letter, open-ended)
        textareas = await page.locator("textarea").all()
        for ta in textareas:
            try:
                label_text = await get_label_for_element(ta, page)
                if not label_text:
                    label_text = "Cover letter or additional information"
                
                questions.append(Question(
                    text=label_text.strip(),
                    field_type="textarea",
                    options=[],
                    element=ta,
                    required=False
                ))
                logger.debug(f"📋 Found textarea: '{label_text[:50]}...'")
                
            except Exception as e:
                logger.debug(f"⚠️ Failed to parse textarea: {e}")

    except Exception as e:
        logger.error(f"❌ Error finding questions: {e}")
    
    logger.info(f"📋 Found {len(questions)} questions on form")
    return questions


async def get_label_for_element(element, page) -> Optional[str]:
    """Get label text for an element. Uses short timeouts to avoid 30s Playwright waits."""
    try:
        # Try by id → label[for="id"]
        elem_id = await element.evaluate("el => el.id")
        if elem_id:
            try:
                label = await page.locator(f"label[for='{elem_id}']").first.text_content(timeout=1500)
                if label:
                    return label.strip()
            except Exception:
                pass  # No label[for=id] — don't wait 30s

        # Try parent label or adjacent label
        parent_label = await element.evaluate("""
            el => {
                const label = el.closest('label');
                if (label) return label.textContent;

                // Try preceding sibling label
                const prev = el.previousElementSibling;
                if (prev && prev.tagName === 'LABEL') return prev.textContent;

                // Try parent's preceding sibling
                const parent = el.parentElement;
                if (parent) {
                    const parentPrev = parent.previousElementSibling;
                    if (parentPrev && parentPrev.tagName === 'LABEL') return parentPrev.textContent;
                }

                return null;
            }
        """)
        if parent_label:
            return parent_label.strip()

        # Try aria-label
        aria_label = await element.evaluate("el => el.getAttribute('aria-label')")
        if aria_label:
            return aria_label.strip()

        # Try placeholder (last resort)
        placeholder = await element.evaluate("el => el.placeholder")
        if placeholder:
            return placeholder.strip()

    except Exception:
        pass

    return None
