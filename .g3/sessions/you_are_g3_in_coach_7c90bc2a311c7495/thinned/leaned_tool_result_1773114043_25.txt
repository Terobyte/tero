# Получить все опции из видимого listbox# Greenhouse использует CSS-классы ('-option') вместо role='option'
GET_OPTS_JS = """
    (el) => {
        // Сначала пробуем aria-owns/aria-controls
        const ownedId = el.getAttribute('aria-owns') || el.getAttribute('aria-controls');
        let list = ownedId ? document.getElementById(ownedId) : null;
        if (!list) {
            // Видимый listbox по role
            for (const lb of document.querySelectorAll("[role='listbox']")) {
                if (lb.offsetParent !== null) { list = lb; break; }            }
        }
        if (!list) {
            // Видимый menu-контейнер (react-select / Greenhouse custom)
            for (const m of document.querySelectorAll("[class*='menu']")) {
                const cls = m.className || '';
                if (m.offsetParent !== null && !cls.match(/control|container|indicator/i)) {
                    list = m; break;
                }
            }
        }
        if (!list) return [];
        // Ищем опции: сначала role='option', потом CSS-class паттерн
        let opts = Array.from(list.querySelectorAll("[role='option']"));
        if (opts.length === 0) {
            opts = Array.from(list.querySelectorAll("[class*='-option']")).filter(o => {
                const cls = o.className || '';
                return o.offsetParent !== null && !cls.match(/control|container|placeholder|group-heading/i);
            });
        }
        return opts.map(o => ({ text: o.innerText.trim(), id: o.id || '' }));
    }
"""

# Найти кнопку "Other" / "I don't see my school" в dropdown
FIND_OTHER_OPTION_JS = """
    () => {
        const keywords = ["other", "not listed", "i don't see", "don't see",
                           "none of the above", "not in the list",
                           "my school", "school not listed", "can't find"];
        // Ищем в любом видимом dropdown/menu/listbox И рядом с ним
        const candidates = document.querySelectorAll(
            "[role='option'], [class*='-option'], " +
            "[role='listbox'] *, [class*='menu'] *, " +
            "[class*='select'] button, [class*='select'] a, " +
            "[class*='menu'] button, [class*='menu'] a, " +
            "[class*='menu'] div, [class*='menu'] span"
        );
        for (const el of candidates) {
            if (el.offsetParent === null) continue;
            const t = (el.innerText || el.textContent || '').trim();
            if (t.length < 3 || t.length > 150) continue;
            const tl = t.toLowerCase();
            for (const kw of keywords) {
                if (tl.includes(kw)) {
                    el.scrollIntoView({block: 'center', behavior: 'instant'});
                    const r = el.getBoundingClientRect();
                    return {
                        text: t, tag: el.tagName,
                        id: el.id || '', cls: el.className || '',
                        x: r.x + r.width / 2, y: r.y + r.height / 2
                    };
                }
            }
        }
        return null;
    }

# Кликнуть по тексту через JS dispatchEvent
CLICK_BY_TEXT_JS = """
    (text) => {
        for (const el of document.querySelectorAll(
            "[role='option'], [class*='-option'], " +
            "[role='listbox'] *, [class*='menu'] *, " +
            "button, a, [class*='select'] *, div, span, li"
        )) {
            if (el.offsetParent === null) continue;
            const t = (el.innerText || '').trim();
            if (t === text || t.toLowerCase() === text.toLowerCase()) {
                // Полная цепочка событий мыши (React слушает именно их)
                el.dispatchEvent(new MouseEvent('mousedown', {bubbles: true}));
                el.dispatchEvent(new MouseEvent('mouseup', {bubbles: true}));
                el.dispatchEvent(new MouseEvent('click', {bubbles: true}));
                return true;
            }
        }
        return false;
    }
"""

# Получить все видимые опции из menu (без role='listbox')
GET_MENU_OPTS_JS = """
    () => {
        const selectors = [
            "[role='option']",
            "[class*='-option']",
            "[class*='menu'] *",
            "[class*='menu'] button",
            "[class*='menu'] a",
            "[class*='menu'] div",
            "[class*='menu'] span"
        ];
        const results = [];
        const seen = new Set();
        for (const sel of selectors) {
            for (const el of document.querySelectorAll(sel)) {
                const t = (el.innerText || '').trim();
                if (t && el.offsetParent !== null && t.length < 100 && !seen.has(t)) {
                    seen.add(t);
                    const r = el.getBoundingClientRect();
                    results.push({
                        text: t, id: el.id || '',
                        x: r.x + r.width / 2, y: r.y + r.height / 2
                    });
                }
            }
            if (results.length) break;
        }
        return results;
    }
"""
