"""
Data models and HITL helpers for universal screening.
"""

from dataclasses import dataclass
from typing import Any, List


def is_hitl_needed(field_type: str, question_text: str) -> bool:
    """True if HITL menu is needed (textarea or motivation question)."""
    if field_type == "textarea":
        return True
    return is_hitl_question(question_text)


def is_hitl_question(text: str) -> bool:
    """True if question is about motivation/culture/finance — needs HITL."""
    t = text.lower()
    return any(kw in t for kw in [
        "why", "describe", "tell us", "explain", "what motivates",
        "why do you want", "what excites", "greatest strength",
        "greatest weakness", "tell me about yourself",
        "salary", "compensation", "expected pay", "pay expectation",
    ])


@dataclass
class Question:
    """Represents a form question."""
    text: str
    field_type: str  # radio, select, text, number, checkbox, textarea
    options: List[str]
    element: Any  # Playwright locator
    required: bool = True
