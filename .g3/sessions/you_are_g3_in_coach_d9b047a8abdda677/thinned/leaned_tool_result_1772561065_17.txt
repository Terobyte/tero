class ProfileInterviewer:
    """
    Telegram bot for profile setup via interactive interview.
    
    Usage:
        # In your Telegram bot handler:
        interviewer = ProfileInterviewer()
        
        # Start interview
        await bot.send_message(chat_id, interviewer.start())
        keyboard = interviewer.get_keyboard()
        
        # Handle answers
        next_msg = interviewer.handle_answer(user_answer)
        if next_msg:
            await bot.send_message(chat_id, next_msg, reply_markup=interviewer.get_keyboard())
        
        # When complete
        if interviewer.is_complete():
            interviewer.save_profile()
            await bot.send_message(chat_id, "✅ Профиль сохранён!")
    """
    
    def __init__(self, profile_path: str = "data/candidate/profile.yaml"):
        self.profile_path = Path(profile_path)
        self.questions = INTERVIEW_QUESTIONS.copy()
        self.current_index = 0
        self.answers: Dict[str, Any] = {}
        self.current_block: Optional[str] = None
        self._started = False
        
    def start(self) -> str:
        """Start the interview and return first question."""
        self._started = True
        self.current_index = 0
        self.current_block = None
        return self._get_next_message()
    
    def _get_next_message(self) -> str:
        """Get the next question message."""
        if self.current_index >= len(self.questions):
            return self._get_completion_message()
        
        q = self.questions[self.current_index]
        
        # Announce new block
        if self.current_block != q.block:
            self.current_block = q.block
            return f"\n📌 [БЛОК: {q.block}]\n\n{q.question}"
        
        return q.question
    
    def _get_completion_message(self) -> str:
        """Get completion message."""
        return "✅ Интервью завершено! Профиль будет сохранён в data/candidate/profile.yaml"
    
    def get_keyboard(self) -> Optional[Dict]:
        """
        Get inline keyboard for current question.
        
        Returns None for text/number questions (user types answer).
        Returns keyboard dict for yes_no questions.
        """
        if self.current_index >= len(self.questions):
            return None
            
        q = self.questions[self.current_index]
        
        if q.question_type == "yes_no":
            return {
                "inline_keyboard": [
                    [
                        {"text": "✅ Да", "callback_data": "yes"},
                        {"text": "❌ Нет", "callback_data": "no"},
                        {"text": "⏭ Пропустить", "callback_data": "skip"}
                    ]
                ]
            }
        
        return None
    
    def handle_answer(self, answer: str) -> Optional[str]:
        """
        Handle user answer and return next question.
        
        Args:
            answer: User's answer (text, "yes", "no", "skip", or number)
            
        Returns:
            Next question message or completion message
        """
        if self.current_index >= len(self.questions):
            return None
        
        q = self.questions[self.current_index]
        
        # Parse answer based on question type
        if q.question_type == "yes_no":
            if answer == "yes":
                self.answers[q.key] = True
            elif answer == "no":
                self.answers[q.key] = False
            elif answer == "skip":
                self.answers[q.key] = q.default
            else:
                # Text answer for yes_no (fallback)
                self.answers[q.key] = answer.lower() in ("да", "yes", "y", "1", "true")
        elif q.question_type == "number":
            try:
                self.answers[q.key] = int(answer.strip())
            except ValueError:
                self.answers[q.key] = q.default
        else:  # text
            self.answers[q.key] = answer.strip() or q.default
        
        # Move to next question
        self.current_index += 1
        return self._get_next_message()
    
    def is_complete(self) -> bool:
        """Check if interview is complete."""
        return self.current_index >= len(self.questions)
    
    def get_progress(self) -> tuple:
        """Get current progress (current, total)."""
        return (self.current_index, len(self.questions))

        """
        Save collected answers to profile.yaml.
        
        Returns True on success.
        """
        if not self.is_complete():

        # AI context
        profile["ai_context"] = {
            "# Читается AI при генерации open-ended ответов": "",
            "skills_dir": "data/candidate/skills/",
            "experience_dir": "data/candidate/experience/",
            "identity_file": "data/candidate/identity.md",
        }
        
        # Ensure directory exists
        self.profile_path.parent.mkdir(parents=True, exist_ok=True)
        
        # Write YAML
        try:
            # Custom representer for None (to make empty lines)
            def none_representer(dumper, data):
                return dumper.represent_scalar('tag:yaml.org,2002:null', '')
            
            yaml.add_representer(type(None), none_representer)
            
            with open(self.profile_path, 'w', encoding='utf-8') as f:
                yaml.dump(profile, f, default_flow_style=False, allow_unicode=True, sort_keys=False)
            
            return True
        except Exception as e:
            print(f"❌ Error saving profile: {e}")
            return False
    
    def load_existing_answers(self) -> bool:
        """
        Load existing profile to pre-populate answers.
        
        Returns True if profile exists and was loaded.
        """
        if not self.profile_path.exists():
            return False
        
        try:
            with open(self.profile_path, 'r', encoding='utf-8') as f:
                profile = yaml.safe_load(f)
            
            # Flatten profile into answers dict
            for section in ['personal', 'experience', 'certifications', 'work_conditions', 'salary']:
                if section in profile:
                    self.answers.update(profile[section])
            
            return True
        except Exception:
            return False
