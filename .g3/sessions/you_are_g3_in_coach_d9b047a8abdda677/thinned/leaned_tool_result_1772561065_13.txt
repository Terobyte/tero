Tool result: """
AI Answerer for Indeed form questions.

Primary: Gemini Flash (gemini-2.0-flash) - key in .env
Fallback: Claude Haiku - when ANTHROPIC_API_KEY is available
"""

import os
import json
import hashlib
import asyncio
from pathlib import Path
from typing import Optional, Dict, Any, List
from dataclasses import dataclass, asdict
from datetime import datetime
import logging

logger = logging.getLogger(__name__)

# Data directory
DATA_DIR = Path(__file__).parent.parent.parent / "data"
CANDIDATE_DIR = DATA_DIR / "candidate"


@dataclass
class AIAnswer:
    """AI-generated answer for a question."""
    answer: str
    confidence: str  # high, medium, low
    reason: str
    cached: bool = False
    provider: str = ""  # gemini, claude


class AIAnswerer:
    """
    Answers unknown form questions using AI.
    
    Primary: Gemini Flash (fast, cheap)
    Fallback: Claude Haiku (when Anthropic key available)
    
    Caches answers to avoid repeated API calls.
    """
    
    SYSTEM_PROMPT = """You help a job candidate fill out application forms on Indeed.

You are given:
1. The candidate's profile (YAML)
2. Their skills context (from markdown files)
3. A question from the application form
4. Available options (if multiple choice)

Rules:
1. Be HONEST - never invent skills or experience not in the profile
2. For yes/no questions - answer based on actual experience
3. For numeric questions - use data from the profile
4. For dropdown ranges - pick the most appropriate tier
5. For open-ended questions - draw from skills context if available
6. Default to "Yes" for comfort/behavioral questions unless clearly irrelevant
7. For certifications/licenses - answer "No" unless explicitly listed

Return ONLY valid JSON:
{"answer": "...", "confidence": "high|medium|low", "reason": "brief explanation"}"""

    def __init__(self):
        self.gemini_api_key = os.environ.get("GEMINI_API_KEY", "")
        self.anthropic_api_key = os.environ.get("ANTHROPIC_API_KEY", "")
        
        # Load profile
        self.profile = self._load_profile()
        self.skills_context = self._load_skills_context()
        
        # Cache file
        self.cache_file = DATA_DIR / "custom_answers.json"
        self.cache = self._load_cache()
        
        # Determine primary provider
        if self.gemini_api_key:
            self.primary_provider = "gemini"
        elif self.anthropic_api_key:
            self.primary_provider = "claude"
        else:
            self.primary_provider = None
            logger.warning("No AI API key found - AI answering disabled")
    
    def _load_profile(self) -> Dict[str, Any]:
        """Load candidate profile from YAML."""
        profile_file = CANDIDATE_DIR / "profile.yaml"
        if not profile_file.exists():
            logger.warning(f"Profile file not found: {profile_file}")
            return {}
        
        try:
            import yaml
            with open(profile_file, 'r', encoding='utf-8') as f:
                return yaml.safe_load(f) or {}
        except ImportError:
            logger.warning("PyYAML not installed - profile loading limited")
            return {}
        except Exception as e:
            logger.error(f"Error loading profile: {e}")
            return {}
    
    def _load_skills_context(self) -> str:
        """Load skills markdown files as context."""
        skills_dir = CANDIDATE_DIR / "skills"
        if not skills_dir.exists():
            return ""
        
        context_parts = []
        for md_file in skills_dir.glob("*.md"):
            try:
                content = md_file.read_text(encoding='utf-8')
                # Take first 500 chars of each skill file
                context_parts.append(f"\n### {md_file.stem.upper()}\n{content[:500]}")
            except Exception as e:
                logger.debug(f"Could not read {md_file}: {e}")
        
        return "\n".join(context_parts)
    
    def _load_cache(self) -> Dict[str, str]:
        """Load cached answers."""
        if not self.cache_file.exists():
            return {}
        
        try:
            with open(self.cache_file, 'r', encoding='utf-8') as f:
                data = json.load(f)
                return data.get("answers", {})
        except Exception as e:
            logger.debug(f"Could not load cache: {e}")
            return {}
    
    def _save_cache(self):
        """Save cached answers."""
        self.cache_file.parent.mkdir(parents=True, exist_ok=True)
        with open(self.cache_file, 'w', encoding='utf-8') as f:
            json.dump({"answers": self.cache}, f, indent=2)
    
    def _question_hash(self, question: str, options: List[str]) -> str:
        """Generate hash for question caching."""
        text = question.lower().strip()
        if options:
            text += "|" + "|".join(str(o) for o in options)
        return hashlib.md5(text.encode()).hexdigest()[:12]
    
    async def ask(
        self,
        question: str,
        field_type: str,
        options: Optional[List[str]] = None,
        context: Optional[Dict[str, Any]] = None
    ) -> Optional[AIAnswer]:
        """
        Ask AI to answer a question.
        
        Args:
            question: The question text
            field_type: radio, select, text, checkbox, textarea
            options: Available options (for radio/select)
            context: Additional context (job title, company, etc.)
            
        Returns:
            AIAnswer or None if unable to answer
        """
        # Check cache first
        q_hash = self._question_hash(question, options or [])
        if q_hash in self.cache:
            cached_answer = self.cache[q_hash]
            logger.info(f"Using cached answer for: {question[:50]}...")
            return AIAnswer(
                answer=cached_answer,
                confidence="high",
                reason="Cached from previous answer",
                cached=True,
                provider="cache"
            )
        
        # Check if AI is available
        if not self.primary_provider:
            logger.warning("No AI provider available")
            return None
        
        # Prepare context
        prompt = self._build_prompt(question, field_type, options, context)
        
        # Try primary provider
        answer = None
        if self.primary_provider == "gemini":
            answer = await self._ask_gemini(prompt)
            if answer:
                answer.provider = "gemini"
        elif self.primary_provider == "claude":
            answer = await self._ask_claude(prompt)
            if answer:
                answer.provider = "claude"
        
        # Fallback to other provider
        if not answer:
            if self.primary_provider == "gemini" and self.anthropic_api_key:
                answer = await self._ask_claude(prompt)
                if answer:
                    answer.provider = "claude"
            elif self.primary_provider == "claude" and self.gemini_api_key:
                answer = await self._ask_gemini(prompt)
                if answer:
                    answer.provider = "gemini"
        
        # Cache successful answer
        if answer:
            self.cache[q_hash] = answer.answer
            self._save_cache()
        
        return answer
    
    def _build_prompt(
        self,
        question: str,
        field_type: str,
        options: Optional[List[str]],
        context: Optional[Dict[str, Any]]
    ) -> str:
        """Build the prompt for AI."""
        # Convert profile to YAML string
        import yaml
        profile_str = yaml.dump(self.profile, default_flow_style=False)
        
        prompt_parts = [
            f"CANDIDATE PROFILE:\n```yaml\n{profile_str}\n```",
        ]
        
        if self.skills_context:
            prompt_parts.append(f"\nSKILLS CONTEXT:\n{self.skills_context}")
        
        if context:
            prompt_parts.append(f"\nAPPLICATION CONTEXT:")
            prompt_parts.append(f"Job Title: {context.get('job_title', 'Unknown')}")
            prompt_parts.append(f"Company: {context.get('company', 'Unknown')}")
        
        prompt_parts.append(f"\nQUESTION:\n{question}")
        prompt_parts.append(f"\nFIELD TYPE: {field_type}")
        
        if options:
            prompt_parts.append(f"\nOPTIONS:\n" + "\n".join(f"- {o}" for o in options))
        
        prompt_parts.append("\n\nReturn ONLY valid JSON:")
        prompt_parts.append('{"answer": "...", "confidence": "high|medium|low", "reason": "..."}')
        
        return "\n".join(prompt_parts)
    
    async def _ask_gemini(self, prompt: str) -> Optional[AIAnswer]:
        """Ask Gemini Flash API."""
        if not self.gemini_api_key:
            return None
        
        try:
            import httpx
            
            url = f"https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key={self.gemini_api_key}"
            
            payload = {
                "contents": [{
                    "parts": [{
                        "text": self.SYSTEM_PROMPT + "\n\n" + prompt
                    }]
                }],
                "generationConfig": {
                    "temperature": 0.3,
                    "maxOutputTokens": 500
                }
            }
            
            async with httpx.AsyncClient(timeout=30.0) as client:
                response = await client.post(url, json=payload)
                response.raise_for_status()
                data = response.json()
            
            # Extract text from Gemini response
            text = data.get("candidates", [{}])[0].get("content", {}).get("parts", [{}])[0].get("text", "")
            
            # Parse JSON response
            return self._parse_ai_response(text)
            
        except Exception as e:
            logger.error(f"Gemini API error: {e}")
            return None
    
    async def _ask_claude(self, prompt: str) -> Optional[AIAnswer]:
        """Ask Claude Haiku API."""
        if not self.anthropic_api_key:
            return None
        
        try:
            import httpx
            
            url = "https://api.anthropic.com/v1/messages"
            
            headers = {
                "x-api-key": self.anthropic_api_key,
                "anthropic-version": "2023-06-01",
                "content-type": "application/json"
            }
            
            payload = {
                "model": "claude-3-haiku-20240307",  # Fast, cheap
                "max_tokens": 500,
                "system": self.SYSTEM_PROMPT,
                "messages": [{
                    "role": "user",
                    "content": prompt
                }]
            }
            
            async with httpx.AsyncClient(timeout=30.0) as client:
                response = await client.post(url, json=payload, headers=headers)
                response.raise_for_status()
                data = response.json()
            
            # Extract text from Claude response
            text = data.get("content", [{}])[0].get("text", "")
            
            # Parse JSON response
            return self._parse_ai_response(text)
            
        except Exception as e:
            logger.error(f"Claude API error: {e}")
            return None
    
    def _parse_ai_response(self, text: str) -> Optional[AIAnswer]:
        """Parse AI JSON response."""
        # Try to extract JSON from response
        text = text.strip()
        
        # Remove markdown code blocks if present
        if text.startswith("```"):
            lines = text.split("\n")
            text = "\n".join(lines[1:-1] if lines[-1] == "```" else lines[1:])
        
        try:
            data = json.loads(text)
            return AIAnswer(
                answer=str(data.get("answer", "")),
                confidence=data.get("confidence", "medium"),
                reason=data.get("reason", "")
            )
        except json.JSONDecodeError:
            # Try to extract answer from text
            logger.debug(f"Could not parse AI response as JSON: {text[:100]}")
            
            # Simple fallback: look for yes/no patterns
            text_lower = text.lower()
            if '"yes"' in text_lower or "'yes'" in text_lower or "answer: yes" in text_lower:
                return AIAnswer(answer="Yes", confidence="low", reason="Extracted from text")
            elif '"no"' in text_lower or "'no'" in text_lower or "answer: no" in text_lower:
                return AIAnswer(answer="No", confidence="low", reason="Extracted from text")
            
            return None


# Convenience function
async def ask_ai(
    question: str,
    field_type: str,
    options: Optional[List[str]] = None,
    context: Optional[Dict[str, Any]] = None
) -> Optional[AIAnswer]:
    """
    Convenience function to ask AI a question.
    
    Args:
        question: The question text
        field_type: radio, select, text, checkbox, textarea
        options: Available options (for radio/select)
        context: Additional context
        
    Returns:
        AIAnswer or None
    """
    answerer = AIAnswerer()
    return await answerer.ask(question, field_type, options, context)
