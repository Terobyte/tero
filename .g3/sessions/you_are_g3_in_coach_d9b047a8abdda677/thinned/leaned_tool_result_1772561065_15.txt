Tool result: """
AnswerBank - Pattern-based question answering for job application forms.

Provides a 3-level fallback system:
1. Extended keyword patterns (fast, free)
2. Custom answers from profile.yaml (user-curated)
3. AI fallback for unknown questions

This module handles the majority (~95%) of standard screening questions
without needing AI assistance.
"""

import re

@dataclass
class QuestionMatch:
    """Result of matching a question to an answer."""
    answer: str
    source: str  # "pattern", "profile", "custom", "ai"
    confidence: str  # "high", "medium", "low"
    profile_key: Optional[str] = None

        # Returns "Yes" or "No" based on profile

    def __init__(self, profile_path: str = "data/candidate/profile.yaml",
                 custom_answers_path: str = "data/custom_answers.json"):
        """Initialize AnswerBank with profile and custom answers."""
        self.profile: Dict[str, Any] = {}
        self.custom_answers: Dict[str, str] = {}
        self.profile_path = Path(profile_path)
        self.custom_answers_path = Path(custom_answers_path)
        
        self._load_profile()
        self._load_custom_answers()
        
    def _load_profile(self) -> None:
        """Load user profile from YAML file."""
        if self.profile_path.exists():
            try:
                with open(self.profile_path, 'r', encoding='utf-8') as f:
                    self.profile = yaml.safe_load(f) or {}
                print(f"[AnswerBank] Loaded profile from {self.profile_path}")
            except Exception as e:
                print(f"[AnswerBank] Warning: Could not load profile: {e}")
                self.profile = {}
        else:
            print(f"[AnswerBank] Profile not found at {self.profile_path}, using defaults")
            self.profile = {}
            
    def _load_custom_answers(self) -> None:
        """Load custom answers from JSON file."""
        if self.custom_answers_path.exists():
            try:
                with open(self.custom_answers_path, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    self.custom_answers = data.get("answers", {})
                print(f"[AnswerBank] Loaded {len(self.custom_answers)} custom answers")
            except Exception as e:
                print(f"[AnswerBank] Warning: Could not load custom answers: {e}")
                self.custom_answers = {}
                
    def _get_profile_value(self, key: str) -> Optional[Any]:
        """
        Get a value from profile using dot notation.
        
        Supports nested keys like 'experience.years_python' or flat keys.
        """
        # Try flat key first
        if key in self.profile:
            return self.profile[key]
            
        # Try nested sections
        for section in ['experience', 'certifications', 'work_conditions', 'work_auth', 
                        'salary', 'availability', 'personal']:
            if section in self.profile:
                if key in self.profile[section]:
                    return self.profile[section][key]
                # Try with/without 'has_' prefix
                if f"has_{key}" in self.profile[section]:
                    return self.profile[section][f"has_{key}"]
                if key.startswith("has_") and key[4:] in self.profile[section]:
                    return self.profile[section][key[4:]]
                    
        return None
    
    def answer(self, question: str, options: Optional[List[str]] = None) -> Optional[QuestionMatch]:
        """
        Find an answer for a screening question.
        
        Args:
            question: The question text
            options: Available answer options (for radio/select fields)
            
        Returns:
            QuestionMatch if found, None otherwise
        """
        question_lower = question.lower().strip()
        
        # Level 1: Try keyword patterns
        for pattern, profile_key in self.KEYWORD_PATTERNS.items():
            if re.search(pattern, question_lower, re.IGNORECASE):
                value = self._get_profile_value(profile_key)
                if value is not None:
                    answer = self._format_answer(value, profile_key, options)
                    if answer:
                        return QuestionMatch(
                            answer=answer,
                            source="pattern",
                            confidence="high",
                            profile_key=profile_key
                        )
        
        # Level 2: Try custom answers (AI cache)
        # Try exact match first
        if question in self.custom_answers:
            return QuestionMatch(
                answer=self.custom_answers[question],
                source="custom",
                confidence="high"
            )
            
        # Try fuzzy match in custom answers
        for cached_q, cached_a in self.custom_answers.items():
            if self._questions_similar(question, cached_q):
                return QuestionMatch(
                    answer=cached_a,
                    source="custom",
                    confidence="medium"
                )
        
        return None
    
    def _format_answer(self, value: Any, profile_key: str, options: Optional[List[str]]) -> Optional[str]:
        """
        Format profile value to match available options.
        
        Args:
            value: The value from profile (bool, int, str)
            profile_key: The key name for context
            options: Available answer options
            
        Returns:
            Formatted answer string or None
        """
        if value is None:
            return None
            
        # Handle boolean values
        if isinstance(value, bool):
            bool_answer = "Yes" if value else "No"
            if options:
                # Find matching option (case-insensitive)
                for opt in options:
                    if opt.lower() == bool_answer.lower():
                        return opt
                # Try alternative formats
                for opt in options:
                    if opt.lower() in ["yes", "no"]:
                        return bool_answer
            return bool_answer
            
        # Handle numeric values (years of experience)
        if isinstance(value, (int, float)):
            return self._format_years_answer(value, options)
            
        # Handle string values
        if isinstance(value, str):
            if options:
                # Find matching option
                for opt in options:
                    if opt.lower() == value.lower():
                        return opt
            return value
            
        return str(value)
    
    def _format_years_answer(self, years: float, options: Optional[List[str]]) -> str:
        """
        Format years of experience to match dropdown options.
        
        Strategy: Round UP to next tier (aggressive approach per requirements).
        0 years → "Less than 1 year" (never "No experience")
        
        Args:
            years: Years of experience
            options: Available dropdown options
            
        Returns:
            Best matching option or formatted string
        """
        if options:
            # Common dropdown patterns
            for opt in options:
                opt_lower = opt.lower()
                
                # 0 years → "Less than 1 year" or "0-1 years"
                if years == 0:
                    if "less than" in opt_lower or "0-1" in opt_lower or "< 1" in opt_lower:
                        return opt
                    if "no experience" in opt_lower:
                        continue  # Skip "no experience" - use "less than 1 year"
                        
                # Match ranges
                import re
                range_match = re.search(r'(\d+)\s*[-–]\s*(\d+)', opt)
                if range_match:
                    low, high = int(range_match.group(1)), int(range_match.group(2))
                    if low <= years <= high:
                        return opt
                    # Round UP: if years > high, check if this is the best option
                    
                # "X+ years" pattern
                plus_match = re.search(r'(\d+)\s*\+\s*years?', opt_lower)
                if plus_match:
                    min_years = int(plus_match.group(1))
                    if years >= min_years:
                        return opt
                        
                # "More than X years" pattern
                more_match = re.search(r'more\s*than\s*(\d+)', opt_lower)
                if more_match:
                    min_years = int(more_match.group(1))
                    if years > min_years:
                        return opt
                        
            # Fallback: find best matching option
            # For 0 years, find "Less than 1 year" or first option
            if years == 0:
                for opt in options:
                    if "less" in opt.lower() or "0" in opt:
                        return opt
                # Return first non-empty option as fallback
                for opt in options:
                    if opt.strip():
                        return opt
                        
            # Round UP: find option where years >= low end of range
            for opt in options:
                range_match = re.search(r'(\d+)\s*[-–]', opt)
                if range_match:
                    low = int(range_match.group(1))
                    if years >= low:
                        return opt
                        
        # No options provided - format as string
        if years == 0:
            return "Less than 1 year"
        return f"{years} years"
    
    def _questions_similar(self, q1: str, q2: str) -> bool:
        """Check if two questions are similar enough to share an answer."""
        # Normalize both questions
        q1_norm = re.sub(r'[^\w\s]', '', q1.lower()).strip()
        q2_norm = re.sub(r'[^\w\s]', '', q2.lower()).strip()
        
        # Exact match after normalization
        if q1_norm == q2_norm:
            return True
            
        # Check word overlap (at least 70% same words)
        words1 = set(q1_norm.split())
        words2 = set(q2_norm.split())
        
        if not words1 or not words2:
            return False
            
        intersection = words1 & words2
        union = words1 | words2
        jaccard = len(intersection) / len(union)
        
        return jaccard >= 0.7
    
    def add_custom_answer(self, question: str, answer: str) -> None:
        """Add a custom answer and save to file."""
        self.custom_answers[question] = answer
        self._save_custom_answers()
        
    def _save_custom_answers(self) -> None:
        """Save custom answers to JSON file."""
        self.custom_answers_path.parent.mkdir(parents=True, exist_ok=True)
        with open(self.custom_answers_path, 'w', encoding='utf-8') as f:
            json.dump({"answers": self.custom_answers}, f, indent=2, ensure_ascii=False)
            
    def get_profile_context(self) -> str:
        """Get profile context for AI prompts."""
        if not self.profile:
            return "No profile loaded."
            
        context_parts = []
        
        # Personal info
        if 'personal' in self.profile:
            p = self.profile['personal']
            context_parts.append(f"Name: {p.get('first_name', '')} {p.get('last_name', '')}")
            
        # Experience summary
        if 'experience' in self.profile:
            exp = self.profile['experience']
            years = [k for k in exp.keys() if k.startswith('years_') and exp[k] > 0]
            if years:
                skills = [k.replace('years_', '').replace('_', ' ').title() for k in years]
                context_parts.append(f"Skills with experience: {', '.join(skills)}")
                
        # Certifications
        if 'certifications' in self.profile:
            certs = self.profile['certifications']
            has_certs = [k for k, v in certs.items() if v and k.startswith('has_')]
            if has_certs:
                cert_names = [k.replace('has_', '').replace('_', ' ').title() for k in has_certs]
                context_parts.append(f"Certifications: {', '.join(cert_names)}")
                
        # Work conditions
        if 'work_conditions' in self.profile:
            wc = self.profile['work_conditions']
            conditions = []
            if wc.get('prefer_remote'): conditions.append("prefers remote")
            if wc.get('accept_hybrid'): conditions.append("accepts hybrid")
            if wc.get('work_weekends'): conditions.append("can work weekends")
            if wc.get('work_overtime'): conditions.append("can work overtime")
            if conditions:
                context_parts.append(f"Work preferences: {', '.join(conditions)}")
                
        return "\n".join(context_parts)


# Convenience function
def answer_question(question: str, options: Optional[List[str]] = None) -> Optional[str]:
    """Quick helper to answer a single question."""
    bank = AnswerBank()
    result = bank.answer(question, options)
    return result.answer if result else None
