Tool result: """
Indeed Easy Apply handler with AI-powered screening question answering.

This module handles Indeed's native Easy Apply forms, including:
- Standard fields (phone, cover letter)
- Screening questions via 3-tier answering system:
  1. AnswerBank (pattern-based)
  2. CustomAnswers (user-curated)
  3. AI Fallback (Gemini Flash)
"""

import asyncio
import os
import re
from typing import Dict, Any, Optional, List
from pathlib import Path
from datetime import datetime
import sys

sys.path.insert(0, str(Path(__file__).parent.parent.parent))

from src.applier.answer_bank import AnswerBank
from src.applier.indeed_screening import IndeedScreeningHandler
from src.applier.ai_answerer import AIAnswerer


# Indeed Easy Apply selectors
IA = {
    "apply_button": "button:has-text('Apply now'), button:has-text('Apply'), .jobsearch-IndeedApplyButton-newIndeedApplyButton",
    "continue_button": "button:has-text('Continue'), button:has-text('Submit'), button[type='submit']",
    "phone": "input[name='phone'], input[id*='phone'], input[placeholder*='phone']",
    "cover_letter": "textarea[name='coverLetter'], textarea[id*='cover'], textarea[placeholder*='cover']",
    "name": "input[name='name'], input[id*='name'], input[placeholder*='name']",
    "email": "input[name='email'], input[type='email']",
    "form_step": "[data-testid='apply-form-step'], .ia-Questions-item, form",
    "radio_group": "fieldset",
    "dropdown": "select",
    "text_input": "input[type='text'], input[type='number']",
    "checkbox": "input[type='checkbox']",
    "textarea": "textarea",
    "success_message": "text=Application submitted, text=Your application has been sent",
    "error_message": ".error, [role='alert']",
}


class IndeedApplier:
    """
    Handles Indeed Easy Apply forms with AI-powered question answering.
    
    Flow:
    1. Click "Apply now" button
    2. Fill standard fields (phone, name, email)
    3. Detect and answer screening questions via 3-tier system
    4. Navigate through form steps until submission
    5. Take screenshot proof
    """
    
    def __init__(
        self,
        headless: bool = True,
        profile_path: str = "data/candidate/profile.yaml",
        custom_answers_path: str = "data/custom_answers.json",
        unknown_log_path: str = "data/unknown_questions.json",
        use_ai: bool = True
    ):
        """
        Initialize Indeed Applier.
        
        Args:
            headless: Run browser in headless mode
            profile_path: Path to profile.yaml
            custom_answers_path: Path to custom_answers.json
            unknown_log_path: Path to unknown_questions.json log
            use_ai: Enable AI fallback for unknown questions
        """
        self.headless = headless
        
        # Initialize answer bank with profile
        self.answer_bank = AnswerBank(profile_path=profile_path)
        self.answer_bank.load_custom_answers(custom_answers_path)
        
        # Initialize AI answerer (Gemini primary, Claude fallback)
        self.ai_answerer = AIAnswerer(profile_path=profile_path) if use_ai else None
        
        # Initialize screening handler
        self.screening = IndeedScreeningHandler(
            answer_bank=self.answer_bank,
            ai_answerer=self.ai_answerer,
            unknown_log_path=unknown_log_path
        )
        
        self.screenshot_dir = Path("data/screenshots")
        self.screenshot_dir.mkdir(parents=True, exist_ok=True)
    
    async def apply(
        self,
        job_url: str,
        job_id: str,
        company: str,
        title: str,
        user_data: Optional[Dict[str, Any]] = None
    ) -> Dict[str, Any]:
        """
        Apply to an Indeed Easy Apply job.
        
        Args:
            job_url: URL of the job posting
            job_id: Indeed job ID
            company: Company name
            title: Job title
            user_data: Optional user data override
            
        Returns:
            dict with keys: success, screenshot_path, error, questions_answered
        """
        result = {
            "success": False,
            "screenshot_path": None,
            "error": None,
            "questions_answered": 0,
            "ai_questions": 0,
            "unknown_questions": []
        }
        
        try:
            from patchright.async_api import async_playwright
            
            async with async_playwright() as p:
                browser = await p.chromium.launch(headless=self.headless)
                context = await browser.new_context()
                page = await context.new_page()
                
                # Navigate to job posting
                print(f"🌐 Navigating to: {job_url}")
                await page.goto(job_url, wait_until="domcontentloaded", timeout=60000)
                await asyncio.sleep(2.0)
                
                # Click Apply button
                apply_clicked = await self._click_apply_button(page)
                if not apply_clicked:
                    # Maybe it's already on apply page
                    if "smartapply" not in page.url:
                        result["error"] = "Could not find Apply button"
                        await browser.close()
                        return result
                
                # Wait for form to load
                await asyncio.sleep(2.0)
                
                # Fill standard fields
                await self._fill_standard_fields(page, user_data or {})
                
                # Process all form steps
                questions_answered, ai_questions, unknown = await self._process_form_steps(
                    page, job_url, company, title
                )
                result["questions_answered"] = questions_answered
                result["ai_questions"] = ai_questions
                result["unknown_questions"] = unknown
                
                # Take screenshot
                screenshot_path = self.screenshot_dir / f"indeed_{job_id}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.png"
                await page.screenshot(path=str(screenshot_path), full_page=True)
                result["screenshot_path"] = str(screenshot_path)
                
                # Check for success
                success = await self._check_success(page)
                result["success"] = success
                
                await browser.close()
                
        except Exception as e:
            result["error"] = str(e)
            import traceback
            traceback.print_exc()
        
        return result
    
    async def _click_apply_button(self, page) -> bool:
        """Click the Apply now button."""
        try:
            apply_btn = page.locator(IA["apply_button"]).first
            if await apply_btn.is_visible(timeout=5000):
                await apply_btn.click()
                await asyncio.sleep(2.0)
                return True
        except Exception:
            pass
        return False
    
    async def _fill_standard_fields(self, page, user_data: Dict[str, Any]):
        """Fill standard form fields."""
        # Phone
        phone = user_data.get("phone") or os.environ.get("APPLICANT_PHONE", "")
        if phone:
            try:
                phone_input = page.locator(IA["phone"]).first
                if await phone_input.is_visible(timeout=3000):
                    await phone_input.fill(phone)
                    print(f"📝 Filled phone: {phone}")
            except Exception:
                pass
        
        # Name
        name = user_data.get("name") or os.environ.get("APPLICANT_NAME", "")
        if name:
            try:
                name_input = page.locator(IA["name"]).first
                if await name_input.is_visible(timeout=3000):
                    await name_input.fill(name)
                    print(f"📝 Filled name: {name}")
            except Exception:
                pass
        
        # Email
        email = user_data.get("email") or os.environ.get("APPLICANT_EMAIL", "")
        if email:
            try:
                email_input = page.locator(IA["email"]).first
                if await email_input.is_visible(timeout=3000):
                    await email_input.fill(email)
                    print(f"📝 Filled email: {email}")
            except Exception:
                pass
    
    async def _process_form_steps(
        self,
        page,
        job_url: str,
        company: str,
        title: str
    ) -> tuple[int, int, list]:
        """
        Process all form steps with screening question handling.
        
        Returns:
            (questions_answered, ai_questions, unknown_questions)
        """
        total_answered = 0
        total_ai = 0
        all_unknown = []
        
        max_steps = 10  # Prevent infinite loops
        
        for step in range(max_steps):
            print(f"\n📋 Processing form step {step + 1}")
            
            # Use screening handler to process all questions on this step
            try:
                answered, ai_count, unknown = await self.screening.handle_all_questions(
                    page=page,
                    job_url=job_url,
                    company=company,
                    job_title=title
                )
                total_answered += answered
                total_ai += ai_count
                all_unknown.extend(unknown)
                
            except Exception as e:
                print(f"⚠️ Error in screening handler: {e}")
            
            # Try to continue/submit
            continued = await self._click_continue(page)
            
            if not continued:
                # Maybe we're done?
                break
            
            await asyncio.sleep(1.5)
        
        return total_answered, total_ai, all_unknown
    
    async def _click_continue(self, page) -> bool:
        """Click Continue or Submit button."""
        try:
            # Try multiple selectors
            selectors = [
                "button:has-text('Continue')",
                "button:has-text('Continue applying')",
                "button:has-text('Submit')",
                "button[type='submit']",
                "button.ia-ContinueButton",
                "button[data-testid='continue-button']"
            ]
            
            for selector in selectors:
                try:
                    btn = page.locator(selector).first
                    if await btn.is_visible(timeout=2000):
                        await btn.click()
                        print(f"➡️ Clicked: {selector}")
                        return True
                except Exception:
                    continue
                    
        except Exception as e:
            print(f"⚠️ Could not click continue: {e}")
        
        return False
    
    async def _check_success(self, page) -> bool:
        """Check if application was successful."""
        try:
            # Check for success indicators
            success_selectors = [
                "[data-testid='application-success']",
                "text=Application submitted",
                "text=Your application has been sent",
                "text=Application sent",
                ".ia-ApplicationSubmitted",
            ]
            
            for selector in success_selectors:
                try:
                    el = page.locator(selector).first
                    if await el.is_visible(timeout=3000):
                        print("✅ Application submitted successfully!")
                        return True
                except Exception:
                    continue
            
            # Check URL for success indicators
            if "application-submitted" in page.url or "success" in page.url.lower():
                return True
                
        except Exception:
            pass
        
        return False


async def apply_to_indeed(
    job: Dict[str, Any],
    user_profile: Optional[Dict[str, Any]] = None,
    headless: bool = True
) -> Dict[str, Any]:
    """
    Convenience function to apply to Indeed Easy Apply job.
    
    Args:
        job: Job dict with keys: id, title, company, url
        user_profile: User profile dict
        headless: Run browser in headless mode
        
    Returns:
        Application result dict
    """
    applier = IndeedApplier(headless=headless)
    return await applier.apply(
        job_url=job.get("url", ""),
        job_id=job.get("id", ""),
        company=job.get("company", ""),
        title=job.get("title", ""),
        user_data=user_profile
    )
