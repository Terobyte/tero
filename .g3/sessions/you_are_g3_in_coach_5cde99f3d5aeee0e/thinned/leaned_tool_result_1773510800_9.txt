"""
StepSnapshot — снэпшот DOM-состояния формы перед/после fill.
Необходим для re-fill после коррекции.
"""
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Optional


@dataclass
class QuestionSnapshot:
    """Снэпшот одного вопроса: ключи для последующего re-fill."""
    text: str
    field_type: str
    options: list[str]
    given_answer: str
    locator_info: dict[str, Any] = field(default_factory=dict)
    # Хранит: {"selector": "...", "nth": 0} для повторного поиска элемента
    corrected_answer: Optional[str] = None


@dataclass
class StepSnapshot:
    """Снэпшот всего шага формы — список вопросов с локаторами."""
    step: int
    questions: list[QuestionSnapshot] = field(default_factory=list)
    job_url: str = ""

    def add(self, text: str, field_type: str,
            options: list[str], answer: str,
            locator_info: dict | None = None):
        self.questions.append(QuestionSnapshot(
            text=text, field_type=field_type,
            options=options, given_answer=answer,
            locator_info=locator_info or {},
        ))

    def get(self, question_text: str) -> Optional[QuestionSnapshot]:
        for q in self.questions:
            if q.text == question_text:
                return q
        return None

    def apply_correction(self, question_text: str, corrected: str) -> bool:
        q = self.get(question_text)
        if q:
            q.corrected_answer = corrected
            return True
        return False
