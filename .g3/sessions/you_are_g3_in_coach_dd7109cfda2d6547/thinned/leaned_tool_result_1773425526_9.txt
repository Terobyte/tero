class TestWizardEngineIntegration:
    """Integration tests for WizardEngine."""

    def test_step_tracker_records_steps(self):
        """Test that StepTracker records and detects loops."""
        tracker = StepTracker(max_steps=5)
        
        # Record first step
        tracker.record_step("https://example.com/step1", "hash1", "screen1")
        assert len(tracker._steps) == 1
        
        # Record second step
        tracker.record_step("https://example.com/step2", "hash2", "screen2")
        assert len(tracker._steps) == 2
        
        # Same DOM hash 2 times should trigger stuck error on 3rd
        tracker.record_step("https://example.com/step3", "hash2", "screen3")
        with pytest.raises(WizardStuckError):
            tracker.record_step("https://example.com/step4", "hash2", "screen4")

    def test_step_tracker_max_steps_limit(self):
        """Test that StepTracker enforces max steps."""
        tracker = StepTracker(max_steps=3)
        
        tracker.record_step("url1", "hash1", "screen1")
        tracker.record_step("url2", "hash2", "screen2")
        tracker.record_step("url3", "hash3", "screen3")
        
        with pytest.raises(WizardMaxStepsError):
            tracker.record_step("url4", "hash4", "screen4")

    def test_step_tracker_get_diff(self):
        """Test getting diff between steps."""
        tracker = StepTracker()
        
        tracker.record_step("url1", "hash1", "screen1")
        tracker.record_step("url2", "hash2", "screen2")
        
        diff = tracker.get_diff(0, 1)
        assert diff is not None

    def test_next_button_finder_patterns(self):
        """Test NextButtonFinder pattern discovery."""
        finder = NextButtonFinder()
        
        # Should have patterns defined
        assert len(finder.NEXT_BUTTON_PATTERNS) > 0
        assert len(finder.SUBMIT_BUTTON_PATTERNS) > 0
        
        # Check priority order
        assert "button:has-text('Next')" in finder.NEXT_BUTTON_PATTERNS
        assert "button:has-text('Submit')" in finder.SUBMIT_BUTTON_PATTERNS

    @pytest.mark.asyncio
    async def test_wizard_engine_init(self):

        engine = WizardEngine(max_steps=10, step_timeout=30)
        
        assert engine.max_steps == 10

        assert engine.step_tracker is not None
