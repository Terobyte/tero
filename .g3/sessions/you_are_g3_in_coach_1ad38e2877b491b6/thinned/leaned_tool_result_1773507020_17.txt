"""
GapRecommender — рекомендации как закрыть пробелы в опыте.

Input: GapAnalysis.gaps (список требований JD которых нет в PKB)
Output: Конкретный план действий для каждого пробела (курсы, проекты, сертификаты)
"""
from __future__ import annotations

import json
from dataclasses import dataclass
from typing import Optional

from loguru import logger

from src.ai.gap_analyser import GapAnalysis


@dataclass
class GapRecommendation:
    """Рекомендация для одного пробела."""
    gap: str                    # Требование из JD (чего нет в профиле)
    priority: str               # "critical", "important", "nice_to_have"
    time_to_close: str          # "1 week", "1 month", "3 months"
    recommendations: list[str]  # Конкретные шаги
    resources: list[str]        # Ссылки на курсы/документацию


@dataclass
class GapReport:
    """Полный отчёт по пробелам."""
    job_title: str
    company: str
    total_gaps: int
    critical_gaps: list[GapRecommendation]
    important_gaps: list[GapRecommendation]
    nice_to_have_gaps: list[GapRecommendation]

    @property
    def match_percent(self) -> float:
        """Процент соответствия (без критических пробелов)."""
        if self.total_gaps == 0:
            return 100.0
        critical = len(self.critical_gaps)
        return round((1 - critical / self.total_gaps) * 100, 1)

    def to_telegram_text(self) -> str:
        """Форматировать для Telegram уведомления."""
        lines = [
            f"📊 Gap Analysis: {self.job_title} @ {self.company}",
            f"Match: {self.match_percent}% | Gaps: {self.total_gaps}",
            "",
        ]
        if self.critical_gaps:
            lines.append("🔴 Critical:")
            for g in self.critical_gaps[:3]:
                lines.append(f"  • {g.gap}: {g.recommendations[0] if g.recommendations else ''}")
        return "\n".join(lines)


# Промпт для LLM
_PROMPT = """You are a career coach. Analyze skill gaps and provide actionable recommendations.

Job: {job_title} at {company}
Skill gaps (requirements NOT in candidate profile):
{gaps_list}

For EACH gap, provide:
1. Priority: critical/important/nice_to_have
2. Time to close: realistic estimate
3. 2-3 specific action steps
4. 1-2 learning resources (course names, not URLs)

Return JSON only:
{{"gaps": [{{"gap": "...", "priority": "critical|important|nice_to_have",
  "time_to_close": "1-2 weeks", "recommendations": ["step1", "step2"],
  "resources": ["resource1"]}}]}}"""


class GapRecommender:
    """
    Генерирует рекомендации по закрытию пробелов в опыте.

    Usage:
        recommender = GapRecommender(gemini_client)
        report = await recommender.recommend(gap_analysis, job)
    """

    def __init__(self, llm_client):
        """
        Args:
            llm_client: любой клиент с методом generate(prompt) → str
                       (GeminiClient, KimiClient, etc.)
        """
        self._llm = llm_client

    async def recommend(
        self,
        gap_analysis: GapAnalysis,
        job: dict,
        max_gaps: int = 10,
    ) -> GapReport:
        """
        Сгенерировать рекомендации для всех пробелов.

        Args:
            gap_analysis: Результат GapAnalyser.analyse()
            job: Словарь вакансии {title, company, ...}
            max_gaps: Максимум пробелов для анализа (экономия токенов)
        """
        all_gaps = gap_analysis.gaps[:max_gaps]
        job_title = job.get("title", "Unknown Role")
        company = job.get("company", "Unknown Company")

        if not all_gaps:
            return GapReport(
                job_title=job_title, company=company,
                total_gaps=0,
                critical_gaps=[], important_gaps=[], nice_to_have_gaps=[],
            )

        gaps_list = "\n".join(f"- {g}" for g in all_gaps)
        prompt = _PROMPT.format(
            job_title=job_title, company=company, gaps_list=gaps_list
        )

        recommendations = []
        try:
            raw = await self._llm.generate(prompt)
            raw = raw.strip()
            if "```json" in raw:
                raw = raw.split("```json")[1].split("```")[0]
            elif "```" in raw:
                raw = raw.split("```")[1].split("```")[0]

            data = json.loads(raw)
            for item in data.get("gaps", []):
                recommendations.append(GapRecommendation(
                    gap=item.get("gap", ""),
                    priority=item.get("priority", "nice_to_have"),
                    time_to_close=item.get("time_to_close", "1 month"),
                    recommendations=item.get("recommendations", []),
                    resources=item.get("resources", []),
                ))
        except Exception as e:
            logger.error(f"GapRecommender LLM error: {e}")
            # Fallback: базовые рекомендации без LLM
            recommendations = [
                GapRecommendation(
                    gap=g, priority="important",
                    time_to_close="1-2 weeks",
                    recommendations=[f"Study {g} fundamentals", f"Build a project using {g}"],
                    resources=[f"{g} documentation", f"Udemy: {g} course"],
                )
                for g in all_gaps[:5]
            ]

        critical = [r for r in recommendations if r.priority == "critical"]
        important = [r for r in recommendations if r.priority == "important"]
        nice = [r for r in recommendations if r.priority == "nice_to_have"]

        report = GapReport(
            job_title=job_title, company=company,
            total_gaps=len(all_gaps),
            critical_gaps=critical,
            important_gaps=important,
            nice_to_have_gaps=nice,
        )
        logger.info(
            f"📊 GapReport: {job_title} @ {company} | "
            f"critical={len(critical)} important={len(important)} nice={len(nice)}"
        )
        return report

    async def quick_score(self, gap_analysis: GapAnalysis) -> float:
        """
        Быстрая оценка match rate без LLM (0.0-1.0).
        Используется для фильтрации перед дорогим LLM вызовом.
        """
        total = len(gap_analysis.matched) + len(gap_analysis.adjacent) + len(gap_analysis.gaps)
        if total == 0:
            return 1.0
        # matched=100%, adjacent=50%, gaps=0%
        score = (len(gap_analysis.matched) + len(gap_analysis.adjacent) * 0.5) / total
        return round(score, 3)
