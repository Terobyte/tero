Tool result: """
Kimi Vision Client - Kimi 2.5 with vision capabilities.

Extends KimiClient to support image+text prompts for visual analysis.
"""

import base64
import logging
from dataclasses import dataclass
from typing import Any

from openai import AsyncOpenAI

logger = logging.getLogger(__name__)


@dataclass
class VisionResponse:
    """Response from vision analysis."""
    text: str
    parsed_json: dict | None = None
    confidence: float = 0.0
    model: str = ""
    tokens_used: int = 0


class KimiVisionClient:
    """
    Kimi 2.5 with vision capabilities.
    
    Supports image+text prompts for visual analysis of screenshots.
    Used for:
    - Site reconnaissance (page type detection)
    - Button/element location (bounding box coordinates)
    - Form verification (empty fields, errors)
    - Outcome detection (success/failure pages)
    
    Model: blackboxai/moonshotai/kimi-k2.5
    """
    
    BASE_URL = "https://api.blackbox.ai"
    MODEL = "blackboxai/moonshotai/kimi-k2.5"
    
    def __init__(
        self,
        api_key: str | None = None,
        max_tokens: int = 2048,
        temperature: float = 0.3,
        timeout: float = 60.0,
    ):
        """
        Initialize Kimi Vision client.
        
        Args:
            api_key: Blackbox API key (defaults to env KIMI_API_KEY or BLACKBOX_API_KEY)
            max_tokens: Maximum tokens in response
            temperature: Sampling temperature (lower = more deterministic)
            timeout: Request timeout in seconds
        """
        import os
        
        self.api_key = api_key or os.getenv("KIMI_API_KEY") or os.getenv("BLACKBOX_API_KEY")
        if not self.api_key:
            logger.warning("No Kimi API key found - vision calls will fail")
        
        self.max_tokens = max_tokens
        self.temperature = temperature
        self.timeout = timeout
        
        self._client: AsyncOpenAI | None = None
    
    async def _get_client(self) -> AsyncOpenAI:
        """Get or create AsyncOpenAI client."""
        if self._client is None:
            self._client = AsyncOpenAI(
                api_key=self.api_key,
                base_url=self.BASE_URL,
                timeout=self.timeout,
            )
        return self._client
    
    async def generate_with_image(
        self,
        prompt: str,
        image_bytes: bytes,
        image_format: str = "png",
        system_prompt: str | None = None,
        json_mode: bool = True,
    ) -> VisionResponse:
        """
        Send image + text prompt to Kimi 2.5 vision.
        
        Args:
            prompt: Text prompt/question about the image
            image_bytes: Raw image bytes (PNG, JPEG, etc.)
            image_format: Image format (png, jpeg, webp)
            system_prompt: Optional system prompt for context
            json_mode: If True, request JSON response
        
        Returns:
            VisionResponse with text and optionally parsed JSON
        """
        client = await self._get_client()
        
        # Encode image to base64
        image_b64 = base64.b64encode(image_bytes).decode()
        media_type = f"image/{image_format.lower()}"
        
        # Build message content
        content: list[dict[str, Any]] = [
            {
                "type": "image_url",
                "image_url": {
                    "url": f"data:{media_type};base64,{image_b64}"
                }
            },
            {
                "type": "text",
                "text": prompt
            }
        ]
        
        # Build messages
        messages = []
        if system_prompt:
            messages.append({
                "role": "system",
                "content": system_prompt
            })
        messages.append({
            "role": "user",
            "content": content
        })
        
        # Add JSON instruction if needed
        if json_mode:
            messages.append({
                "role": "system",
                "content": "IMPORTANT: You must respond with valid JSON only. No markdown, no explanation, just the JSON object."
            })
        
        try:
            response = await client.chat.completions.create(
                model=self.MODEL,
                messages=messages,
                max_tokens=self.max_tokens,
                temperature=self.temperature,
            )
            
            text = response.choices[0].message.content or ""
            
            # Parse JSON if requested
            parsed_json = None
            confidence = 0.0
            if json_mode:
                parsed_json = self._extract_json(text)
                if parsed_json and "confidence" in parsed_json:
                    confidence = float(parsed_json.get("confidence", 0))
            
            return VisionResponse(
                text=text,
                parsed_json=parsed_json,
                confidence=confidence,
                model=self.MODEL,
                tokens_used=response.usage.total_tokens if response.usage else 0,
            )
            
        except Exception as e:
            logger.error(f"Kimi vision API error: {e}")
            raise
    
    async def find_element_bbox(
        self,
        image_bytes: bytes,
        target_description: str,
    ) -> dict | None:
        """
        Find element bounding box in screenshot.
        
        Args:
            image_bytes: Screenshot bytes
            target_description: Description of element to find (e.g., "Apply Now button")
        
        Returns:
            Dict with {x, y, width, height} or None if not found
        """
        prompt = f"""Look at this screenshot. Find the element: {target_description}

Return ONLY JSON with the bounding box coordinates relative to the top-left corner of the image:
{{"x": 150, "y": 300, "width": 120, "height": 40, "confidence": 0.9}}

If not found, return: {{"not_found": true, "confidence": 0.5}}"""
        
        response = await self.generate_with_image(
            prompt=prompt,
            image_bytes=image_bytes,
            json_mode=True,
        )
        
        if response.parsed_json:
            if response.parsed_json.get("not_found"):
                return None
            return response.parsed_json
        
        return None
    
    async def analyze_page_type(
        self,
        image_bytes: bytes,
        dom_snapshot: dict | None = None,
    ) -> dict:
        """
        Analyze screenshot to determine page type and key features.
        
        Args:
            image_bytes: Screenshot bytes
            dom_snapshot: Optional DOM snapshot for additional context
        
        Returns:
            Dict with page_type, apply_button, account_required, etc.
        """
        dom_context = ""
        if dom_snapshot:
            import json
            dom_context = f"\n\nDOM Summary:\n{json.dumps(dom_snapshot, indent=2)[:2000]}"
        
        prompt = f"""Analyze this screenshot of a company career/jobs website.{dom_context}

I need to determine:
1. PAGE_TYPE: Is this a job_listing, login_required, form_single, form_wizard, confirmation, or error page?
2. APPLY_BUTTON: Is there an "Apply" or "Apply Now" button visible? Describe its location and text.
3. ACCOUNT_REQUIRED: Does the page require creating an account or signing in before applying?
4. FORM_FIELDS_COUNT: How many form fields are visible (approximate)?
5. ESTIMATED_STEPS: Does this look like a single-page form (1) or multi-step wizard (>1)?
6. LANGUAGE: What language is the interface in (2-letter code)?
7. BLOCKER: Is there a CAPTCHA, geo-block, error message, or 404 visible?

Respond in strict JSON only:
{{
    "page_type": "job_listing|login_required|form_single|form_wizard|confirmation|error",
    "apply_button": {{"visible": true, "text": "Apply Now", "location": "top-right"}},
    "account_required": false,
    "form_fields_count": 5,
    "estimated_steps": 1,
    "language": "en",
    "blocker": null,
    "confidence": 0.85
}}"""
        
        response = await self.generate_with_image(
            prompt=prompt,
            image_bytes=image_bytes,
            json_mode=True,
        )
        
        return response.parsed_json or {}
    
    async def verify_form_state(
        self,
        image_bytes: bytes,
    ) -> dict:
        """
        Verify form filling state - check for empty fields, errors, etc.
        
        Args:
            image_bytes: Screenshot of form
        
        Returns:
            Dict with empty_required_fields, validation_errors, etc.
        """
        prompt = """Look at this screenshot of a partially filled form.

QUESTIONS:
1. Are there any empty required fields (marked with * or "Required")? List them.
2. Are there any validation errors visible (red borders, error messages)?
3. Is there a "Next" / "Continue" / "Submit" button? What is its text?
4. Are all dropdowns properly selected (not showing "Select..." or placeholder)?
5. Are there any file upload fields that need a file?

Return JSON:
{
    "empty_required_fields": ["First Name", "Phone"],
    "validation_errors": ["Email is required"],
    "next_button": {"text": "Continue", "visible": true},
    "unselected_dropdowns": ["Country"],
    "file_upload_needed": false,
    "form_looks_complete": false,
    "confidence": 0.8
}"""
        
        response = await self.generate_with_image(
            prompt=prompt,
            image_bytes=image_bytes,
            json_mode=True,
        )
        
        return response.parsed_json or {}
    
    async def detect_outcome(
        self,
        image_bytes: bytes,
    ) -> dict:
        """
        Detect submission outcome (success/failure/loading/etc).
        
        Args:
            image_bytes: Screenshot after submit action
        
        Returns:
            Dict with outcome, message, reference_id, etc.
        """
        prompt = """Look at this screenshot after clicking "Submit" on a job application form.

Determine the outcome:
1. SUCCESS — confirmation page, "thank you" message, reference number
2. ERROR — validation error, red error messages, "required field" warnings
3. LOADING — spinner, "processing..." message
4. SAME_PAGE — nothing changed, still on the same form
5. REDIRECT — completely different page (not related to application)

Return JSON only:
{
    "outcome": "SUCCESS|ERROR|LOADING|SAME_PAGE|REDIRECT",
    "confidence": 0.95,
    "message": "Application submitted successfully",
    "reference_id": "APP-12345",
    "detected_elements": ["thank you message", "confirmation number"]
}"""
        
        response = await self.generate_with_image(
            prompt=prompt,
            image_bytes=image_bytes,
            json_mode=True,
        )
        
        return response.parsed_json or {}
    
    def _extract_json(self, text: str) -> dict | None:
        """Extract JSON from response text (handles markdown code blocks)."""
        import json
        import re
        
        # Try direct parse
        text = text.strip()
        
        # Remove markdown code blocks if present
        if "```json" in text:
            match = re.search(r"```json\s*([\s\S]*?)\s*```", text)
            if match:
                text = match.group(1)
        elif "```" in text:
            match = re.search(r"```\s*([\s\S]*?)\s*```", text)
            if match:
                text = match.group(1)
        
        try:
            return json.loads(text)
        except json.JSONDecodeError:
            # Try to find JSON object in text
            match = re.search(r"\{[\s\S]*\}", text)
            if match:
                try:
                    return json.loads(match.group())
                except json.JSONDecodeError:
                    pass
            return None
    
    async def close(self):
        """Close the client."""
        if self._client:
            await self._client.close()
            self._client = None


# Singleton instance
_vision_client: KimiVisionClient | None = None


def get_kimi_vision_client() -> KimiVisionClient:
    """Get or create singleton KimiVisionClient."""
    global _vision_client
    if _vision_client is None:
        _vision_client = KimiVisionClient()
    return _vision_client
