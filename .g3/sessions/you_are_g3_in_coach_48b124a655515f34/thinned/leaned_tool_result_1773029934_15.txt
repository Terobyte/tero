IndeedRouter - Degradation-aware apply orchestration.

Phase 4 of the Indeed MTD implementation - coordinates apply flows
with graceful fallback when selectors degrade.
"""

from __future__ import annotations

import logging
from typing import TYPE_CHECKING, Any, Dict, List, Optional

if TYPE_CHECKING:
    from patchright.async_api import Page

from .selector_engine import SelectorEngine
from .selector_health import SelectorHealthMonitor

logger = logging.getLogger(__name__)


class IndeedRouter:
    """
    Routes Indeed job applications with degradation awareness.
    
    Checks selector health before apply attempts and provides
    graceful fallback URLs for manual completion when needed.
    """
    
    def __init__(
        self,
        selector_engine: Optional[SelectorEngine] = None,
        health_monitor: Optional[SelectorHealthMonitor] = None,
    ) -> None:
        """
        Initialize router with optional dependency injection.
        
        Args:
            selector_engine: Optional pre-configured SelectorEngine
            health_monitor: Optional pre-configured SelectorHealthMonitor
        """
        self.selector = selector_engine or SelectorEngine()
        self.health = health_monitor
    
    def check_degradation(self) -> List[str]:
        """
        Check for degraded selectors.
        
        Returns:
            List of intent names that have degraded hit rates.
        """
        if not self.health:
            return []
        
        alerts = self.health.check_degradation()
        return [alert.intent for alert in alerts]
    
    async def apply(
        self,
        job,  # Job dataclass from base_scraper
        page: "Page",
        resume_path: str,
        **kwargs,
    ) -> Dict[str, Any]:
        """
        Apply to a job with degradation awareness.
        
        Args:
            job: Job object with URL, title, company, etc.
            page: Playwright Page instance
            resume_path: Path to resume file
            **kwargs: Additional arguments for IndeedApplier
            
        Returns:
            Dict with:
                - success: bool
                - error: Optional[str]
                - open_url: Optional[str] (for manual fill on failure)
                - degraded_selectors: List[str] (if degradation detected)
                - skip: bool (if early exit for already_applied/job_expired)
        """
        # Import here to avoid circular dependency
        from .indeed_apply import IndeedApplier
        
        # Check degradation before apply
        degraded = self.check_degradation()
        if degraded:
            logger.warning(f"⚠️ Degraded selectors detected: {degraded}")
        
        # Create applier with injected dependencies
        applier = IndeedApplier(
            selector_engine=self.selector,
            health_monitor=self.health,
        )
        
        # Run apply flow
        result = await applier.apply_on_page(job, page, resume_path, **kwargs)
        
        # If failed and we have degradation, add open_url for manual fill
        if not result.get("success") and degraded:
            result["degraded_selectors"] = degraded
            if "open_url" not in result and hasattr(job, "url"):
                result["open_url"] = job.url
                logger.info(f"Manual completion recommended: {job.url}")
        
        return result
    
    def get_health_report(self) -> Dict[str, Any]:
        """
        Get current health report.
        
        Returns:
            Dict with health stats and degraded selectors.
        """
        return self.selector.get_health_report()
    
    def reset_session_budget(self) -> None:
        """
        Reset LLM call counter for a new session.
        
        Note: This creates a new HealthMonitor if one wasn't provided.
        """
        if self.health:
            # HealthMonitor doesn't have a reset method - LLM budget is per-session
            # by default (new instance = fresh budget)
            pass
        else:
            # Create fresh monitor for new session
            self.health = SelectorHealthMonitor()
            self.selector._health = self.health
