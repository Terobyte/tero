Tool result: """
Form Pattern Store - Persists successful form filling patterns.

Enables self-learning: first submission uses LLM, subsequent submissions
reuse confirmed patterns without LLM calls.

Cycle:
  First time:  LLM analyzes → save_pattern(status="pending")
  On success:  confirm_all_pending() → status="confirmed"
  Next time:   load_pattern() → fills without LLM (fast, free)
"""

import hashlib
import json
import re
from pathlib import Path
from typing import Optional, List, Dict, Any

from loguru import logger


class FormPatternStore:
    """
    Persistent storage for successful form filling patterns.
    
    Key format: "{domain}:{fp}:{step_num}"
    - domain: e.g., "lever.co", "indeed.com"
    - fp: MD5 hash of normalized question labels
    - step_num: step number in multi-step forms
    
    Status flow: pending → confirmed
    - pending: saved during incomplete/unknown outcome submission
    - confirmed: validated after successful submission
    """
    
    DATA_FILE = Path("data/form_patterns.json")
    
    def __init__(self, data_file: Optional[Path] = None):
        """Initialize store with optional custom data file path."""
        self.data_file = Path(data_file) if data_file else self.DATA_FILE
        self._ensure_data_file()
    
    def _ensure_data_file(self) -> None:
        """Create data directory and file if they don't exist."""
        self.data_file.parent.mkdir(parents=True, exist_ok=True)
        if not self.data_file.exists():
            self._write_data({})
    
    def _read_data(self) -> Dict[str, Any]:
        """Read patterns from JSON file."""
        try:
            with open(self.data_file, "r", encoding="utf-8") as f:
                return json.load(f)
        except (json.JSONDecodeError, FileNotFoundError):
            return {}
    
    def _write_data(self, data: Dict[str, Any]) -> None:
        """Write patterns to JSON file."""
        with open(self.data_file, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2, ensure_ascii=False)
    
    @staticmethod
    def _normalize_label(label: str) -> str:
        """
        Normalize label for consistent fingerprinting.
        
        - lowercase
        - strip spaces, asterisks, "(required)"
        - collapse multiple spaces to single space
        """
        # Remove asterisks and (required) markers
        normalized = label.replace("*", "").replace("(required)", "")
        # Collapse whitespace and strip
        normalized = re.sub(r'\s+', ' ', normalized.lower()).strip()
        return normalized
    
    def fingerprint_step(self, question_labels: List[str]) -> str:
        """
        Generate MD5 fingerprint from question labels.
        
        Args:
            question_labels: List of question label texts
            
        Returns:
            MD5 hash string (first 12 chars for readability)
        """
        if not question_labels:
            return "empty"
        
        # Sort and normalize labels for consistent fingerprinting
        normalized = sorted(self._normalize_label(l) for l in question_labels)
        combined = "|".join(normalized)
        fp_hash = hashlib.md5(combined.encode()).hexdigest()
        return fp_hash[:12]  # Short hash for readability
    
    def save_pattern(
        self,
        domain: str,
        fp: str,
        step_num: int,
        actions: List[Dict[str, Any]],
        status: str = "pending",
    ) -> None:
        """
        Save a form filling pattern.
        
        Args:
            domain: Platform domain (e.g., "lever.co")
            fp: Fingerprint from fingerprint_step()
            step_num: Step number in multi-step form
            actions: List of FormAction dicts [{label, field_type, value}]
            status: "pending" or "confirmed"
        
        Note: Does NOT overwrite confirmed → pending (prevents degradation)
        """
        key = f"{domain}:{fp}:{step_num}"
        data = self._read_data()
        
        # Protection: don't downgrade confirmed to pending
        if key in data and data[key].get("status") == "confirmed" and status == "pending":
            logger.debug(f"Pattern {key} already confirmed, skipping pending save")
            return
        
        data[key] = {
            "domain": domain,
            "fp": fp,
            "step_num": step_num,
            "actions": actions,
            "status": status,
        }
        
        self._write_data(data)
        logger.debug(f"Saved pattern {key} with status={status}")
    
    def load_pattern(
        self,
        domain: str,
        fp: str,
        step_num: int,
    ) -> Optional[List[Dict[str, Any]]]:
        """
        Load a confirmed pattern.
        
        Args:
            domain: Platform domain
            fp: Fingerprint from fingerprint_step()
            step_num: Step number
            
        Returns:
            List of actions if pattern exists AND is confirmed, else None
            
        Note: pending patterns are NOT returned - they may be erroneous
        """
        key = f"{domain}:{fp}:{step_num}"
        data = self._read_data()
        
        if key not in data:
            return None
        
        entry = data[key]
        if entry.get("status") != "confirmed":
            return None
        
        logger.info(f"✓ Loaded confirmed pattern: {key}")
        return entry.get("actions", [])
    
    def confirm_all_pending(self, domain: str, fps_list: List[str]) -> int:
        """
        Confirm all pending patterns matching domain and fingerprints.
        
        Args:
            domain: Platform domain
            fps_list: List of fingerprints to confirm
            
        Returns:
            Number of patterns confirmed
            
        Note: Uses prefix match on "{domain}:{fp}:" to find patterns
        regardless of step_num (since step_num isn't in fps_list)
        """
        if not fps_list:
            return 0
        
        data = self._read_data()
        count = 0
        
        for key, entry in data.items():
            if (
                entry.get("status") == "pending"
                and entry.get("domain") == domain
                and entry.get("fp") in fps_list
            ):
                entry["status"] = "confirmed"
                count += 1
                logger.debug(f"Confirmed pattern: {key}")
        
        if count > 0:
            self._write_data(data)
            logger.success(f"✅ Confirmed {count} patterns for {domain}")
        
        return count
    
    def get_all_pending(self, domain: Optional[str] = None) -> List[Dict[str, Any]]:
        """
        Get all pending patterns, optionally filtered by domain.
        
        Args:
            domain: Optional domain filter
            
        Returns:
            List of pending pattern entries
        """
        data = self._read_data()
        pending = []
        
        for key, entry in data.items():
            if entry.get("status") == "pending":
                if domain is None or entry.get("domain") == domain:
                    pending.append({"key": key, **entry})
        
        return pending
    
    def clear_all(self) -> None:
        """Clear all stored patterns (for testing)."""
        self._write_data({})
