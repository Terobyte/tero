    def _load_applied_jobs(self):
        """Load applied jobs from database."""
        # Clear existing widgets
        for widget in self.applied_list_frame.winfo_children():
            widget.destroy()
        
        if not self.db_path.exists():
            self.applied_count_var.set("0")
            return
        
        try:
            conn = sqlite3.connect(self.db_path)
            conn.row_factory = sqlite3.Row
            cursor = conn.cursor()
            
            # Get all applied jobs
            cursor.execute("""
                SELECT id, company, title, url, platform, applied_at, screenshot_path
                FROM jobs
                WHERE status = 'applied'
                ORDER BY applied_at DESC
            """)
            
            applied_jobs = cursor.fetchall()
            self.applied_count_var.set(str(len(applied_jobs)))
            
            for job in applied_jobs:
                self._create_applied_job_card(job)
            
            conn.close()
            
        except Exception as e:
            messagebox.showerror("Ошибка", f"Не удалось загрузить заявки: {e}")
    
    def _create_applied_job_card(self, job: sqlite3.Row):

        card_frame = ttk.LabelFrame(
            self.applied_list_frame,

            padding="5"

        card_frame.pack(fill=tk.X, pady=(0, 5), padx=5)
        
        # Date
        date_str = job['applied_at'] or "Дата неизвестна"

            date_str = date_str[:19]  # Truncate to YYYY-MM-DD HH:MM:SS
        ttk.Label(card_frame, text=f"Дата: {date_str}").pack(anchor=tk.W)
        
        # Platform
        ttk.Label(card_frame, text=f"Платформа: {job['platform'] or 'Неизвестно'}").pack(anchor=tk.W)
        
        # Buttons
        btn_frame = ttk.Frame(card_frame)
        btn_frame.pack(fill=tk.X, pady=(5, 0))
        
        # URL button
        if job['url']:
            ttk.Button(
                btn_frame,
                text="Открыть URL",
                command=lambda u=job['url']: webbrowser.open(u)
            ).pack(side=tk.LEFT)
        
        # Screenshot button
        screenshot_path = job['screenshot_path']
        if screenshot_path and Path(screenshot_path).exists():
            ttk.Button(
                btn_frame,
                text="Скриншот",
                command=lambda p=screenshot_path: self._open_screenshot(p)

    def _open_screenshot(self, path: str):
        """Open a screenshot file."""
        full_path = self.project_root / path
        if full_path.exists():
            if PIL_AVAILABLE:
                # Show in a new window
                self._show_screenshot_window(full_path)
            else:
                # Open with system default
                webbrowser.open(f"file://{full_path}")
        else:
            messagebox.showwarning("Не найдено", f"Файл скриншота не существует: {path}")
    
    def _show_screenshot_window(self, path: Path):
        """Show screenshot in a new window using PIL."""
        window = tk.Toplevel(self.root)
        window.title(f"Скриншот: {path.name}")
        
        try:
            img = Image.open(path)
            
            # Resize if too large
            max_width = 800
            max_height = 600
            if img.width > max_width or img.height > max_height:
                img.thumbnail((max_width, max_height), Image.Resampling.LANCZOS)
            
            photo = ImageTk.PhotoImage(img)
            
            label = ttk.Label(window, image=photo)
            label.image = photo  # Keep reference
            label.pack(padx=10, pady=10)
            
        except Exception as e:
            ttk.Label(window, text=f"Ошибка загрузки: {e}").pack(padx=10, pady=10)
    
    def _start_scan(self):
        """Start the job scan in a background thread."""
        if self.scanning:
            return
        
        # Get selected providers
        selected_providers = [p for p, var in self.provider_vars.items() if var.get()]
        if not selected_providers:
            messagebox.showwarning("Внимание", "Выберите хотя бы один провайдер")
            return
        
        # Get search query
        query = self.query_var.get().strip()
        if not query:
            messagebox.showwarning("Внимание", "Введите поисковый запрос")
            return
        
        # Update UI state
        self.scanning = True
        self.scan_btn.config(state=tk.DISABLED)
        self.stop_btn.config(state=tk.NORMAL)
        self.progress_var.set(0)
        self.status_var.set("Сканирование...")
        self.scan_results = []
        
        # Start scan thread
        thread = threading.Thread(
            target=self._run_scan,
            args=(query, selected_providers),
            daemon=True
        )
        thread.start()
    
    def _run_scan(self, query: str, providers: List[str]):
        """Run the actual scan (called in background thread)."""
        try:
            results = []
            total_providers = len(providers)
            
            for i, provider in enumerate(providers):
                if not self.scanning:
                    break
                
                # Update status
                self.root.after(0, lambda p=provider: self.status_var.set(f"Сканирование {p}..."))
                
                # Run actual scraper
                provider_results = self._scan_provider(query, provider)
                results.extend(provider_results)
                
                # Update progress
                progress = ((i + 1) / total_providers) * 100
                self.root.after(0, lambda v=progress: self.progress_var.set(v))
                
                # Update summary
                summary = f"{provider.capitalize()}: {len(provider_results)} найдено"
                self.root.after(0, lambda s=summary: self._update_scan_summary(s))
            
            # Store results
            self.scan_results = results
            
            # Update UI on completion
            self.root.after(0, self._scan_complete)
            
        except Exception as e:
            self.root.after(0, lambda: self._scan_error(str(e)))
    
    def _scan_provider(self, query: str, provider: str) -> List[Dict[str, Any]]:
        """Scan a specific provider for jobs."""
        results = []
        
        try:
            if provider == "indeed":
                results = self._scan_indeed(query)
            elif provider == "ashby":
                results = self._scan_ashby(query)
            elif provider == "greenhouse":
                results = self._scan_greenhouse(query)
            elif provider == "lever":
                results = self._scan_lever(query)
            elif provider == "bamboohr":
                results = self._scan_bamboohr(query)
        except Exception as e:
            print(f"Error scanning {provider}: {e}")
        
        return results

    def _update_scan_summary(self, message: str):
        """Update the scan summary."""
        current = self.scan_summary_var.get()
        if current == "Сканирование не запущено":
            self.scan_summary_var.set(message)
        else:
            self.scan_summary_var.set(current + "\n" + message)
    
    def _scan_complete(self):
        """Handle scan completion."""
        self.scanning = False
        self.scan_btn.config(state=tk.NORMAL)
        self.stop_btn.config(state=tk.DISABLED)
        self.progress_var.set(100)
        
        total = len(self.scan_results)
        self.status_var.set(f"Сканирование завершено: {total} вакансий")
        
        # Update summary
        self.scan_summary_var.set(f"Всего найдено: {total} вакансий")
        
        # Show random job if results available
        if total > 0:
            self._show_random_job()
    
    def _scan_error(self, error: str):
        """Handle scan error."""
        self.scanning = False
        self.scan_btn.config(state=tk.NORMAL)
        self.stop_btn.config(state=tk.DISABLED)
        self.status_var.set("Ошибка сканирования")
        messagebox.showerror("Ошибка", f"Сканирование не удалось: {error}")
    
    def _stop_scan(self):
        """Stop the current scan."""

        self.scan_btn.config(state=tk.NORMAL)

    def _start_apply(self):
        """Start applying to current job in background thread."""
        if not self.current_job:
            messagebox.showwarning("Внимание", "Нет выбранной вакансии")
            return
        
        # Update UI state
        self.apply_job_btn.config(state=tk.DISABLED)
        self.status_var.set("Подача заявки...")
        
        # Start apply thread
        thread = threading.Thread(
            target=self._run_apply,
            daemon=True
        )
        thread.start()
    
    def _run_apply(self):
        """Run the actual apply in background thread."""
        try:
            import asyncio
            result = asyncio.run(self._apply_job_async())
            self.root.after(0, lambda: self._apply_complete(result))
        except Exception as e:
            self.root.after(0, lambda: self._apply_error(str(e)))
    
    async def _apply_job_async(self):
        """Apply to the current job asynchronously.
        
        Routes to the correct applier based on platform/ATS type.
        """
        import sys
        sys.path.insert(0, str(self.project_root / "src"))
        
        from src.applier.ats_detector import ATSDetector
        from database.db import Database
        
        job = self.current_job
        if not job:
            return {"success": False, "error": "No job selected"}
        
        # Determine ATS type from URL
        detector = ATSDetector()
        url = job.get("url", "")
        ats_type = detector.detect_from_url(url)
        platform = job.get("provider", "").lower()
        
        # Prepare job data for applier
        job_data = {
            "id": job.get("id", ""),
            "job_id": job.get("id", ""),
            "url": url,
            "apply_url": url,
            "title": job.get("title", ""),
            "company": job.get("company", ""),
        }
        
        # Load user profile from config
        user_profile = self._load_user_profile()
        job_data["user_data"] = user_profile
        
        # Route to appropriate applier
        if ats_type == "workday":
            from src.applier.workday_apply import WorkdayApplier
            applier = WorkdayApplier(headless=False)
            return await applier.apply(job_data)
        
        elif ats_type == "greenhouse":
            from src.applier.greenhouse_apply import apply_to_greenhouse
            return await apply_to_greenhouse(job_data, user_profile)
        
        elif ats_type == "lever":
            from src.applier.lever_apply import LeverApplier
            applier = LeverApplier(headless=False)
            success, screenshot = await applier.apply(
                job_url=url,
                job_id=job.get("id", ""),
                company=job.get("company", ""),
                position=job.get("title", ""),
                user_data=user_profile
            )
            return {"success": success, "screenshot_path": screenshot, "error": None}
        
        elif ats_type == "ashby":
            from src.applier.ashby_apply import AshbyApplier
            applier = AshbyApplier(headless=False)
            return await applier.apply(
                job_url=url,
                job_id=job.get("id", ""),
                company=job.get("company", "")
            )
        
        elif ats_type == "bamboohr":
            from src.applier.bamboohr_apply import BambooHRApply
            applier = BambooHRApply(headless=False)
            return await applier.apply(
                job_id=job.get("id", ""),
                job_url=url,
                company=job.get("company", ""),
                title=job.get("title", ""),
                applicant_data=user_profile
            )
        
        elif platform == "indeed" or ats_type is None:
            # Use Indeed Router for unknown ATS or Indeed jobs
            from src.applier.indeed_router import IndeedRouter
            router = IndeedRouter(headless=False)
            return await router.route_and_apply(job_data, user_profile)
        
        else:
            return {
                "success": False,
                "error": f"Unsupported ATS type: {ats_type or platform}"
            }
    
    def _load_user_profile(self) -> Dict[str, Any]:
        """Load user profile from config and environment."""
        import os
        
        profile = {
            "first_name": os.getenv("FIRST_NAME", ""),
            "last_name": os.getenv("LAST_NAME", ""),
            "email": os.getenv("EMAIL", ""),
            "phone": os.getenv("PHONE", ""),
            "address": os.getenv("ADDRESS", ""),
            "city": os.getenv("CITY", ""),
            "state": os.getenv("STATE", ""),
            "postal_code": os.getenv("POSTAL_CODE", ""),
            "country": os.getenv("COUNTRY", "United States"),
            "linkedin_url": os.getenv("LINKEDIN_URL", ""),
            "website_url": os.getenv("WEBSITE_URL", ""),
            "resume_path": os.getenv("RESUME_PATH", ""),
        }
        
        # Merge with config if available
        if self.config:
            profile.update(self.config.get("user_profile", {}))
        
        return profile
    
    def _apply_complete(self, result: Dict[str, Any]):
        """Handle apply completion."""
        self.apply_job_btn.config(state=tk.NORMAL)
        
        if result.get("success"):
            self.status_var.set("✅ Заявка подана успешно!")
            
            # Show screenshot if available
            screenshot_path = result.get("screenshot_path")
            if screenshot_path:
                messagebox.showinfo(
                    "Успех",
                    f"Заявка успешно подана!\n\nСкриншот: {screenshot_path}"
                )
            
            # Reload applied jobs
            self._load_applied_jobs()
            
            # Show next random job
            if self.scan_results:
                self._show_random_job()
        else:
            error = result.get("error", "Unknown error")
            self.status_var.set(f"❌ Ошибка: {error}")
            messagebox.showerror("Ошибка", f"Не удалось подать заявку:\n{error}")
    
    def _apply_error(self, error: str):
        """Handle apply error."""
        self.apply_job_btn.config(state=tk.NORMAL)
        self.status_var.set(f"❌ Ошибка: {error}")
        messagebox.showerror("Ошибка", f"Подача заявки не удалась:\n{error}")
