logger = logging.getLogger(__name__)


async def verify_element_state(element, should_check: bool) -> Optional[bool]:
    """
    Verify element state through 3 methods.
    
    Returns:
        True/False if verified, None if unknown (couldn't verify).
    
    Methods:
        1. Native is_checked() for input[type=checkbox/radio]
        2. aria-checked attribute for div[role=checkbox/radio]
        3. CSS class analysis (checked, selected, active, -on)
    """
    # Method 1: Native (input[type=checkbox])
    try:
        native_checked = await element.is_checked()
        return native_checked == should_check
    except Exception:
        pass  # Not a native checkbox, try next method

    # Method 2: aria-checked (div[role=checkbox/radio])
    try:
        aria = await element.get_attribute("aria-checked")
        if aria is not None:
            return (aria == "true") == should_check
    except Exception:
        pass

    # Method 3: CSS classes (checked, selected, active, "on")
    try:
        classes = await element.get_attribute("class") or ""
        class_lower = classes.lower()
        has_checked = any(
            marker in class_lower
            for marker in ["checked", "selected", "active", "-on"]
        )
        return has_checked == should_check
    except Exception:
        pass

    return None  # Could not verify

async def _strategy_label_js(element, should_check: bool) -> None:
    """Click via JavaScript on associated label."""
    page = element.page
    # Try to find associated label
    element_id = await element.get_attribute("id")
    if element_id:
        label = page.locator(f'label[for="{element_id}"]')
        if await label.count() > 0:
            await label.evaluate("el => el.click()")
            return
    
    # Try parent label
    parent = element.locator("xpath=..")
    tag_name = await parent.evaluate("el => el.tagName.toLowerCase()")
    if tag_name == "label":
        await parent.evaluate("el => el.click()")
        return
    
    # Fallback: click the element itself via JS
    await element.evaluate("el => el.click()")


async def _strategy_label_pw(element, should_check: bool) -> None:
    """Click via Playwright on associated label."""
    page = element.page
    element_id = await element.get_attribute("id")
    if element_id:
        label = page.locator(f'label[for="{element_id}"]')
        if await label.count() > 0:
            await label.click()
            return
    
    # Try parent label
    parent = element.locator("xpath=..")
    tag_name = await parent.evaluate("el => el.tagName.toLowerCase()")
    if tag_name == "label":
        await parent.click()
        return
    
    # Fallback: direct click
    await element.click()


async def _strategy_direct_click(element, should_check: bool) -> None:
    """Direct Playwright click on element."""
    await element.click()


async def _strategy_force_click(element, should_check: bool) -> None:
    """Force click with {force: true} bypassing visibility checks."""
    await element.click(force=True)


async def _strategy_js_click(element, should_check: bool) -> None:
    """JavaScript click directly on element."""
    await element.evaluate("el => el.click()")


async def _strategy_set_property(element, should_check: bool) -> None:
    """
    Set checked property via JavaScript.
    Note: This triggers React/Angular change detection properly.
    """
    await element.evaluate(
        f"el => {{ el.checked = {'true' if should_check else 'false'}; "
        f"el.dispatchEvent(new Event('change', {{ bubbles: true }})); "
        f"el.dispatchEvent(new Event('input', {{ bubbles: true }})); }}"
    )


CHECKBOX_STRATEGIES = [
    ("label_js", _strategy_label_js),
    ("label_pw", _strategy_label_pw),
    ("direct_click", _strategy_direct_click),
    ("force_click", _strategy_force_click),
    ("js_click", _strategy_js_click),
    ("set_property", _strategy_set_property),
]


async def fill_checkbox_verified(
    element,
    answer: str,
    strategy_store=None,
    domain: Optional[str] = None,
    page=None,
) -> bool:
    """
    Fill checkbox with verified state change.
    
    Args:
        element: Playwright element handle
        answer: "yes", "true", "1", "no", "false", "0"
        strategy_store: Optional StrategyStore for learning
        domain: Domain for strategy learning
    
    Returns:
        True if verified, False if failed
    """
    should_check = answer.lower() in ["yes", "true", "1", "checked", "on"]
    
    # Already in desired state?
    state = await verify_element_state(element, should_check)
    if state is True:
        logger.debug("✓ Checkbox already in correct state")
        return True
    
    # Try winning strategy first if available
    winning_strategy = None
    if strategy_store and domain:
        winning_strategy = strategy_store.get_winning_strategy(domain, "checkbox")
    
    strategies_to_try = list(CHECKBOX_STRATEGIES)
    
    # Move winning strategy to front
    if winning_strategy:
        for i, (name, _) in enumerate(strategies_to_try):
            if name == winning_strategy:
                strategies_to_try.insert(0, strategies_to_try.pop(i))
                break
    
    for name, strategy in strategies_to_try:
        try:
            await strategy(element, should_check)
        except Exception as e:
            logger.debug(f"Strategy {name} failed: {e}")
            continue
        
        # Give DOM time to update
        await asyncio.sleep(0.15)
        
        verified = await verify_element_state(element, should_check)
        
        if verified is True:
            if strategy_store and domain:
                strategy_store.record_success(domain, "checkbox", name)
            logger.debug(f"✓ Checkbox verified via {name}")
            return True
        elif verified is False:
            # Strategy executed but state didn't change
            if strategy_store and domain:
                strategy_store.record_failure(domain, "checkbox", name)
            continue
        # verified is None: couldn't verify, assume success
        logger.warning(f"⚠ Checkbox unverified after {name}, assuming success")
        return True
    
    logger.warning("⚠ fill_checkbox_verified: all 6 strategies failed")
    return False
