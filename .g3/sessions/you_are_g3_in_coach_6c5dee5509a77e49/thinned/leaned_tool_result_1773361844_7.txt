Answer Bank for Screening Questions

Provides quick answers for known screening questions based on profile data.
Uses pattern matching with explicit priority ordering to correctly classify
compound questions.

CRITICAL: Pattern order matters! Sponsorship patterns must be checked BEFORE
work_authorization patterns to handle compound questions like:

"Will you now or in the future require sponsorship for a work visa, 
including the transfer of an existing work authorization?"

This question asks about SPONSORSHIP, not work authorization. The answer
should be based on sponsorship needs, NOT work authorization status.
"""

    def __init__(self, profile: Optional[Dict[str, Any]] = None):
        """
        Initialize the answer bank.
        
        Args:
            profile: User profile data for profile-based answers

        self.profile = profile or {}
    
    def _normalize_text(self, text: str) -> str:
        """Normalize question text for matching."""
        text = text.lower().strip()
        text = re.sub(r'\s+', ' ', text)
        # Remove punctuation except question marks
        text = re.sub(r'[^\w\s\?]', '', text)
        return text
    
    def get_answer(
        self, 
        question_text: str,
        options: Optional[List[str]] = None
    ) -> Optional[AnswerMatch]:
        """
        Get an answer for a screening question.
        
        Uses three-tier matching:
        1. Exact match (confidence=1.0)
        2. Pattern match (confidence=0.80-0.98)
        3. Profile-based match (confidence=0.90)
        
        Args:
            question_text: The question to answer
            options: Available options (for validation)
            
        Returns:
            AnswerMatch if found, None otherwise
        """
        normalized = self._normalize_text(question_text)
        
        # Tier 1: Exact match
        if normalized in self.EXACT_MATCHES:
            intent, default_answer = self.EXACT_MATCHES[normalized]
            answer = self._get_profile_answer(intent, default_answer)
            return AnswerMatch(
                intent=intent,
                answer=answer,
                confidence=1.0,
                source="exact"
            )
        
        # Tier 2: Pattern match (priority ordered)
        # CRITICAL: Patterns are ordered - sponsorship before work_authorization
        for pattern, intent, default_answer, confidence in self.PATTERNS:
            if re.search(pattern, normalized):
                answer = self._get_profile_answer(intent, default_answer)
                
                # Validate against options if provided
                if options and answer not in options:
                    # Try to find a matching option
                    answer = self._fuzzy_match_option(answer, options)
                    if not answer:
                        return None
                
                return AnswerMatch(
                    intent=intent,
                    answer=answer,
                    confidence=confidence,
                    source="pattern",
                    pattern=pattern
                )
        
        # Tier 3: Profile-based match
        profile_answer = self._check_profile_patterns(question_text)
        if profile_answer:
            return profile_answer
        
        return None

    def _fuzzy_match_option(
        self, 
        target: str, 
        options: List[str]
    ) -> Optional[str]:
        """
        Fuzzy match target answer to available options.
        
        Args:
            target: The target answer (e.g., "No")
            options: Available options on the page
            
        Returns:
            Best matching option or None
        """
        target_lower = target.lower().strip()
        
        # Exact match
        for opt in options:
            if opt.lower().strip() == target_lower:
                return opt
        
        # Partial match (target is contained in option)
        for opt in options:
            opt_lower = opt.lower().strip()
            if target_lower in opt_lower or opt_lower in target_lower:
                return opt
        
        # For Yes/No, try to find options that contain the answer
        if target_lower in ("yes", "no"):
            for opt in options:
                opt_lower = opt.lower()
                if target_lower in opt_lower:
                    return opt
        
        return None
    
    def get_yes_no(
        self, 
        question_text: str,
        prefer_no: bool = False
    ) -> Optional[str]:
        """
        Get a Yes/No answer for a question.
        
        This is a convenience method for yes/no questions.
        
        CRITICAL: For critical intents (sponsorship), never return an
        optimistic "Yes" as a default. If the intent is critical and
        confidence is low, return None to trigger validation.
        
        Args:
            question_text: The question text
            prefer_no: If True, prefer "No" for uncertain matches
            
        Returns:
            "Yes", "No", or None if uncertain
        """
        match = self.get_answer(question_text)
        if not match:
            return None
        
        # Check for critical intent with low confidence
        if match.intent in self.CRITICAL_INTENTS:
            # For critical intents, require high confidence
            if match.confidence < 0.90:
                # Don't guess on critical questions
                return None
        
        # Ensure answer is Yes/No
        answer = match.answer
        if answer not in ("Yes", "No"):
            # Try to convert
            if answer.lower() in ("yes", "true", "1"):
                return "Yes"
            elif answer.lower() in ("no", "false", "0"):
                return "No"
            return None
        
        return answer
    
    def is_critical_intent(self, intent: str) -> bool:

def fuzzy_match_option(target: str, options: List[str]) -> Optional[str]:
    """
    Standalone function for fuzzy matching options.
    
    Used by correction_executor for matching correct_answer to page options.
    
    Args:
        target: The target answer
        options: Available options
        
    Returns:
        Best matching option or None
    """
    target_lower = target.lower().strip()
    
    # Exact match
    for opt in options:
        if opt.lower().strip() == target_lower:
            return opt
    
    # Partial match
    for opt in options:
        opt_lower = opt.lower().strip()
        if target_lower in opt_lower or opt_lower in target_lower:
            return opt
    
    # For Yes/No, find options containing the answer
    if target_lower in ("yes", "no"):
        for opt in options:
            opt_lower = opt.lower()
            if target_lower in opt_lower:
                return opt
        # Try "I do/don't" style options
        if target_lower == "no":
            for opt in options:
                if "do not" in opt.lower() or "don't" in opt.lower():
                    return opt
        elif target_lower == "yes":
            for opt in options:
                if "i do" in opt.lower() and "not" not in opt.lower():
                    return opt
    
    return None
