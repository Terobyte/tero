Tool result: """
AI Answerer - Gemini-based Q&A with caching.

Provides intelligent answers to custom application questions with:
- AnswerBank for static responses
- Gemini cache for previously answered questions
- Gemini API for new questions
- Multi-variant generation for HITL (Human-in-the-Loop) prompts
"""

import json
import hashlib
import os
from pathlib import Path
from typing import Optional, List, Dict, Any, Tuple
from dataclasses import dataclass
from loguru import logger


# System prompt for single-answer generation
SYSTEM_PROMPT = """You are an AI assistant helping a job applicant fill out job application forms.
Answer questions concisely and professionally based on the candidate's profile.
Be honest and factual. If you don't know something, say so.
Format answers appropriately for the field type (text, textarea, select, etc.)."""

# Prompt template for single answer
ANSWER_PROMPT = """Answer this job application question based on the candidate profile.

Question: {question}
Field Type: {field_type}
{options_section}
{context_section}

Profile:
{profile}

Provide a single, best answer. Be concise (under 200 words for textareas).
Answer:"""


# Prompt for generating 4 variants with different styles
VARIANTS_PROMPT = """Generate exactly 4 different answer variants for this job application question.

Question: {question}
Field Type: {field_type}
{options_section}
{context_section}

Profile:
{profile}

Create 4 variants with these styles:
1. FORMAL: Professional, technical, 3-4 sentences with facts from profile
2. FRIENDLY: Conversational, engaging, 2-3 sentences focusing on growth/team
3. CONCISE: Brief, business-like, 1-2 sentences with key facts only
4. ATS_OPTIMAL: Bullet-point or keyword-rich, single line, skills comma-separated

Respond in JSON format only:
{{"variants": ["variant1 text", "variant2 text", "variant3 text", "variant4 text"], "best": 0}}

"best" is the 0-based index of the most ATS-optimized variant (usually index 3).
JSON response:"""


@dataclass
class AIAnswer:
    """Container for AI-generated answer."""
    answer: str
    question_hash: str
    field_type: str
    cached: bool = False
    confidence: str = "high"


class AIAnswerer:
    """
    Gemini-powered Q&A system with multi-level caching.
    
    Levels:
    1. AnswerBank (static) - from data/answers.json
    2. Gemini Cache - from data/custom_answers.json
    3. Gemini API - live generation
    
    Also provides ask_variants() for HITL prompts with 4 style variants.
    """
    
    def __init__(
        self,
        profile: Dict[str, Any],
        gemini_client: Optional[Any] = None,
        answers_file: str = "data/answers.json",
        cache_file: str = "data/custom_answers.json",
    ):
        """
        Initialize AI Answerer.
        
        Args:
            profile: Candidate profile dict (from profile.yaml)
            gemini_client: GeminiClient instance (optional, lazy-loaded if None)
            answers_file: Path to AnswerBank JSON
            cache_file: Path to custom answers cache
        """
        self.profile = profile
        self._gemini = gemini_client
        self.answers_file = Path(answers_file)
        self.cache_file = Path(cache_file)
        
        # Load static answer bank
        self._bank = self._load_json(self.answers_file)
        
        # Load or create cache
        self._cache = self._load_json(self.cache_file)
        
        logger.info(f"AIAnswerer initialized: {len(self._bank)} bank answers, {len(self._cache)} cached")
    
    def _load_json(self, path: Path) -> Dict[str, Any]:
        """Load JSON file or return empty dict."""
        if path.exists():
            try:
                with open(path, 'r', encoding='utf-8') as f:
                    return json.load(f)
            except (json.JSONDecodeError, IOError) as e:
                logger.warning(f"Failed to load {path}: {e}")
        return {}
    
    def _save_cache(self) -> None:
        """Persist cache to disk."""
        self.cache_file.parent.mkdir(parents=True, exist_ok=True)
        with open(self.cache_file, 'w', encoding='utf-8') as f:
            json.dump(self._cache, f, indent=2, ensure_ascii=False)
    
    def _hash_question(self, question: str, options: Optional[List[str]] = None) -> str:
        """Generate MD5 hash for question + options."""
        content = question.lower().strip()
        if options:
            content += "|".join(sorted(options))
        return hashlib.md5(content.encode()).hexdigest()[:12]
    
    @property
    def gemini(self) -> Any:
        """Lazy-load GeminiClient."""
        if self._gemini is None:
            from src.ai.llm_client import GeminiClient
            self._gemini = GeminiClient()
        return self._gemini
    
    # -------------------------------------------------------------------------
    # Level 1: AnswerBank (static)
    # -------------------------------------------------------------------------
    
    def get_bank_answer(self, question: str, options: Optional[List[str]] = None) -> Optional[str]:
        """
        Get answer from static AnswerBank.
        
        Uses fuzzy matching on question keywords.
        """
        q_lower = question.lower()
        
        # Direct key match
        for key, value in self._bank.items():
            if key.lower() in q_lower or q_lower in key.lower():
                # For select/radio, validate against options
                if options and value not in options:
                    continue
                return value
        
        # Fuzzy keyword match
        keywords = set(q_lower.split())
        for key, value in self._bank.items():
            key_words = set(key.lower().split())
            overlap = keywords & key_words
            if len(overlap) >= 2:  # At least 2 matching words
                if options and value not in options:
                    continue
                return value
        
        return None
    
    # -------------------------------------------------------------------------
    # Level 2: Gemini Cache
    # -------------------------------------------------------------------------
    
    def get_cached_answer(self, question: str, options: Optional[List[str]] = None) -> Optional[str]:
        """Get previously cached Gemini answer."""
        q_hash = self._hash_question(question, options)
        entry = self._cache.get(q_hash)
        
        if entry:
            # Validate against options if provided
            if options and entry.get("answer") not in options:
                return None
            return entry.get("answer")
        return None
    
    def set_cached_answer(
        self,
        question: str,
        answer: str,
        options: Optional[List[str]] = None,
    ) -> None:
        """Cache an answer for future use."""
        q_hash = self._hash_question(question, options)
        self._cache[q_hash] = {
            "question": question[:200],  # Truncate for storage
            "answer": answer,
            "options": options or [],
        }
        self._save_cache()
        logger.debug(f"Cached answer: {q_hash} -> {answer[:50]}...")
    
    # -------------------------------------------------------------------------
    # Level 3: Gemini API (single answer)
    # -------------------------------------------------------------------------
    
    async def ask(
        self,
        question: str,
        field_type: str = "text",
        options: Optional[List[str]] = None,
        context: Optional[Dict[str, Any]] = None,
    ) -> AIAnswer:
        """
        Generate a single answer using Gemini.
        
        Args:
            question: The question text
            field_type: Field type (text, textarea, select, radio, etc.)
            options: Available options for select/radio fields
            context: Additional context (company, job_title, etc.)
        
        Returns:
            AIAnswer with the generated response
        """
        q_hash = self._hash_question(question, options)
        
        # Build prompt
        options_section = f"Options: {', '.join(options)}" if options else ""
        context_section = ""
        if context:
            parts = []
            if context.get("company"):
                parts.append(f"Company: {context['company']}")
            if context.get("job_title"):
                parts.append(f"Position: {context['job_title']}")
            context_section = " | ".join(parts)
        
        prompt = ANSWER_PROMPT.format(
            question=question,
            field_type=field_type,
            options_section=options_section,
            context_section=context_section,
            profile=json.dumps(self.profile, indent=2, ensure_ascii=False),
        )
        
        try:
            response = await self.gemini.generate(SYSTEM_PROMPT + "\n\n" + prompt)
            answer = response.strip()
            
            # Validate against options
            if options and answer not in options:
                # Try to find closest match
                answer_lower = answer.lower()
                for opt in options:
                    if opt.lower() == answer_lower:
                        answer = opt
                        break
            
            return AIAnswer(
                answer=answer,
                question_hash=q_hash,
                field_type=field_type,
                cached=False,
                confidence="high",
            )
        except Exception as e:
            logger.error(f"Gemini API error: {e}")
            return AIAnswer(
                answer="",
                question_hash=q_hash,
                field_type=field_type,
                cached=False,
                confidence="low",
            )
    
    # -------------------------------------------------------------------------
    # Level 3 Alternative: Gemini API (4 variants for HITL)
    # -------------------------------------------------------------------------
    
    async def ask_variants(
        self,
        question: str,
        field_type: str = "textarea",
        options: Optional[List[str]] = None,
        context: Optional[Dict[str, Any]] = None,
    ) -> Tuple[List[str], int]:
        """
        Generate 4 answer variants for HITL prompt.
        
        Returns tuple of (variants_list, best_index).
        Falls back to single ask() on error.
        
        Args:
            question: The question text
            field_type: Field type
            options: Available options for select/radio
            context: Additional context
        
        Returns:
            (["variant1", "variant2", ...], best_index)
        """
        # Build prompt
        options_section = f"Options: {', '.join(options)}" if options else ""
        context_section = ""
        if context:
            parts = []
            if context.get("company"):
                parts.append(f"Company: {context['company']}")
            if context.get("job_title"):
                parts.append(f"Position: {context['job_title']}")
            context_section = " | ".join(parts)
        
        prompt = VARIANTS_PROMPT.format(
            question=question,
            field_type=field_type,
            options_section=options_section,
            context_section=context_section,
            profile=json.dumps(self.profile, indent=2, ensure_ascii=False),
        )
        
        try:
            response = await self.gemini.generate(prompt)
            
            # Parse JSON response
            # Try to extract JSON from response (may have markdown fences)
            json_str = response.strip()
            if "```json" in json_str:
                json_str = json_str.split("```json")[1].split("```")[0]
            elif "```" in json_str:
                json_str = json_str.split("```")[1].split("```")[0]
            
            result = json.loads(json_str.strip())
            
            variants = result.get("variants", [])
            best = result.get("best", 0)
            
            # Validate
            if not variants or len(variants) < 1:
                raise ValueError("No variants in response")
            
            # Clamp best index
            best = max(0, min(best, len(variants) - 1))
            
            # Validate options for select/radio
            if options and field_type in ("select", "radio"):
                # Check if variants match options
                valid_variants = []
                for v in variants:
                    v_lower = v.lower()
                    for opt in options:
                        if opt.lower() == v_lower:
                            valid_variants.append(opt)
                            break
                    else:
                        valid_variants.append(v)
                variants = valid_variants if valid_variants else variants
            
            logger.debug(f"Generated {len(variants)} variants for: {question[:50]}...")
            return variants, best
            
        except (json.JSONDecodeError, ValueError, KeyError) as e:
            logger.warning(f"Failed to parse variants JSON: {e}")
            # Fallback to single ask()
            answer = await self.ask(question, field_type, options, context)
            return [answer.answer], 0
            
        except Exception as e:
            logger.error(f"Gemini variants error: {e}")
            # Last resort fallback
            logger.warning(f"ask_variants fallback to hardcoded string: {question[:60]}")
            return ["Please see my resume for details."], 0
