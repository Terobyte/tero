class TestParseJsonLd:
    """Tests for _parse_jsonld_to_job method."""
    
    def test_parse_valid_jsonld_returns_job(self, scraper, sample_jsonld):
        """Valid JSON-LD parses to Job dataclass."""
        job = scraper._parse_jsonld_to_job(
            sample_jsonld,
            "https://www.indeed.com/viewjob?jk=abc123"
        )
        
        assert job is not None
        assert job.title == "Senior Software Engineer"
        assert job.company == "TechCorp Inc"
        assert "San Francisco" in job.location  # May include country code
        assert job.url == "https://www.indeed.com/viewjob?jk=abc123"
        assert "Python" in job.description
    
    def test_parse_jsonld_missing_fields_uses_defaults(self, scraper):
        """JSON-LD with missing fields uses defaults."""
        minimal_jsonld = {
            "@type": "JobPosting",
            "title": "Developer",
            "hiringOrganization": {"name": "Acme"},
        }
        
        job = scraper._parse_jsonld_to_job(
            minimal_jsonld,
            "https://www.indeed.com/viewjob?jk=test"
        )
        
        assert job is not None
        assert job.title == "Developer"
        assert job.company == "Acme"
        assert job.location == ""  # default
        assert job.description == ""  # default
    
    def test_parse_jsonld_wrong_type_returns_none(self, scraper):
        """Non-JobPosting JSON-LD returns None."""
        wrong_type = {
            "@type": "BreadcrumbList",
            "itemListElement": []
        }
        
        job = scraper._parse_jsonld_to_job(wrong_type, "https://example.com")
        
        assert job is None
    
    def test_parse_jsonld_nested_location(self, scraper):
        """Nested location object is flattened correctly."""
        jsonld = {
            "@type": "JobPosting",
            "title": "Engineer",
            "hiringOrganization": {"name": "Corp"},
            "jobLocation": {
                "@type": "Place",
                "address": {
                    "@type": "PostalAddress",
                    "addressLocality": "New York",
                    "addressRegion": "NY"
                }
            }
        }
        
        job = scraper._parse_jsonld_to_job(jsonld, "https://example.com")
        
        assert job.location == "New York, NY"
    
    def test_parse_jsonld_salary_extraction(self, scraper):
        """Salary range is extracted from JSON-LD."""
        jsonld = {
            "@type": "JobPosting",
            "title": "Engineer",
            "hiringOrganization": {"name": "Corp"},
            "baseSalary": {
                "@type": "MonetaryAmount",
                "value": {
                    "@type": "QuantitativeValue",
                    "minValue": 100000,
                    "maxValue": 150000,
                    "unitText": "YEAR"
                }
            }
        }
        
        job = scraper._parse_jsonld_to_job(jsonld, "https://example.com")
        
        assert job.salary == "$100,000 - $150,000/year"
