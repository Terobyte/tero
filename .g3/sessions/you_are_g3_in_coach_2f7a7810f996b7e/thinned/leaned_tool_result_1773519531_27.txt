"""
APScheduler-based task scheduler for CareerBot.

Scheduled tasks:
  - Daily insights at 9:00 (send daily questions to Telegram)
  - Weekly market report on Friday at 18:00

Usage:
    from src.utils.scheduler import setup_scheduler
    
    scheduler = setup_scheduler(bot, chat_id)
    scheduler.start()
"""
from __future__ import annotations

import asyncio
import logging
from datetime import datetime
from typing import Optional

from apscheduler.schedulers.asyncio import AsyncIOScheduler
from apscheduler.triggers.cron import CronTrigger
from apscheduler.jobstores.memory import MemoryJobStore

logger = logging.getLogger(__name__)

# Global scheduler instance
_scheduler: Optional[AsyncIOScheduler] = None


def setup_scheduler(bot, chat_id: int) -> AsyncIOScheduler:
    """
    Set up the scheduler with all scheduled tasks.
    
    Args:
        bot: aiogram Bot instance
        chat_id: Telegram chat ID to send messages to
        
    Returns:
        Configured AsyncIOScheduler instance
    """
    global _scheduler
    
    if _scheduler is not None:
        logger.warning("Scheduler already initialized, returning existing instance")
        return _scheduler
    
    # Create scheduler with memory job store
    _scheduler = AsyncIOScheduler(
        jobstores={"default": MemoryJobStore()},
        timezone="UTC",
    )
    
    # Daily insights at 9:00 UTC
    _scheduler.add_job(
        func=lambda: _send_daily_insights(bot, chat_id),
        trigger=CronTrigger(hour=9, minute=0),
        id="daily_insights",
        name="Daily Insight Questions",
        replace_existing=True,
        max_instances=1,
    )
    logger.info("Scheduled: daily_insights at 9:00 UTC")
    
    # Weekly market report on Friday at 18:00 UTC
    _scheduler.add_job(
        func=lambda: _send_weekly_market_report(bot, chat_id),
        trigger=CronTrigger(day_of_week="fri", hour=18, minute=0),
        id="weekly_market_report",
        name="Weekly Market Report",
        replace_existing=True,
        max_instances=1,
    )
    logger.info("Scheduled: weekly_market_report on Friday at 18:00 UTC")
    
    return _scheduler


def get_scheduler() -> Optional[AsyncIOScheduler]:
    """Get the current scheduler instance."""
    return _scheduler


async def _send_daily_insights(bot, chat_id: int) -> None:
    """Send daily insight questions to the user."""
    from src.insights.daily_engine import DailyInsightEngine
    
    try:
        engine = DailyInsightEngine()
        questions = engine.get_todays_questions(count=2)
        
        if not questions:
            logger.info("No daily insight questions available")
            return
        
        for q in questions:
            await bot.send_message(
                chat_id,
                f"☀️ *Daily [{q.category.title()}]:*\n\n{q.question}\n\n"
                f"_Reply with your answer or use /insights for more options_",
                parse_mode="Markdown",
            )
        
        logger.info(f"Sent {len(questions)} daily insight questions")
        
    except Exception as e:
        logger.error(f"Failed to send daily insights: {e}")


async def _send_weekly_market_report(bot, chat_id: int) -> None:
    """Send weekly market insights report to the user."""
    from src.analytics.market_insights import MarketInsights
    
    try:
        insights = MarketInsights()
        report = insights.generate_report()
        
        header = (
            f"📊 *Weekly Market Report*\n"
            f"_Generated: {datetime.utcnow().strftime('%Y-%m-%d %H:%M')} UTC_\n\n"
        )
        
        await bot.send_message(
            chat_id,
            header + report.markdown,
            parse_mode="Markdown",
        )
        
        logger.info("Sent weekly market report")
        
    except Exception as e:
        logger.error(f"Failed to send weekly market report: {e}")


def shutdown_scheduler() -> None:
    """Gracefully shutdown the scheduler."""
    global _scheduler
    
    if _scheduler is not None:
        _scheduler.shutdown(wait=True)
        logger.info("Scheduler shutdown complete")
        _scheduler = None


# Convenience function for manual trigger
async def trigger_daily_insights_now(bot, chat_id: int) -> None:
    """Manually trigger daily insights (for testing)."""
    await _send_daily_insights(bot, chat_id)


async def trigger_market_report_now(bot, chat_id: int) -> None:
    """Manually trigger market report (for testing)."""
    await _send_weekly_market_report(bot, chat_id)
