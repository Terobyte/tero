# Preamble Detection Constants
# ─────────────────────────────────────────────────────────────────────────────

_PREAMBLE_MARKERS = (
    "based on", "i would like to", "as per", "according to",
    "as mentioned", "given your", "looking at", "given the",
    "thank you for", "i am pleased", "i am writing",
    "in response to", "with reference to", "regarding your",
    "i believe that", "it is worth noting",
)

_ABBREVIATIONS = frozenset({
    "ph.d", "u.s", "dr", "mr", "mrs", "ms", "jr", "sr",
    "inc", "ltd", "corp", "e.g", "i.e", "vs", "etc",
})

_PREAMBLE_FIELD_TYPES = {"textarea"}

def _find_sentence_end(text: str) -> int:
    """
    Find index of first real sentence-ending '. ' in text.
    Skips abbreviations (Ph.D., U.S., etc.).

    Returns:
        Index of '. ' or -1 if not found.

    Examples:
        >>> _find_sentence_end("Based on my Ph.D. research. I am skilled.")
        26
        >>> _find_sentence_end("Hello world")
        -1
        >>> _find_sentence_end("Dr. Smith is here. Welcome.")
        17
    """
    idx = 0
    while idx < len(text):
        dot_pos = text.find(". ", idx)
        if dot_pos == -1:
            return -1
        word_start = text.rfind(" ", 0, dot_pos) + 1
        word_before = text[word_start:dot_pos].lower().rstrip(".")
        if word_before in _ABBREVIATIONS:
            idx = dot_pos + 2
            continue
        return dot_pos
    return -1


def _strip_preamble(text: str) -> str:
    """
    Remove AI preamble phrases from the start of textarea answers.

    Safety guarantees:
    1. Never returns empty string — falls back to original text
    2. Never returns text shorter than 10 chars — falls back to original
    3. Skips abbreviations (Ph.D., U.S., Dr., e.g.)
    4. Handles chained preambles via loop
    5. Case-insensitive matching, preserves original casing

    Examples:
        >>> _strip_preamble("Based on your requirements. I have 5 years of Python.")
        'I have 5 years of Python.'

        >>> _strip_preamble("I want to build great products.")
        'I want to build great products.'

        >>> _strip_preamble("Based on my Ph.D. research. I developed novel algos.")
        'I developed novel algos.'

        >>> _strip_preamble("")
        ''
    """
    if not text or len(text) < 10:
        return text

    original = text

    changed = True
    while changed:
        changed = False
        lower = text.lower()
        for marker in _PREAMBLE_MARKERS:
            if lower.startswith(marker):
                dot_idx = _find_sentence_end(text)
                if dot_idx != -1 and dot_idx < 150:
                    remainder = text[dot_idx + 2:].strip()
                    if len(remainder) >= 10:
                        text = remainder
                        changed = True
                        break
                break  # Marker found but cut is unsafe — exit

    if text != original:
        logger.debug(f"🧹 _strip_preamble: '{original[:60]}' → '{text[:60]}'")

    return text

    def __init__(self, profile: dict, provider: str = "gemini"):
        """
        Initialize AIAnswerer.

        Args:
            profile: Candidate profile dictionary
            provider: AI provider to use ("gemini", "kimi", or "glm5")

    def _get_profile_text(self) -> str:
        """Convert profile dict to formatted text."""
        lines = []
        for section, data in self.profile.items():
            if isinstance(data, dict):
                lines.append(f"\n[{section.upper()}]")
                for key, value in data.items():
                    lines.append(f"  {key}: {value}")
            else:
                lines.append(f"{section}: {data}")
        return "\n".join(lines)

    async def ask(
        self,
        question: str,
        field_type: str = "text",
        options: list[str] | None = None,
        context: str | None = None,
    ) -> "AIAnswer | None":
        """
        Ask AI to answer a form question.

        Args:
            question: The form question
            field_type: Type of field (text, textarea, radio, select, checkbox, date)
            options: Available options for select/radio fields
            context: Additional context for the question

        Returns:
            AIAnswer object or None if failed
        """
        profile_text = self._get_profile_text()

        # Build context section
        context_section = ""
        if options:
            context_section += f"ВАРИАНТЫ: {', '.join(options)}\n"
        if context:
            context_section += f"КОНТЕКСТ: {context}\n"

        # Simulate AI response for greenfield implementation

        import json

            result = json.loads(result_json.strip())

            logger.error(f"Failed to parse AI response: {result_json[:100]}")
            return None

        answer_text = str(result.get("answer", ""))

        # Postprocessing: remove preamble phrases for textarea fields

                question=question,

            answer=answer_text,

            reason=result.get("reason", ""),

    async def ask_variants(
        self,
        question: str,
        field_type: str = "textarea",
        context: str | None = None,
    ) -> tuple[list[str], int]:
        """
        Generate 4 answer variants for a question.

        Args:
            question: The form question
            field_type: Type of field (default: textarea)
            context: Additional context

        Returns:
            Tuple of (variants list, best index)

        context_section = f"КОНТЕКСТ: {context}\n" if context else ""

        # Build prompt using .replace()
        prompt = VARIANTS_PROMPT.replace("{profile}", profile_text)
        prompt = prompt.replace("{question}", question)
        prompt = prompt.replace("{field_type}", field_type)
        prompt = prompt.replace("{context_section}", context_section)

        # Simulate AI response
        result_json = await self._generate_variants_response(prompt, question)

        if not result_json:
            # Fallback
            return ["Please see my resume for details."], 0

        import json

        try:
            result = json.loads(result_json.strip())
        except json.JSONDecodeError:
            logger.error(f"Failed to parse variants response: {result_json[:100]}")
            return ["Please see my resume for details."], 0

        variants = result.get("variants", [])
        best = result.get("best", 0)

        if not variants:
            raise ValueError("No variants in response")

        # Postprocessing: remove preamble only for textarea
        if field_type in _PREAMBLE_FIELD_TYPES:
            raw_variants = variants.copy()
            variants = [_strip_preamble(v) for v in variants]

            # Log telemetry for variants
            for i, (raw, final) in enumerate(zip(raw_variants, variants)):
                if raw != final:
                    logger.debug(
                        f"🧹 Variant {i} preamble stripped: '{raw[:40]}...' → '{final[:40]}...'"
                    )

        return variants, best

class AIAnswer:
    """Data class for AI-generated answer."""

    def __init__(self, answer: str, confidence: str, reason: str):
        self.answer = answer
        self.confidence = confidence
        self.reason = reason

    def __repr__(self) -> str:
        return f"AIAnswer(answer={self.answer!r}, confidence={self.confidence!r})"

    def __eq__(self, other: object) -> bool:
        if not isinstance(other, AIAnswer):
            return False
        return (
            self.answer == other.answer
            and self.confidence == other.confidence
            and self.reason == other.reason
        )
