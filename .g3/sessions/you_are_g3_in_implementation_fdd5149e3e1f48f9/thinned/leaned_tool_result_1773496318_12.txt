"""
AnswerBank — keyword-based answer lookup for screening questions.

Provides pre-defined answers to common screening questions using regex patterns.
Used as a fallback by SemanticAnswerBank when semantic search doesn't find a match.
"""
from __future__ import annotations

import re
from dataclasses import dataclass, field
from typing import Optional
from loguru import logger


@dataclass
class AnswerProfile:
    """User profile containing standard answers."""
    email: str = ""
    phone: str = ""
    first_name: str = ""
    last_name: str = ""
    city: str = ""
    state: str = ""
    country: str = "US"
    linkedin_url: str = ""
    portfolio_url: str = ""
    years_experience: int = 5
    work_authorization: str = "yes"  # yes, no
    visa_sponsorship: str = "no"     # yes, no
    security_clearance: str = "no"   # yes, no
    willing_to_relocate: str = "no"  # yes, no
    drivers_license: str = "yes"     # yes, no
    salary_expectations: str = ""    # e.g., "$120,000 - $150,000"
    start_date: str = "2 weeks"      # availability

class AnswerBank:
    """
    Keyword-based answer lookup for screening questions.
    
    Usage:
        bank = AnswerBank()
        bank.set_profile(AnswerProfile(
            email="john@example.com",
            work_authorization="yes",
            visa_sponsorship="no",
        ))
        
        answer = bank.get_answer("Are you legally authorized to work in the US?")
        # Returns: "Yes"
    """
    
    def __init__(self, profile: Optional[AnswerProfile] = None):
        self._profile = profile or AnswerProfile()
        self._compiled_patterns: dict[re.Pattern, str] = {}
        self._compile_patterns()
    
    def _compile_patterns(self):
        """Compile regex patterns for efficient matching."""
        for pattern, key in KEYWORD_PATTERNS.items():
            try:
                compiled = re.compile(pattern, re.IGNORECASE)
                self._compiled_patterns[compiled] = key
            except re.error:
                logger.warning(f"Invalid pattern: {pattern}")
    
    def set_profile(self, profile: AnswerProfile):
        """Set the answer profile."""
        self._profile = profile
    
    def get_answer(self, question: str) -> Optional[str]:
        """
        Get answer for a screening question.
        
        Args:
            question: The screening question text
            
        Returns:
            Answer string or None if no match found
        """
        question_lower = question.lower().strip()
        
        for pattern, key in self._compiled_patterns.items():
            if pattern.search(question_lower):
                answer = self._get_profile_value(key)
                if answer:
                    logger.debug(f"AnswerBank match: '{question}' -> {key} = {answer}")
                    return answer
        
        return None
    
    def _get_profile_value(self, key: str) -> Optional[str]:
        """Get value from profile, formatting yes/no questions."""
        value = getattr(self._profile, key, None)
        if value is None:
            return None
        
        # Convert boolean-like values
        if isinstance(value, bool):
            return "Yes" if value else "No"
        
        # Convert yes/no string values
        if isinstance(value, str):
            value_lower = value.lower().strip()
            if value_lower in ("yes", "y", "true", "1"):
                return "Yes"
            if value_lower in ("no", "n", "false", "0"):
                return "No"
            return value
        
        # Numbers (years of experience, etc.)
        if isinstance(value, (int, float)):
            return str(value)
        
        return str(value)
    
    def get_yes_no(self, question: str, default: Optional[str] = None) -> Optional[str]:
        """
        Get Yes/No answer for a question.
        
        Args:
            question: The screening question
            default: Default value if no match found
            
        Returns:
            "Yes", "No", or default
        """
        answer = self.get_answer(question)
        if answer in ("Yes", "No"):
            return answer
        return default
    
    def get_answer_for_options(
        self,
        question: str,
        options: list[str],
        default: Optional[str] = None,
    ) -> Optional[str]:
        """
        Get answer matching one of the provided options.
        
        Args:
            question: The screening question
            options: List of valid option strings
            default: Default value if no match found
            
        Returns:
            The matching option or default
        """
        answer = self.get_answer(question)
        if not answer:
            return default
        
        # Try exact match
        answer_lower = answer.lower()
        for opt in options:
            if opt.lower() == answer_lower:
                return opt
        
        # Try partial match
        for opt in options:
            if answer_lower in opt.lower() or opt.lower() in answer_lower:
                return opt
        
        return default
    
    def get_years_for_dropdown(
        self,
        question: str,
        options: Optional[list[str]] = None,
    ) -> Optional[str]:
        """
        Get years of experience, formatted for dropdown.
        
        Args:
            question: The screening question
            options: Valid dropdown options (e.g., ["1-2", "3-5", "5+"])
            
        Returns:
            The best matching option or None
        """
        years = self._profile.years_experience
        
        if options:
            # Find best matching option
            for opt in options:
                opt_lower = opt.lower()
                if "+" in opt or "plus" in opt_lower:
                    # Parse "5+" format
                    try:
                        min_years = int(opt_lower.replace("+", "").replace("plus", "").strip())
                        if years >= min_years:
                            return opt
                    except ValueError:
                        continue
                elif "-" in opt:
                    # Parse "3-5" format
                    try:
                        parts = opt_lower.split("-")
                        min_years = int(parts[0].strip())
                        max_years = int(parts[1].strip())
                        if min_years <= years <= max_years:
                            return opt
                    except (ValueError, IndexError):
                        continue
                else:
                    # Try exact match
                    try:
                        if years == int(opt_lower.strip()):
                            return opt
                    except ValueError:
                        continue
        
        # No matching option, return the years as string
        return str(years)
    
    def set_answer(self, key: str, value: str):
        """
        Set a profile value by key.
        
        Args:
            key: Profile field name
            value: New value
        """
        if hasattr(self._profile, key):
            setattr(self._profile, key, value)
        else:
            logger.warning(f"Unknown profile key: {key}")
    
    def get_all(self) -> dict[str, str]:
        """
        Get all profile values as a dictionary.
        
        Returns:
            Dict mapping field names to string values
        """
        result = {}
        for key in KEYWORD_PATTERNS.values():
            value = self._get_profile_value(key)
            if value:
                result[key] = value
        return result


# Singleton instance for convenience
_default_bank: Optional[AnswerBank] = None


def get_default_bank() -> AnswerBank:
    """Get or create the default AnswerBank instance."""
    global _default_bank
    if _default_bank is None:
        _default_bank = AnswerBank()
    return _default_bank
