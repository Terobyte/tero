Tier 4 LLM Discovery - Gemini Flash Vision API

This module provides visual element discovery using Gemini Flash.
Called as a last resort when semantic and structural selectors fail.
"""
from __future__ import annotations

import asyncio
import logging
import os
from typing import Optional

logger = logging.getLogger(__name__)


async def gemini_find_element(screenshot: bytes, hint: str) -> Optional[str]:
    """
    Send a page screenshot to Gemini Flash with a hint describing the target element.
    Returns a CSS selector string if found, or None.
    
    Called only from SelectorEngine._llm_discover() — always wrapped in asyncio.timeout(10).
    
    Args:
        screenshot: PNG/ JPEG screenshot bytes from page.screenshot()
        hint: Human-readable description of the target element (from IntentConfig.llm_hint)
    
    Returns:
        CSS selector string, or None if not found/ error
    """
    api_key = os.environ.get("GEMINI_API_KEY")
    if not api_key:
        logger.warning("GEMINI_API_KEY not set - Tier 4 LLM discovery unavailable")
        return None
    
    try:
        import google.generativeai as genai
        import PIL.Image
        import io
        
        genai.configure(api_key=api_key)
        model = genai.GenerativeModel("gemini-2.0-flash")
        
        prompt = (
            f"Look at this screenshot of a web page. Find the element: {hint}. "
            "Reply with ONLY a CSS selector string that uniquely identifies it, nothing else. "
            "If you cannot find it, reply with the single word: NONE"
        )
        
        img = PIL.Image.open(io.BytesIO(screenshot))
        
        # Gemini SDK is synchronous - run in thread pool
        response = await asyncio.to_thread(
            model.generate_content,
            [prompt, img]
        )
        
        text = response.text.strip()
        
        if text == "NONE" or not text:
            return None
            
        # Validate the selector doesn't look like hallucinated prose
        if len(text) > 200 or " " in text[:20]:
            # Likely prose description, not a selector
            logger.warning(f"Gemini returned prose instead of selector: {text[:50]}...")
            return None
            
        return text
        
    except ImportError:
        logger.error("google-generativeai not installed. Run: pip install google-generativeai")
        return None
    except Exception as e:
        logger.warning(f"Gemini API error: {e}")
        return None


async def gemini_find_elements(screenshot: bytes, hint: str) -> list[str]:
    """
    Find multiple elements via Gemini Flash.
    Returns a list of CSS selector strings.
    
    Args:
        screenshot: PNG/ JPEG screenshot bytes
        hint: Description of elements to find
    
    Returns:
        List of CSS selector strings (may be empty)
    """
    api_key = os.environ.get("GEMINI_API_KEY")
    if not api_key:
        return []
    
    try:
        import google.generativeai as genai
        import PIL.Image
        import io
        
        genai.configure(api_key=api_key)
        model = genai.GenerativeModel("gemini-2.0-flash")
        
        prompt = (
            f"Look at this screenshot of a web page. Find all elements matching: {hint}. "
            "Reply with ONLY a JSON array of CSS selector strings, nothing else. "
            "Example: [\"div.card\", \"article.item\"]. "
            "If you cannot find any, reply with: []"
        )
        
        img = PIL.Image.open(io.BytesIO(screenshot))
        
        response = await asyncio.to_thread(
            model.generate_content,
            [prompt, img]
        )
        
        text = response.text.strip()
        
        # Try to parse as JSON array
        import json
        if text.startswith("["):
            selectors = json.loads(text)
            return [s for s in selectors if isinstance(s, str) and len(s) < 200]
        
        return []
        
    except Exception as e:
        logger.warning(f"Gemini multi-element API error: {e}")
        return []
