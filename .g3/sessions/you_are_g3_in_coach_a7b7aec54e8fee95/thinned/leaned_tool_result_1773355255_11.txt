_PREAMBLE_MARKERS: tuple[str, ...] = (
    "based on", "i would like to", "as per", "according to",
    "as mentioned", "given your", "looking at", "given the",
    "thank you for", "i am pleased", "i am writing",

_ABBREVIATIONS: frozenset[str] = frozenset({
    "ph.d", "u.s", "dr", "mr", "mrs", "ms", "jr", "sr",
    "inc", "ltd", "corp", "e.g", "i.e", "vs", "etc",
})

def _find_sentence_end(text: str) -> int:
    """
    Find index of first real sentence-ending '. ' in text.
    Skips abbreviations (Ph.D., U.S., etc.).

    Returns:
        Index of '. ' or -1 if not found.

    Examples:
        >>> _find_sentence_end("Based on my Ph.D. research. I am skilled.")
        26
        >>> _find_sentence_end("Hello world")
        -1
        >>> _find_sentence_end("Dr. Smith is here. Welcome.")
        17
    """
    idx = 0
    while idx < len(text):
        dot_pos = text.find(". ", idx)
        if dot_pos == -1:
            return -1
        word_start = text.rfind(" ", 0, dot_pos) + 1
        word_before = text[word_start:dot_pos].lower().rstrip(".")
        if word_before in _ABBREVIATIONS:
            idx = dot_pos + 2
            continue
        return dot_pos
    return -1


def _strip_preamble(text: str) -> str:
    """
    Remove AI preamble phrases from the start of textarea answers.

    Safety guarantees:
    1. Never returns empty string — falls back to original text
    2. Never returns text shorter than 10 chars — falls back to original
    3. Skips abbreviations (Ph.D., U.S., Dr., e.g.)
    4. Handles chained preambles via loop
    5. Case-insensitive matching, preserves original casing

    Examples:
        >>> _strip_preamble("Based on your requirements. I have 5 years of Python.")
        'I have 5 years of Python.'

        >>> _strip_preamble("I want to build great products.")
        'I want to build great products.'

        >>> _strip_preamble("Based on my Ph.D. research. I developed novel algos.")
        'I developed novel algos.'

        >>> _strip_preamble("")
        ''
    """
    if not text or len(text) < 10:
        return text

    original = text

    changed = True
    while changed:
        changed = False
        lower = text.lower()
        for marker in _PREAMBLE_MARKERS:
            if lower.startswith(marker):
                dot_idx = _find_sentence_end(text)
                if dot_idx != -1 and dot_idx < 150:
                    remainder = text[dot_idx + 2:].strip()
                    if len(remainder) >= 10:
                        text = remainder
                        changed = True
                        break
                break  # Marker found but unsafe to cut — exit

    if text != original:
        logger.debug(f"🧹 _strip_preamble: '{original[:60]}' → '{text[:60]}'")

    return text

class AIAnswerer:

    def __init__(
        self,
        profile: dict[str, Any],
        provider: str = "gemini",
        unknown_log_path: Optional[Path] = None,
    ):
        """
        Initialize AI Answerer.

        Args:
            profile: Candidate profile dictionary
            provider: AI provider to use ("gemini", "kimi", "glm5")
            unknown_log_path: Path to log unknown questions (optional)
        """
        self.profile = profile
        self.provider = provider
        self._unknown_log: list[dict[str, Any]] = []
        self._unknown_log_path = unknown_log_path or Path("data/unknown_questions.json")

        # Lazy-loaded clients
        self._gemini: Any = None
        self._kimi: Any = None
        self._glm5: Any = None

    def _get_profile_text(self) -> str:
        """Convert profile dict to formatted text for prompts."""
        lines = []
        for section, data in self.profile.items():
            if isinstance(data, dict):
                lines.append(f"\n[{section.upper()}]")
                for key, value in data.items():
                    lines.append(f"  {key}: {value}")
            else:
                lines.append(f"{section}: {data}")
        return "\n".join(lines)

    def _question_hash(self, question: str) -> str:
        """Generate short hash for question caching."""
        return hashlib.md5(question.encode()).hexdigest()[:12]

    def _save_unknown_log(self) -> None:
        """Save unknown questions log to file."""
        self._unknown_log_path.parent.mkdir(parents=True, exist_ok=True)
        with open(self._unknown_log_path, "w", encoding="utf-8") as f:
            json.dump(self._unknown_log, f, indent=2, ensure_ascii=False)

    async def _get_gemini_client(self) -> Any:
        """Lazy-load Gemini client."""
        if self._gemini is None:
            from ..llm.gemini_client import GeminiClient
            self._gemini = GeminiClient()
        return self._gemini

    async def _get_kimi_client(self) -> Any:
        """Lazy-load Kimi client."""
        if self._kimi is None:
            from ..llm.kimi_client import KimiClient
            self._kimi = KimiClient()
        return self._kimi

    async def _get_glm5_client(self) -> Any:
        """Lazy-load GLM5 client."""
        if self._glm5 is None:
            from ..llm.glm5_client import GLM5Client
            self._glm5 = GLM5Client()
        return self._glm5

    async def ask(
        self,
        question: str,
        field_type: str = "text",
        options: Optional[list[str]] = None,
    ) -> Optional[AIAnswer]:
        """
        Generate a single answer for a form question.

        Args:
            question: The form question text
            field_type: Type of form field ("text", "textarea", "radio", "select")
            options: Available options for select/radio fields

        Returns:
            AIAnswer object or None if generation fails
        """
        profile_text = self._get_profile_text()

        # Use .replace() instead of .format() to avoid issues with curly braces
        prompt = self.SYSTEM_PROMPT.replace("{profile}", profile_text)

        # Add question-specific context
        question_context = f"\n\nВОПРОС: {question}"
        question_context += f"\nТип поля: {field_type}"
        if options:
            question_context += f"\nВарианты ответа: {', '.join(options)}"
        prompt += question_context

        try:
            gemini = await self._get_gemini_client()
            raw_text = await gemini.generate(prompt)

            if not raw_text:
                logger.warning(f"Empty response from Gemini for: {question[:50]}...")
                return None

            # Parse JSON response
            result = json.loads(raw_text.strip())
            answer_text = str(result.get("answer", ""))
            raw_answer = answer_text  # Keep original for telemetry

            # Post-processing: strip preamble for textarea fields
            if field_type in _PREAMBLE_FIELD_TYPES:
                answer_text = _strip_preamble(answer_text)

            answer = AIAnswer(
                answer=answer_text,
                confidence=result.get("confidence", "medium"),
                reason=result.get("reason", ""),
            )

            # Log telemetry
            preamble_stripped = raw_answer != answer_text
            record = QuestionRecord(
                question=question,
                field_type=field_type,
                raw_answer=raw_answer,
                ai_answered=answer_text,
                preamble_stripped=preamble_stripped,
                provider=self.provider,
                timestamp=datetime.now(timezone.utc).isoformat(),
            )
            self._unknown_log.append(record.__dict__)

            if preamble_stripped:
                logger.info(f"📝 Preamble stripped for question: {question[:50]}...")

            return answer

        except json.JSONDecodeError as e:
            logger.error(f"Failed to parse AI response as JSON: {e}")
            return None
        except Exception as e:
            logger.error(f"Error generating answer: {e}")
            return None

    async def ask_variants(
        self,
        question: str,
        field_type: str = "textarea",
        options: Optional[list[str]] = None,
    ) -> tuple[list[str], int]:
        """
        Generate multiple answer variants for a form question.

        Args:
            question: The form question text
            field_type: Type of form field
            options: Available options for select/radio fields

        Returns:
            Tuple of (list of variant answers, index of best variant)

        # Build prompt with .replace()
        prompt = VARIANTS_PROMPT.replace("{profile}", profile_text)
        prompt = prompt.replace("{question}", question)
        prompt = prompt.replace("{field_type}", field_type)

        if options:
            prompt += f"\nВарианты ответа: {', '.join(options)}"

        try:
            # Use appropriate client based on provider
            if self.provider == "kimi":
                client = await self._get_kimi_client()
            else:
                client = await self._get_glm5_client()

            raw_text = await client.generate(prompt)

            if not raw_text:
                logger.warning(f"Empty variants response for: {question[:50]}...")
                # Fallback
                return ["Please see my resume for details."], 0

            result = json.loads(raw_text.strip())
            variants = result.get("variants", [])
            best = result.get("best", 0)

            if not variants:
                logger.warning(f"No variants in response for: {question[:50]}...")
                return ["Please see my resume for details."], 0

            # Post-processing: strip preamble only for textarea
            if field_type in _PREAMBLE_FIELD_TYPES:
                variants = [_strip_preamble(v) for v in variants]

            # Validate best index
            if best < 0 or best >= len(variants):
                best = 0

            return variants, best

        except json.JSONDecodeError as e:
            logger.error(f"Failed to parse variants response as JSON: {e}")
            return ["Please see my resume for details."], 0
        except Exception as e:
            logger.error(f"Error generating variants: {e}")
            return ["Please see my resume for details."], 0
