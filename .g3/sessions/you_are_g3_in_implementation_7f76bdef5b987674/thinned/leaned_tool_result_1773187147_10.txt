Tool result: """
SelectorEngine - 4-tier element discovery with Shadow DOM support.

Tier 1: CSS selectors (fast, reliable)
Tier 2: ARIA/AOM selectors (semantic, MTD-resistant)
Tier 3: Intent-based selectors (from registry)
Tier 4: LLM Vision (fallback, uses bounding boxes)

Includes:
- Shadow DOM piercing with `>>>` selector syntax
- Element freshness retry (re-locate on stale)
- scroll_into_view_if_needed for reliable clicks
"""

@dataclass
class SelectorResult:
    """Result of selector discovery."""
    element: Any  # Playwright Locator or ElementHandle
    selector: str
    tier: int  # Which tier found it (1-4)
    method: str  # Method used within tier
    confidence: float = 1.0


class SelectorEngine:
    """
    4-tier element discovery engine.
    
    Tier 1: Direct CSS selectors
    Tier 2: ARIA/AOM (Accessibility Object Model)
    Tier 3: Intent-based (from registry)
    Tier 4: LLM Vision (with bounding boxes)
    """
    
    MAX_REFRESH_ATTEMPTS = 3  # Element freshness retry attempts
    STALE_RETRY_DELAY = 0.1  # Seconds between freshness retries
    
    def __init__(self, page, llm_client=None):
        """
        Initialize SelectorEngine.
        
        Args:
            page: Playwright Page object
            llm_client: Optional LLM client for Tier 4
        """
        self.page = page
        self.llm_client = llm_client
        self._cache: dict[str, SelectorResult] = {}
    
    async def find_element(
        self,
        intent: str,
        timeout: float = 5000,
        refresh_on_stale: bool = True
    ) -> Optional[SelectorResult]:
        """
        Find element using 4-tier discovery.
        
        Args:
            intent: Semantic intent (e.g., "apply_button", "email_input")
            timeout: Timeout in milliseconds
            refresh_on_stale: Whether to re-locate stale elements
        
        Returns:
            SelectorResult or None if not found
        """
        # Check cache first
        cache_key = f"{intent}_{timeout}"
        if cache_key in self._cache:
            result = self._cache[cache_key]
            if await self._verify_freshness(result):
                return result
            else:
                del self._cache[cache_key]
        
        # Tier 1: Intent-based selectors (CSS)
        result = await self._tier1_intent_selectors(intent, timeout)
        if result:
            self._cache[cache_key] = result
            return result
        
        # Tier 2: ARIA/AOM
        result = await self._tier2_aom_selectors(intent, timeout)
        if result:
            self._cache[cache_key] = result
            return result
        
        # Tier 3: Accessibility snapshot
        result = await self._tier3_accessibility_snapshot(intent, timeout)
        if result:
            self._cache[cache_key] = result
            return result
        
        # Tier 4: LLM Vision
        result = await self._tier4_llm_vision(intent, timeout)
        if result:
            self._cache[cache_key] = result
            return result
        
        logger.warning(f"⚠️ SelectorEngine: element not found for intent '{intent}'")
        return None
    
    async def _tier1_intent_selectors(
        self,
        intent: str,
        timeout: float
    ) -> Optional[SelectorResult]:
        """Tier 1: Try selectors from intent registry."""
        mapping = INTENT_REGISTRY.get(intent)
        if not mapping:
            logger.debug(f"No intent mapping for '{intent}'")
            return None
        
        # Try regular selectors
        for selector in mapping.selectors:
            try:
                element = self.page.locator(selector).first
                if await element.count() > 0:
                    await self._scroll_into_view_if_needed(element)
                    return SelectorResult(
                        element=element,
                        selector=selector,
                        tier=1,
                        method="intent_css"
                    )
            except Exception as e:
                logger.debug(f"Tier 1 selector '{selector}' failed: {e}")
                continue
        
        # Try Shadow DOM selectors with `>>>`
        for shadow_selector in mapping.shadow_selectors:
            try:
                element = await self._find_in_shadow_dom(shadow_selector)
                if element:
                    await self._scroll_into_view_if_needed(element)
                    return SelectorResult(
                        element=element,
                        selector=shadow_selector,
                        tier=1,
                        method="intent_shadow"
                    )
            except Exception as e:
                logger.debug(f"Tier 1 shadow selector '{shadow_selector}' failed: {e}")
                continue
        
        return None
    
    async def _tier2_aom_selectors(
        self,
        intent: str,
        timeout: float
    ) -> Optional[SelectorResult]:
        """Tier 2: Use Accessibility Object Model."""
        try:
            snapshot = await self.page.accessibility.snapshot()
            if not snapshot:
                return None
            
            # Map intent to AOM role and name patterns
            aom_patterns = self._get_aom_patterns(intent)
            
            for pattern in aom_patterns:
                node = self._find_in_aom_tree(snapshot, pattern)
                if node:
                    # Get locator from AOM node
                    selector = self._aom_to_selector(node)
                    if selector:
                        element = self.page.locator(selector).first
                        if await element.count() > 0:
                            await self._scroll_into_view_if_needed(element)
                            return SelectorResult(
                                element=element,
                                selector=selector,
                                tier=2,
                                method="aom"
                            )
        except Exception as e:
            logger.debug(f"Tier 2 AOM failed: {e}")
        
        return None
    
    async def _tier3_accessibility_snapshot(
        self,
        intent: str,
        timeout: float
    ) -> Optional[SelectorResult]:
        """Tier 3: Accessibility snapshot with semantic matching."""
        try:
            # Get page text content for semantic matching
            content = await self.page.content()
            
            # Try ARIA labels and roles
            aria_selectors = self._get_aria_selectors(intent)
            for selector in aria_selectors:
                element = self.page.locator(selector).first
                if await element.count() > 0:
                    await self._scroll_into_view_if_needed(element)
                    return SelectorResult(
                        element=element,
                        selector=selector,
                        tier=3,
                        method="aria"
                    )
        except Exception as e:
            logger.debug(f"Tier 3 accessibility failed: {e}")
        
        return None
    
    async def _tier4_llm_vision(
        self,
        intent: str,
        timeout: float
    ) -> Optional[SelectorResult]:
        """Tier 4: Use LLM vision to find element coordinates."""
        if not self.llm_client:
            logger.debug("No LLM client available for Tier 4")
            return None
        
        try:
            # Inject bounding boxes into page
            boxes = await self._inject_bounding_boxes()
            if not boxes:
                return None
            
            # Take screenshot
            screenshot = await self.page.screenshot()
            
            # Ask LLM which box contains the target element
            prompt = f"""Find the element that matches: {intent}
            
Available elements with numbered bounding boxes:
{self._format_boxes(boxes)}

Return ONLY the number of the correct element, nothing else."""

            response = await self.llm_client.analyze_image(screenshot, prompt)
            
            # Parse response to get box number
            try:
                box_num = int(response.strip())
                if box_num in boxes:
                    box = boxes[box_num]
                    # Return coordinate-based result
                    return SelectorResult(
                        element=None,  # Will use coordinates directly
                        selector=f"coordinates({box['x']}, {box['y']})",
                        tier=4,
                        method="llm_vision",
                        confidence=box.get('confidence', 0.5)
                    )
            except (ValueError, KeyError) as e:
                logger.debug(f"Failed to parse LLM response: {e}")
                
        except Exception as e:
            logger.warning(f"Tier 4 LLM vision failed: {e}")
        
        return None
    
    async def _inject_bounding_boxes(self) -> dict[int, dict]:
        """
        Inject numbered bounding boxes into the page for LLM vision.
        
        Returns:
            Dict mapping box numbers to coordinates
        """
        boxes = {}
        
        try:
            # JavaScript to inject bounding boxes
            js_code = """
            () => {
                const elements = document.querySelectorAll('button, input, select, textarea, a, [role="button"], [role="checkbox"], [role="radio"]');
                const boxes = {};
                let num = 1;
                
                elements.forEach(el => {
                    const rect = el.getBoundingClientRect();
                    if (rect.width > 0 && rect.height > 0) {
                        // Create visual indicator
                        const overlay = document.createElement('div');
                        overlay.style.cssText = `
                            position: absolute;
                            left: ${rect.left}px;
                            top: ${rect.top}px;
                            width: ${rect.width}px;
                            height: ${rect.height}px;
                            border: 2px solid red;
                            background: rgba(255,0,0,0.1);
                            z-index: 99999;
                            pointer-events: none;
                        `;
                        overlay.textContent = num;
                        document.body.appendChild(overlay);
                        
                        boxes[num] = {
                            x: rect.left + rect.width / 2,
                            y: rect.top + rect.height / 2,
                            width: rect.width,
                            height: rect.height,
                            tagName: el.tagName,
                            text: el.textContent?.substring(0, 50) || ''
                        };
                        num++;
                    }
                });
                
                return boxes;
            }
            """
            
            boxes = await self.page.evaluate(js_code)
            await asyncio.sleep(0.1)  # Let overlays render
            
        except Exception as e:
            logger.debug(f"Failed to inject bounding boxes: {e}")
        
        return boxes
    
    def _format_boxes(self, boxes: dict[int, dict]) -> str:
        """Format boxes for LLM prompt."""
        lines = []
        for num, box in boxes.items():
            lines.append(f"[{num}] {box['tagName']}: '{box['text']}'")
        return "\n".join(lines)
    
    async def _find_in_shadow_dom(self, selector: str) -> Optional[Any]:
        """
        Find element in Shadow DOM using `>>>` piercing selector.
        
        Args:
            selector: Selector with `>>>` to pierce shadow boundaries
        
        Returns:
            Playwright locator or None
        """
        if ">>>" not in selector:
            return self.page.locator(selector).first
        
        # Split by `>>>` for shadow piercing
        parts = selector.split(">>>")
        
        try:
            # Start with the host element
            host = self.page.locator(parts[0].strip()).first
            if await host.count() == 0:
                return None
            
            # Navigate through shadow roots
            current = host
            for part in parts[1:]:
                part = part.strip()
                # Get shadow root
                shadow_root = await current.evaluate_handle("el => el.shadowRoot")
                if not shadow_root:
                    return None
                # Find element in shadow root
                current = shadow_root.as_element()
                if not current:
                    return None
                current = current.locator(part).first
                if await current.count() == 0:
                    return None
            
            return current
            
        except Exception as e:
            logger.debug(f"Shadow DOM traversal failed: {e}")
            return None
    
    async def _scroll_into_view_if_needed(self, element: Any) -> bool:
        """
        Scroll element into view if needed.
        
        Args:
            element: Playwright locator or element handle
        
        Returns:
            True if element is now visible, False otherwise
        """
        try:
            await element.scroll_into_view_if_needed(timeout=2000)
            await asyncio.sleep(0.05)  # Small delay for scroll to complete
            return True
        except Exception as e:
            logger.debug(f"Scroll into view failed: {e}")
            return False
    
    async def _verify_freshness(
        self,
        result: SelectorResult,
        max_attempts: int = None
    ) -> bool:
        """
        Verify element is still fresh (not stale).
        
        Args:
            result: Previously cached selector result
            max_attempts: Max refresh attempts
        
        Returns:
            True if element is still valid, False if stale
        """
        max_attempts = max_attempts or self.MAX_REFRESH_ATTEMPTS
        
        try:
            # Try to interact with element to check freshness
            count = await result.element.count()
            return count > 0
        except Exception as e:
            if "stale" in str(e).lower() or "detached" in str(e).lower():
                logger.debug(f"Element is stale: {e}")
                return False
            raise
    
    async def refresh_if_stale(
        self,
        result: SelectorResult,
        intent: str
    ) -> Optional[SelectorResult]:
        """
        Re-locate element if stale.
        
        Args:
            result: Previously found result (possibly stale)
            intent: Original intent for re-location
        
        Returns:
            Fresh result or None if can't re-locate
        """
        if await self._verify_freshness(result):
            return result
        
        logger.info(f"🔄 Element stale, re-locating for intent '{intent}'")
        
        for attempt in range(self.MAX_REFRESH_ATTEMPTS):
            await asyncio.sleep(self.STALE_RETRY_DELAY * (attempt + 1))
            
            # Try same selector first
            try:
                element = self.page.locator(result.selector).first
                if await element.count() > 0:
                    await self._scroll_into_view_if_needed(element)
                    return SelectorResult(
                        element=element,
                        selector=result.selector,
                        tier=result.tier,
                        method=result.method + "_refreshed"
                    )
            except Exception:
                pass
            
            # Full re-discovery
            new_result = await self.find_element(intent, timeout=3000)
            if new_result:
                logger.info(f"✅ Re-located element on attempt {attempt + 1}")
                return new_result
        
        logger.warning(f"⚠️ Could not re-locate stale element after {self.MAX_REFRESH_ATTEMPTS} attempts")
        return None
    
    def _get_aom_patterns(self, intent: str) -> list[dict]:
        """Get AOM search patterns for intent."""
        patterns = []
        
        if "button" in intent:
            patterns.append({"role": "button"})
            patterns.append({"role": "link"})
        
        if "input" in intent:
            patterns.append({"role": "textbox"})
            patterns.append({"role": "searchbox"})
        
        if "checkbox" in intent:
            patterns.append({"role": "checkbox"})
        
        if "radio" in intent:
            patterns.append({"role": "radio"})
        
        if "select" in intent or "dropdown" in intent:
            patterns.append({"role": "combobox"})
            patterns.append({"role": "listbox"})
        
        return patterns
    
    def _find_in_aom_tree(
        self,
        node: dict,
        pattern: dict
    ) -> Optional[dict]:
        """Recursively search AOM tree for matching pattern."""
        if not node:
            return None
        
        # Check if this node matches
        matches = True
        for key, value in pattern.items():
            if node.get(key) != value:
                matches = False
                break
        
        if matches:
            return node
        
        # Search children
        for child in node.get("children", []):
            result = self._find_in_aom_tree(child, pattern)
            if result:
                return result
        
        return None
    
    def _aom_to_selector(self, node: dict) -> Optional[str]:
        """Convert AOM node to CSS selector."""
        if not node:
            return None
        
        # Try role selector first
        role = node.get("role")
        name = node.get("name", "")
        
        if role:
            if name:
                return f'[role="{role}"]:has-text("{name}")'
            return f'[role="{role}"]'
        
        return None
    
    def _get_aria_selectors(self, intent: str) -> list[str]:
        """Get ARIA-based selectors for intent."""
        selectors = []
        
        if "button" in intent:
            if "apply" in intent:
                selectors.extend([
                    '[aria-label*="Apply" i]',
                    'button[aria-label*="Apply" i]',
                    '[role="button"][aria-label*="Apply" i]',
                ])
            elif "submit" in intent:
                selectors.extend([
                    '[aria-label*="Submit" i]',
                    'button[aria-label*="Submit" i]',
                    '[role="button"][aria-label*="Submit" i]',
                ])
        
        if "input" in intent:
            if "email" in intent:
                selectors.extend([
                    '[aria-label*="email" i]',
                    'input[aria-label*="email" i]',
                ])
            elif "phone" in intent:
                selectors.extend([
                    '[aria-label*="phone" i]',
                    'input[aria-label*="phone" i]',
                ])
        
        return selectors


class IntentRegistry:
    """
    Registry for intent-to-selector mappings.
    Allows dynamic registration and domain-specific overrides.
    """
    
    def __init__(self):
        self._registry: dict[str, IntentMapping] = dict(INTENT_REGISTRY)
        self._domain_overrides: dict[str, dict[str, IntentMapping]] = {}
    
    def register(self, intent: str, selectors: list[str], shadow_selectors: list[str] = None):
        """Register a new intent mapping."""
        self._registry[intent] = IntentMapping(
            intent=intent,
            selectors=selectors,
            shadow_selectors=shadow_selectors or []
        )
    
    def register_domain_override(
        self,
        domain: str,
        intent: str,
        selectors: list[str]
    ):
        """Register domain-specific selector override."""
        if domain not in self._domain_overrides:
            self._domain_overrides[domain] = {}
        
        self._domain_overrides[domain][intent] = IntentMapping(
            intent=intent,
            selectors=selectors,
            shadow_selectors=[]
        )
    
    def get(self, intent: str, domain: str = None) -> Optional[IntentMapping]:
        """Get intent mapping, with domain override if available."""
        # Check domain override first
        if domain and domain in self._domain_overrides:
            if intent in self._domain_overrides[domain]:
                return self._domain_overrides[domain][intent]
        
        return self._registry.get(intent)
    
    def clear_cache(self):
        """Clear any cached selectors."""
        pass  # Registry is stateless


# Utility functions

async def scroll_into_view_if_needed(page, element: Any) -> bool:
    """
    Convenience function to scroll element into view.
    
    Args:
        page: Playwright page
        element: Playwright locator or element handle
    
    Returns:
        True if scrolled successfully, False otherwise
    """
    try:
        await element.scroll_into_view_if_needed(timeout=2000)
        await asyncio.sleep(0.05)
        return True
    except Exception as e:
        logger.debug(f"scroll_into_view_if_needed failed: {e}")
        return False


async def find_with_retry(
    page,
    selector: str,
    max_attempts: int = 3,
    retry_delay: float = 0.1
) -> Optional[Any]:
    """
    Find element with freshness retry.
    
    Args:
        page: Playwright page
        selector: CSS selector
        max_attempts: Number of retry attempts
        retry_delay: Delay between retries
    
    Returns:
        Playwright locator or None
    """
    for attempt in range(max_attempts):
        try:
            element = page.locator(selector).first
            if await element.count() > 0:
                return element
        except Exception as e:
            if "stale" in str(e).lower() and attempt < max_attempts - 1:
                await asyncio.sleep(retry_delay)
                continue
            logger.debug(f"find_with_retry failed on attempt {attempt + 1}: {e}")
    
    return None
