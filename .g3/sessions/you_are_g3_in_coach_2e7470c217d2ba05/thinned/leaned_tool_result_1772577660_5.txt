    def _load_applied_jobs(self):
        """Load applied jobs from database."""
        # Clear existing widgets
        for widget in self.applied_list_frame.winfo_children():
            widget.destroy()
        
        if not self.db_path.exists():
            self.applied_count_var.set("0")
            return
        
        try:
            conn = sqlite3.connect(self.db_path)
            conn.row_factory = sqlite3.Row
            cursor = conn.cursor()
            
            # Get all applied jobs
            cursor.execute("""
                SELECT id, company, title, url, platform, applied_at, screenshot_path
                FROM jobs
                WHERE status = 'applied'
                ORDER BY applied_at DESC
            """)
            
            applied_jobs = cursor.fetchall()
            self.applied_count_var.set(str(len(applied_jobs)))
            
            for job in applied_jobs:
                self._create_applied_job_card(job)
            
            conn.close()
            
        except Exception as e:
            messagebox.showerror("Ошибка", f"Не удалось загрузить заявки: {e}")

    def _start_scan(self):
        """Start the job scan in a background thread."""
        if self.scanning:
            return
        
        # Get selected providers
        selected_providers = [p for p, var in self.provider_vars.items() if var.get()]
        if not selected_providers:
            messagebox.showwarning("Внимание", "Выберите хотя бы один провайдер")
            return
        
        # Get search query
        query = self.query_var.get().strip()
        if not query:
            messagebox.showwarning("Внимание", "Введите поисковый запрос")
            return
        
        # Update UI state
        self.scanning = True
        self.scan_btn.config(state=tk.DISABLED)
        self.stop_btn.config(state=tk.NORMAL)
        self.progress_var.set(0)
        self.status_var.set("Сканирование...")
        self.scan_results = []
        
        # Start scan thread
        thread = threading.Thread(
            target=self._run_scan,
            args=(query, selected_providers),
            daemon=True
        )
        thread.start()
    
    def _run_scan(self, query: str, providers: List[str]):
        """Run the actual scan (called in background thread)."""
        try:
            results = []
            total_providers = len(providers)
            
            for i, provider in enumerate(providers):
                if not self.scanning:
                    break
                
                # Update status
                self.root.after(0, lambda p=provider: self.status_var.set(f"Сканирование {p}..."))
                
                # Run actual scraper
                provider_results = self._scan_provider(query, provider)
                results.extend(provider_results)
                
                # Update progress
                progress = ((i + 1) / total_providers) * 100
                self.root.after(0, lambda v=progress: self.progress_var.set(v))
                
                # Update summary
                summary = f"{provider.capitalize()}: {len(provider_results)} найдено"
                self.root.after(0, lambda s=summary: self._update_scan_summary(s))
            
            # Store results
            self.scan_results = results
            
            # Update UI on completion
            self.root.after(0, self._scan_complete)
            
        except Exception as e:
            self.root.after(0, lambda: self._scan_error(str(e)))

        """Update the scan summary."""
        current = self.scan_summary_var.get()

    def _run_apply(self):
        """Run the actual apply in background thread."""
        try:
            import asyncio
            result = asyncio.run(self._apply_job_async())
            self.root.after(0, lambda: self._apply_complete(result))
        except Exception as e:
            self.root.after(0, lambda: self._apply_error(str(e)))

        Routes to the correct applier based on platform/ATS type.
        """
        import sys

        from src.applier.ats_detector import ATSDetector
        from database.db import Database
        
        job = self.current_job
        if not job:
            return {"success": False, "error": "No job selected"}
        
        # Determine ATS type from URL
        detector = ATSDetector()
        url = job.get("url", "")
        ats_type = detector.detect_from_url(url)
        platform = job.get("provider", "").lower()
        
        # Prepare job data for applier
        job_data = {
            "id": job.get("id", ""),
            "job_id": job.get("id", ""),
            "url": url,
            "apply_url": url,
            "title": job.get("title", ""),
            "company": job.get("company", ""),
        }
        
        # Load user profile from config
        user_profile = self._load_user_profile()
        job_data["user_data"] = user_profile
        
        # Route to appropriate applier
        if ats_type == "workday":
            from src.applier.workday_apply import WorkdayApplier
            applier = WorkdayApplier(headless=False)
            return await applier.apply(job_data)
        
        elif ats_type == "greenhouse":
            from src.applier.greenhouse_apply import apply_to_greenhouse
            return await apply_to_greenhouse(job_data, user_profile)
        
        elif ats_type == "lever":
            from src.applier.lever_apply import LeverApplier
            applier = LeverApplier(headless=False)
            success, screenshot = await applier.apply(
                job_url=url,
                job_id=job.get("id", ""),
                company=job.get("company", ""),
                position=job.get("title", ""),
                user_data=user_profile
            )
            return {"success": success, "screenshot_path": screenshot, "error": None}
        
        elif ats_type == "ashby":
            from src.applier.ashby_apply import AshbyApplier
            applier = AshbyApplier(headless=False)
            return await applier.apply(
                job_url=url,
                job_id=job.get("id", ""),
                company=job.get("company", "")
            )
        
        elif ats_type == "bamboohr":
            from src.applier.bamboohr_apply import BambooHRApply
            applier = BambooHRApply(headless=False)
            return await applier.apply(
                job_id=job.get("id", ""),
                job_url=url,
                company=job.get("company", ""),
                title=job.get("title", ""),
                applicant_data=user_profile
            )
        
        elif platform == "indeed" or ats_type is None:
            # Use Indeed Router for unknown ATS or Indeed jobs
            from src.applier.indeed_router import IndeedRouter
            router = IndeedRouter(headless=False)
            return await router.route_and_apply(job_data, user_profile)
        
        else:
            return {
                "success": False,
                "error": f"Unsupported ATS type: {ats_type or platform}"
            }

    def _evaluate_tailoring(self):
        """Запустить 5-уровневую оценку тейлоринга для текущей вакансии."""
        job = self.current_job
        if not job:
            messagebox.showwarning("Внимание", "Нет выбранной вакансии для оценки")
            return

        # job — словарь с данными вакансии
        job_dict = {
            "title":       job.get("title", ""),
            "company":     job.get("company", ""),
            "description": job.get("description", "") or "",  # description может быть None
        }

        # ── Открыть окно ──────────────────────────────────────────────────────
        win = tk.Toplevel(self.root)
        win.title("🎯 Оценка тейлоринга")
        win.geometry("560x520")
        win.configure(bg="#1e1e2e")
        win.grab_set()

        # Заголовок вакансии
        ttk.Label(
            win,
            text=f"{job_dict.get('title','?')} @ {job_dict.get('company','?')}",
            foreground="#cba6f7", font=("SF Pro Text", 12, "bold"),
        ).pack(pady=(12, 4))

        # Live-индикатор ||||
        bars_var = tk.StringVar(value="...")
        bars_lbl = tk.Label(
            win, textvariable=bars_var,
            bg="#1e1e2e", fg="#89b4fa",
            font=("Menlo", 32, "bold"),
        )
        bars_lbl.pack(pady=8)

        status_var = tk.StringVar(value="Запуск...")
        ttk.Label(win, textvariable=status_var, foreground="#6c7086").pack()

        # Область результатов
        result_frame = ttk.Frame(win, padding=8)
        result_frame.pack(fill=tk.BOTH, expand=True, padx=12, pady=8)

        result_text = tk.Text(
            result_frame,
            bg="#181825", fg="#cdd6f4",
            font=("Menlo", 10),
            relief=tk.FLAT, state=tk.DISABLED,
        )
        result_text.pack(fill=tk.BOTH, expand=True)

        def append_result(text: str):
            result_text.configure(state=tk.NORMAL)
            result_text.insert(tk.END, text + "\n")
            result_text.see(tk.END)
            result_text.configure(state=tk.DISABLED)

        ttk.Button(win, text="Закрыть", command=win.destroy).pack(pady=(0, 12))

        def format_result(role_key: str, bars: str, label: str, data: dict) -> str:
            lines = [f"\n{'─'*40}", f"{bars}  {label}"]
            if "error" in data:
                lines.append(f"  ❌ Ошибка: {data['error']}")
                return "\n".join(lines)

            if role_key == "ats":
                score = data.get("ats_score", "?")
                present = ", ".join(data.get("present_keywords", [])[:5])
                missing = ", ".join(data.get("missing_keywords", [])[:5])
                lines.append(f"  Скор: {score}%")
                if present: lines.append(f"  ✅ Есть: {present}")
                if missing: lines.append(f"  ❌ Нет:  {missing}")
                lines.append(f"  → {data.get('verdict','')}")

            elif role_key == "hr":
                lines.append(f"  Скор: {data.get('hr_score','?')}/10  |  AI: {data.get('ai_suspicion','?')}")
                issues = data.get("authenticity_issues", [])
                if issues: lines.append(f"  ⚠️  {'; '.join(issues[:3])}")
                lines.append(f"  → {data.get('verdict','')}")

            elif role_key == "recruiter":
                lines.append(f"  Фит: {data.get('fit_score','?')}/10  |  Карьера: {data.get('career_coherence','?')}")
                concerns = data.get("concerns", [])
                if concerns: lines.append(f"  ⚠️  {'; '.join(concerns[:2])}")
                lines.append(f"  → {data.get('verdict','')}")

            elif role_key == "manager":
                d1 = "✅" if data.get("day_one_ready") else "❌"
                lines.append(f"  Технически: {data.get('technical_score','?')}/10  |  День1: {d1}")
                lines.append(f"  → {data.get('verdict','')}")

            elif role_key == "devil":
                flags = data.get("red_flags", [])
                for f in flags[:4]:
                    sev = f.get("severity", "")
                    icon = "🔴" if sev == "high" else "🟡" if sev == "medium" else "🟢"
                    lines.append(f"  {icon} {f.get('detail','')}")
                lines.append(f"  Главное: {data.get('biggest_concern','')}")

            return "\n".join(lines)

        # ── Флаг живости окна (защита от закрытия во время работы потока) ─────────
        win_alive = [True]
        win.protocol("WM_DELETE_WINDOW", lambda: (win_alive.__setitem__(0, False), win.destroy()))

        def safe_after(fn):
            """Вызвать fn в main thread только если окно ещё открыто."""
            if win_alive[0]:
                self.root.after(0, fn)

        def worker():
            from src.ai.high_effort import TailoringRunner, load_profile_context

            profile_ctx = load_profile_context()
            runner = TailoringRunner(job_dict, profile_ctx)

            # ── Параллельный запуск всех 5 ролей через asyncio.gather (~5 сек вместо ~25)
            safe_after(lambda: (bars_var.set("..."), status_var.set("⏳ Опрашиваем 5 ролей параллельно...")))
            all_results = runner.run_all_parallel()  # {role_key: result_dict}

            # Вывести результаты в порядке ROLES (ats → hr → recruiter → manager → devil)
            for role_key, bars, label in runner.ROLES:
                result = all_results.get(role_key, {"error": "no result"})
                formatted = format_result(role_key, bars, label, result)
                safe_after(lambda t=formatted: append_result(t))

            safe_after(lambda: (bars_var.set("|||||"), status_var.set("✅ Готово")))
            self.status_var.set(f"🎯 Тейлоринг оценён: {job_dict.get('title','')} @ {job_dict.get('company','')}")

        threading.Thread(target=worker, daemon=True).start()
