Tool result: """
Extract and convert Deep Research report to Markdown.
"""

import re
from datetime import datetime
from pathlib import Path
from typing import Optional

from html2text import HTML2Text
from loguru import logger
from slugify import slugify


class ReportExtractor:
    """Extract report from Gemini Deep Research page and convert to Markdown."""

    def __init__(self):
        self.h2t = HTML2Text()
        self.h2t.ignore_links = False
        self.h2t.ignore_images = False
        self.h2t.ignore_emphasis = False
        self.h2t.body_width = 0  # Don't wrap lines
        self.h2t.unicode_snob = True
        self.h2t.skip_internal_links = False

    def extract_report_html(self, page) -> Optional[str]:
        """
        Extract the report HTML from the Gemini page.
        
        The report is typically in a specific container after Deep Research completes.
        """
        # Try multiple selectors for the report content
        selectors = [
            # Main response container
            "response-element",
            # Fallback: message content
            ".message-content",
            # Fallback: response container
            "[data-response]",
            # Generic response area
            "model-response",
            # Last resort: main content area
            ".response-container",
        ]
        
        for selector in selectors:
            try:
                element = page.query_selector(selector)
                if element:
                    html = element.inner_html()
                    if html and len(html) > 100:  # Ensure we got meaningful content
                        logger.info(f"Found report content with selector: {selector}")
                        return html
            except Exception as e:
                logger.debug(f"Selector {selector} failed: {e}")
                continue
        
        # If no specific selector works, try to get the full page content
        logger.warning("Could not find specific report container, extracting full page")
        return page.content()

    def html_to_markdown(self, html: str) -> str:
        """Convert HTML to Markdown using html2text."""
        try:
            markdown = self.h2t.handle(html)
            # Clean up excessive newlines
            markdown = re.sub(r'\n{3,}', '\n\n', markdown)
            return markdown.strip()
        except Exception as e:
            logger.error(f"Failed to convert HTML to Markdown: {e}")
            return html

    def generate_filename(self, query: str) -> str:
        """Generate a safe filename from the query."""
        date_str = datetime.now().strftime("%Y-%m-%d")
        slug = slugify(query, max_length=50)
        return f"{date_str}_{slug}.md"

    def create_report(
        self,
        markdown_content: str,
        query: str,
        output_dir: Path = None
    ) -> Path:
        """
        Create a Markdown report file with frontmatter.
        
        Args:
            markdown_content: The report content in Markdown
            query: The original search query
            output_dir: Output directory (default: ~/Desktop/Personal Researchs/)
            
        Returns:
            Path to the created file
        """
        if output_dir is None:
            output_dir = Path.home() / "Desktop" / "Personal Researchs"
        
        output_dir.mkdir(parents=True, exist_ok=True)
        
        # Create frontmatter
        date_str = datetime.now().strftime("%Y-%m-%d")
        frontmatter = f"""---
date: {date_str}
query: "{query}"
---

"""
        
        # Combine frontmatter and content
        full_content = frontmatter + markdown_content
        
        # Generate filename and save
        filename = self.generate_filename(query)
        output_path = output_dir / filename
        
        # Handle duplicate filenames
        counter = 1
        while output_path.exists():
            base = filename.rsplit(".", 1)[0]
            output_path = output_dir / f"{base}_{counter}.md"
            counter += 1
        
        output_path.write_text(full_content, encoding="utf-8")
        logger.info(f"Report saved to: {output_path}")
        
        return output_path

    def extract_and_save(self, page, query: str) -> Optional[Path]:
        """
        Full extraction pipeline: extract HTML, convert, save.
        
        Args:
            page: Playwright page object
            query: The original search query
            
        Returns:
            Path to saved report or None if extraction failed
        """
        logger.info("Extracting report from page...")
        
        html = self.extract_report_html(page)
        if not html:
            logger.error("Failed to extract report HTML")
            return None
        
        logger.info("Converting HTML to Markdown...")
        markdown = self.html_to_markdown(html)
        
        if not markdown or len(markdown) < 50:
            logger.error("Converted markdown is too short, extraction may have failed")
            return None
        
        logger.info("Saving report...")
        output_path = self.create_report(markdown, query)
        
        return output_path
