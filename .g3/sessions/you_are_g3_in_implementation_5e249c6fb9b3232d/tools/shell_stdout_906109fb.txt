def _parse_greenhouse_url(url: str) -> Tuple[Optional[str], Optional[str]]:
    """Извлечь board_token и job_id из любого Greenhouse URL."""
    patterns = [
        r'boards-api\.greenhouse\.io/v1/boards/([^/?#]+)/jobs/(\d+)',
        r'(?:boards|job-boards)\.greenhouse\.io/([^/?#]+)/jobs/(\d+)',
        r'([^./?#]+)\.greenhouse\.io/jobs/(\d+)',
    ]
    for pat in patterns:
        m = re.search(pat, url)
        if m:
            return m.group(1), m.group(2)
    return None, None


class GreenhouseApplier:
    """
    Подаёт заявку на Greenhouse вакансию через patchright stealth-браузер.

    Открывает job-boards.greenhouse.io/{token}/jobs/{id},
    заполняет форму через Playwright, отправляет Submit.
    """

    def __init__(self):
        self.bank = AnswerBank()
        self.ai = AIAnswerer(profile=self.bank._profile)
        self._hitl = HumanReviewPrompter()  # bot_mode читается из BOT_MODE env

    def _page_url(self, apply_url: str) -> str:
        """Конвертировать любой Greenhouse URL в job-boards.greenhouse.io для браузера."""
        token, job_id = _parse_greenhouse_url(apply_url)
        if token and job_id:
            return f"https://job-boards.greenhouse.io/{token}/jobs/{job_id}"
        # Если не распарсили — используем как есть (браузер сам сделает редиректы)
        return apply_url

    async def apply_http(self, job: Job, resume_path: str = None) -> Dict[str, Any]:
        """
        Подать заявку через Playwright stealth-браузер.
        Метод называется apply_http для совместимости с IndeedRouter и GreenhouseScraper.
        """
        self._job_context = {
            "job_title": job.title,
            "company": job.company,
            "url": job.apply_url or "",
        }
        resume_file = Path(resume_path) if resume_path else RESUME_PATH
        if not resume_file.exists():
            logger.error(f"❌ Greenhouse: резюме не найдено: {resume_file}")
            return {"success": False, "error": "resume_not_found"}

        apply_url = job.apply_url
        if not apply_url:
            logger.error(f"❌ Greenhouse: нет apply_url у вакансии {job.id}")
            return {"success": False, "error": "no_apply_url"}

        page_url = self._page_url(apply_url)
        logger.info(f"🟢 Greenhouse apply: {job.title} @ {job.company} → {page_url}")

        try:
            async with StealthBrowser(headed=True) as browser:
                async with await browser.new_context() as ctx:
                    page = await browser.new_page(ctx)
                    await page.goto(page_url, wait_until="domcontentloaded", timeout=20000)
                    await asyncio.sleep(gaussian_delay(2.0, 3.5))

                    # Стандартные поля (React UI — по id)
                    await self._fill(page, SELECTORS["first_name"],
                                     self.bank.get_answer("first_name") or "")
                    await self._fill(page, SELECTORS["last_name"],
                                     self.bank.get_answer("last_name") or "")
                    await self._fill(page, SELECTORS["email"],
                                     self.bank.get_answer("email") or "")
                    await self._fill(page, SELECTORS["phone"],
                                     self.bank.get_answer("phone") or "")

                    # Country — combobox (React dropdown)
                    await self._fill_combobox(page, SELECTORS["country"],
                                             self.bank.get_answer("country") or "United States")

                    # EEO / Demographic fields (по прямому id, не question_ префикс)
                    for field_id in ("gender", "hispanic_ethnicity", "veteran_status", "disability_status"):
                        answer = self.bank.get_answer(field_id)
                        if answer:
                            await self._fill_combobox(page, f"#{field_id}", answer)

                    # Загрузить резюме
                    await self._upload_resume(page, resume_file)

                    # Cover letter если есть
                    if COVER_LETTER_PATH.exists():
                        logger.debug(f"📎 Greenhouse: загружаем cover letter: {COVER_LETTER_PATH}")
                        await self._upload_cover_letter(page, COVER_LETTER_PATH)

                    # Кастомные вопросы
                    await self._fill_custom_questions(page)

                    # Скриншот ДО submit — чтобы видеть заполненную форму

                    pre_screenshot = SCREENSHOTS_DIR / f"gh_{safe_company}_{job_id_short}_pre.png"
                    await page.screenshot(path=str(pre_screenshot), full_page=True)
                    logger.debug(f"📸 Pre-submit screenshot: {pre_screenshot}")

                    # ── Проверка пустых полей перед Submit ─────────────────
                    empty_required = await self._find_empty_required_fields(page)
                    if empty_required:
                        logger.warning(
                            f"⚠️ Greenhouse: {len(empty_required)} незаполненных обязательных полей:"
                        )
                        for lbl in empty_required:
                            logger.warning(f"   ❌ {lbl}")
                        logger.warning(
                            f"⏸️  Пауза 90 сек — заполните поля вручную в браузере.\n"
                            f"   Скриншот: {pre_screenshot.name}"
                        )
                        await asyncio.sleep(90)
                        # Пересканируем после паузы
                        still_empty = await self._find_empty_required_fields(page)
                        if still_empty:
                            logger.error(f"🛑 Поля всё ещё пусты: {still_empty} — Submit отменён")
                            return {
                                "success": False, "error": "required_fields_empty",
                                "missing": still_empty, "screenshot_path": str(pre_screenshot),
                            }
                        logger.info("✅ Все поля заполнены после паузы — продолжаем Submit")
                    # ────────────────────────────────────────────────────────

                    # Submit
                    submit_btn = await page.query_selector(SELECTORS["submit"])
                    if not submit_btn:
                        logger.warning("⚠️ Greenhouse: кнопка Submit не найдена")
                        return {"success": False, "error": "no_submit_button",
                                "screenshot_path": str(pre_screenshot)}
