async def apply_http(self, job: Job, resume_path: str = None) -> Dict[str, Any]:
        """
        Подать заявку через Playwright stealth-браузер.
        Метод называется apply_http для совместимости с IndeedRouter и GreenhouseScraper.
        """
        self._job_context = {
            "job_title": job.title,
            "company": job.company,
            "url": job.apply_url or "",
        }
        resume_file = Path(resume_path) if resume_path else RESUME_PATH
        if not resume_file.exists():
            logger.error(f"❌ Greenhouse: резюме не найдено: {resume_file}")
            return {"success": False, "error": "resume_not_found"}

        apply_url = job.apply_url
        if not apply_url:
            logger.error(f"❌ Greenhouse: нет apply_url у вакансии {job.id}")
            return {"success": False, "error": "no_apply_url"}

        page_url = self._page_url(apply_url)
        logger.info(f"🟢 Greenhouse apply: {job.title} @ {job.company} → {page_url}")

        try:
            async with StealthBrowser(headed=True) as browser:
                async with await browser.new_context() as ctx:
                    page = await browser.new_page(ctx)
                    await page.goto(page_url, wait_until="domcontentloaded", timeout=20000)
                    await asyncio.sleep(gaussian_delay(2.0, 3.5))

                    # Стандартные поля (React UI — по id)
                    await self._fill(page, SELECTORS["first_name"],
                                     self.bank.get_answer("first_name") or "")
                    await self._fill(page, SELECTORS["last_name"],
                                     self.bank.get_answer("last_name") or "")
                    await self._fill(page, SELECTORS["email"],
                                     self.bank.get_answer("email") or "")
                    await self._fill(page, SELECTORS["phone"],
                                     self.bank.get_answer("phone") or "")

                    # Country — combobox (React dropdown)
                    await self._fill_combobox(page, SELECTORS["country"],
                                             self.bank.get_answer("country") or "United States")

                    # EEO / Demographic fields (по прямому id, не question_ префикс)
                    for field_id in ("gender", "hispanic_ethnicity", "veteran_status", "disability_status"):
                        answer = self.bank.get_answer(field_id)
                        if answer:
                            await self._fill_combobox(page, f"#{field_id}", answer)

                    # Загрузить резюме
                    await self._upload_resume(page, resume_file)

                    # Cover letter если есть
                    if COVER_LETTER_PATH.exists():
                        logger.debug(f"📎 Greenhouse: загружаем cover letter: {COVER_LETTER_PATH}")
                        await self._upload_cover_letter(page, COVER_LETTER_PATH)

                    # Кастомные вопросы
                    await self._fill_custom_questions(page)

                    # Скриншот ДО submit — чтобы видеть заполненную форму
                    safe_company = job.company.lower().replace(' ', '_').replace('/', '_')
                    job_id_short = str(job.id)[:8]
                    pre_screenshot = SCREENSHOTS_DIR / f"gh_{safe_company}_{job_id_short}_pre.png"
                    await page.screenshot(path=str(pre_screenshot), full_page=True)
                    logger.debug(f"📸 Pre-submit screenshot: {pre_screenshot}")

                    # ── Проверка пустых полей перед Submit ─────────────────
                    empty_required = await self._find_empty_required_fields(page)
                    if empty_required:
                        logger.warning(
                            f"⚠️ Greenhouse: {len(empty_required)} незаполненных обязательных полей:"
                        )
                        for lbl in empty_required:
                            logger.warning(f"   ❌ {lbl}")
                        logger.warning(
                            f"⏸️  Пауза 90 сек — заполните поля вручную в браузере.\n"
                            f"   Скриншот: {pre_screenshot.name}"
                        )
                        await asyncio.sleep(90)
                        # Пересканируем после паузы
                        still_empty = await self._find_empty_required_fields(page)
                        if still_empty:
                            logger.error(f"🛑 Поля всё ещё пусты: {still_empty} — Submit отменён")
                            return {
                                "success": False, "error": "required_fields_empty",
                                "missing": still_empty, "screenshot_path": str(pre_screenshot),
                            }
                        logger.info("✅ Все поля заполнены после паузы — продолжаем Submit")
                    # ────────────────────────────────────────────────────────

                    # Submit
                    submit_btn = await page.query_selector(SELECTORS["submit"])
                    if not submit_btn:
                        logger.warning("⚠️ Greenhouse: кнопка Submit не найдена")
                        return {"success": False, "error": "no_submit_button",
                                "screenshot_path": str(pre_screenshot)}

                    # Явный scroll к кнопке Submit + пауза для рендера.
                    # Страховка: human_click_safe тоже делает scroll, но здесь
                    # даём браузеру время перерисовать layout после заполнения полей.
                    await submit_btn.scroll_into_view_if_needed()
                    await asyncio.sleep(0.5)

                    submit_text = await submit_btn.inner_text()
                    logger.info(f"🖱️ Greenhouse: Submit найден ('{submit_text.strip()}'), кликаем...")
                    await human_click_safe(page, submit_btn)
                    await asyncio.sleep(gaussian_delay(3.0, 5.0))

                    # ── Security Code (Greenhouse Email Verification) ────────
                    # После первого submit Greenhouse может показать поле для
                    # ввода кода из письма — проверяем это и обрабатываем через HITL
                    security_handled = await self._handle_security_code(page, job)
                    if security_handled:
                        # После ввода кода ждём ещё немного
                        await asyncio.sleep(gaussian_delay(3.0, 5.0))
                    # ────────────────────────────────────────────────────────

                    # Проверить успех: URL меняется на /confirmation
                    current_url = page.url
                    success_el = await page.query_selector(SELECTORS["success"])
                    logger.debug(f"Greenhouse: URL после submit = {current_url}")

                    # Скриншот ПОСЛЕ submit — видим результат
                    post_screenshot = SCREENSHOTS_DIR / f"gh_{safe_company}_{job_id_short}_post.png"
                    await page.screenshot(path=str(post_screenshot), full_page=True)
                    logger.info(f"📸 Post-submit screenshot: {post_screenshot}")

                    success_confirmed = (
                        "/confirmation" in current_url
                        or "thank" in current_url.lower()
                        or success_el is not None
                    )
                    if success_confirmed:
                        logger.success(f"✅ Greenhouse: подали на {job.title} @ {job.company}")
                        return {"success": True, "screenshot_path": str(post_screenshot)}

                    logger.warning(
                        f"⚠️ Greenhouse: успех не подтверждён. URL={current_url}  "
                        f"success_el={'да' if success_el else 'нет'}. "
                        f"Смотри скриншоты: {pre_screenshot.name} / {post_screenshot.name}"
                    )
                    return {"success": False, "warning": "no_confirmation",
                            "screenshot_path": str(post_screenshot)}

        except Exception as e:
            logger.error(f"❌ Greenhouse apply error: {e}")
            return {"success": False, "error": str(e)}

    # ─── Helpers ────────────────────────────────────────────────────────────

    async def _find_empty_required_fields(self, page) -> list:
        """
        Найти все видимые обязательные поля которые пустые.
        Возвращает список label-строк пустых полей.
        """
        empty = []
        seen_ids = set()

        # Все input/textarea/select с aria-required или required
        candidates = await page.query_selector_all(
            "input[aria-required='true'], input[required], "
            "select[aria-required='true'], select[required], "
            "textarea[aria-required='true'], textarea[required]"
        )
        for el in candidates:
            el_type = (await el.get_attribute("type") or "text").lower()
            if el_type in ("hidden", "file", "submit", "button", "reset", "image"):
                continue
            if not await el.is_visible():
                continue
            # Combobox инпуты (role='combobox') обрабатываются отдельно ниже —
            # input_value() для react-select всегда "" даже после выбора
            el_role = (await el.get_attribute("role") or "").lower()
            if el_role == "combobox":
                continue
            el_id = await el.get_attribute("id") or ""
            if el_id in seen_ids:
                continue
            seen_ids.add(el_id)

            tag = await el.evaluate("e => e.tagName.toLowerCase()")
            if tag == "select":
                val = await el.evaluate(
                    "e => e.selectedIndex <= 0 ? '' : (e.options[e.selectedIndex]?.value || '')"
                )
            else:
                val = await el.input_value()

            if not val.strip():
                label = await self._get_label(page, el)
                empty.append(label or el_id or "unknown")

        # Также проверяем combobox (React Select — не имеет native required)
        # Их обязательность определяем по aria-required на самом input
        # ВАЖНО: после выбора input_value() = "" (react-select очищает текстовый инпут).
        # Реальное значение — в [class*='single-value'] контейнере рядом с инпутом.
        combos = await page.query_selector_all("input[role='combobox'][aria-required='true']")
