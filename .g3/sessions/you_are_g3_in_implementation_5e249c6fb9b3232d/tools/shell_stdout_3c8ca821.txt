    66	def _parse_greenhouse_url(url: str) -> Tuple[Optional[str], Optional[str]]:
    67	    """Извлечь board_token и job_id из любого Greenhouse URL."""
    68	    patterns = [
    69	        r'boards-api\.greenhouse\.io/v1/boards/([^/?#]+)/jobs/(\d+)',
    70	        r'(?:boards|job-boards)\.greenhouse\.io/([^/?#]+)/jobs/(\d+)',
    71	        r'([^./?#]+)\.greenhouse\.io/jobs/(\d+)',
    72	    ]
    73	    for pat in patterns:
    74	        m = re.search(pat, url)
    75	        if m:
    76	            return m.group(1), m.group(2)
    77	    return None, None

    93	    def _page_url(self, apply_url: str) -> str:
    94	        """Конвертировать любой Greenhouse URL в job-boards.greenhouse.io для браузера."""
    95	        token, job_id = _parse_greenhouse_url(apply_url)
    96	        if token and job_id:
    97	            return f"https://job-boards.greenhouse.io/{token}/jobs/{job_id}"
    98	        # Если не распарсили — используем как есть (браузер сам сделает редиректы)
    99	        return apply_url

   101	    async def apply_http(self, job: Job, resume_path: str = None) -> Dict[str, Any]:
