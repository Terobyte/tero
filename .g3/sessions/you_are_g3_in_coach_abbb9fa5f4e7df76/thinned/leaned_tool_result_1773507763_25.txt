"""
StepAuditLog — хранит аудит всех ответов и коррекций.
Пишет в data/screening_audit.jsonl (append-only).
"""
from __future__ import annotations
import json
import time
from pathlib import Path
from typing import Optional
from loguru import logger

from src.applier.universal_screening.batch_validator import Correction

AUDIT_PATH = Path("data/screening_audit.jsonl")


class StepAuditLog:
    """Append-only audit log для screening answers."""

    def __init__(self, job_id: str = "", job_url: str = ""):
        self.job_id = job_id
        self.job_url = job_url
        self._entries: list[dict] = []

    def record_answer(self, question: str, field_type: str,
                      answer: str, intent: Optional[str] = None,
                      is_critical: bool = False):
        self._entries.append({
            "ts": time.time(), "event": "answer",
            "job_id": self.job_id, "question": question,
            "field_type": field_type, "answer": answer,
            "intent": intent, "is_critical": is_critical,
        })

    def record_correction(self, correction: Correction, success: bool):
        self._entries.append({
            "ts": time.time(), "event": "correction",
            "job_id": self.job_id,
            "question": correction.question_text,
            "from": correction.current_answer,
            "to": correction.corrected_answer,
            "reason": correction.reason,
            "severity": correction.severity,
            "success": success,
        })

    def flush(self):
        """Записать все записи в JSONL файл."""
        try:
            AUDIT_PATH.parent.mkdir(parents=True, exist_ok=True)
            with open(AUDIT_PATH, "a", encoding="utf-8") as f:
                for entry in self._entries:
                    f.write(json.dumps(entry, ensure_ascii=False) + "\n")
            self._entries.clear()
        except Exception as e:
            logger.debug(f"AuditLog flush error: {e}")
