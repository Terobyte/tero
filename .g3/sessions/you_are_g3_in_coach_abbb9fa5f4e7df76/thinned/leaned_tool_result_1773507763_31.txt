"""
AIAnswerer — отвечает на screening вопросы с помощью LLM.

Интеграции:
- VectorKnowledgeBase для семантического поиска похожих вопросов
- NaturalLanguageGenerator для уникальных ответов на textarea
- Cost tracking через _track()
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Optional

from loguru import logger


@dataclass
class AIAnswer:
    """Ответ от AI на screening вопрос."""
    answer: str
    confidence: str  # "high", "medium", "low"
    reason: str = ""
    cached: bool = False


class AIAnswerer:
    """
    Отвечает на screening вопросы с помощью LLM.

    Usage:
        answerer = AIAnswerer(profile, llm_client)
        answer = await answerer.ask("Do you have Python experience?")
    """

    def __init__(self, profile: dict, llm_client=None, kb=None):
        """
        Args:
            profile: Профиль кандидата из profile.yaml
            llm_client: LLM клиент (KimiClient, GeminiClient, etc.)
            kb: VectorKnowledgeBase (опционально)
        """
        self.profile = profile
        self._llm = llm_client
        self._kb = kb
        self._nlg = None

        # Инициализировать NLG если есть LLM
        if llm_client:
            from src.ai.nlg import NaturalLanguageGenerator
            self._nlg = NaturalLanguageGenerator(llm_client, kb=kb)

    async def ask(
        self,
        question: str,
        field_type: str = "radio",
        options: Optional[list[str]] = None,
        context: Optional[dict] = None,
    ) -> Optional[AIAnswer]:
        """
        Ответить на screening вопрос.

        Args:
            question: Текст вопроса
            field_type: Тип поля (radio, checkbox, select, text, textarea)
            options: Список опций для radio/select
            context: Контекст вакансии {title, company, requirements}

        Returns:
            AIAnswer или None если не удалось ответить
        """
        # 1. Попробовать семантический поиск в VectorKB
        if self._kb:
            similar = self._kb.find_similar_question(question, k=1, threshold=0.85)
            if similar:
                best = similar[0]
                logger.info(f"🧠 VectorKB hit (score={best['score']:.2f}): '{question[:40]}'")
                return AIAnswer(
                    answer=best["answer"],
                    confidence="high",
                    reason=f"VectorKB semantic match (score={best['score']:.2f})",
                    cached=True,
                )

        # 2. Для textarea — использовать NLG
        if field_type == "textarea" and self._nlg:
            nlg_result = await self._nlg.generate(
                question=question,
                job=context or {},
                field_type=field_type,
                profile=self.profile,
                requirements=context.get("requirements") if context else None,
            )
            if nlg_result.confidence != "low":
                return AIAnswer(
                    answer=nlg_result.answer,
                    confidence=nlg_result.confidence,
                    reason=f"NLG ({nlg_result.word_count}w, {len(nlg_result.keywords_used)} kw)",
                )

        # 3. Стандартный LLM запрос
        if not self._llm:
            return None

        prompt = self._build_prompt(question, field_type, options, context)
        try:
            response = await self._llm.generate(prompt)
            answer = self._parse_response(response, field_type, options)

            # Track cost
            from src.ai.llm_client import _track
            _track("unknown", getattr(self._llm, "MODEL", "unknown"), response, module="ai_answerer")

            return AIAnswer(
                answer=answer,
                confidence="medium",
                reason="LLM generated",
            )
        except Exception as e:
            logger.error(f"AIAnswerer error: {e}")
            return None

    def _build_prompt(
        self,
        question: str,
        field_type: str,
        options: Optional[list[str]],
        context: Optional[dict],
    ) -> str:
        """Построить промпт для LLM."""
        lines = [
            "You are answering a job application screening question as the candidate.",
            f"Question: {question}",
        ]

        if options:
            lines.append(f"Options: {', '.join(options)}")

        if field_type == "radio":
            lines.append("Choose ONE option from the list. Reply with the exact option text only.")
        elif field_type == "checkbox":
            lines.append("Select ALL that apply. Reply with the exact option texts, comma-separated.")
        elif field_type == "select":
            lines.append("Choose ONE option. Reply with the exact option text only.")
        elif field_type == "text":
            lines.append("Reply with a brief answer (1-2 sentences).")
        elif field_type == "textarea":
            lines.append("Reply with a detailed answer (2-4 sentences).")

        # Add profile context
        if self.profile:
            exp = self.profile.get("experience", {})
            skills = [k.replace("years_", "") for k in exp if k.startswith("years_") and exp[k]]
            if skills:
                lines.append(f"\nCandidate skills: {', '.join(skills[:10])}")

        # Add job context
        if context:
            if context.get("title"):
                lines.append(f"\nJob: {context['title']} at {context.get('company', 'company')}")

        lines.append("\nAnswer:")
        return "\n".join(lines)

    def _parse_response(
        self,
        response: str,
        field_type: str,
        options: Optional[list[str]],
    ) -> str:
        """Parse LLM response to extract the answer."""
        answer = response.strip()

        # For radio/select, try to match to an option
        if field_type in ("radio", "select") and options:
            answer_lower = answer.lower()
            for opt in options:
                if opt.lower() in answer_lower or answer_lower in opt.lower():
                    return opt
            # Fallback to first option if no match
            if options:
                return options[0]

        # For checkbox, extract all matched options
        if field_type == "checkbox" and options:
            matched = []
            answer_lower = answer.lower()
            for opt in options:
                if opt.lower() in answer_lower:
                    matched.append(opt)
            if matched:
                return ", ".join(matched)

        return answer

    async def batch_ask(
        self,
        questions: list[dict],
        context: Optional[dict] = None,
    ) -> list[Optional[AIAnswer]]:
        """
        Ответить на несколько вопросов одновременно.

        Args:
            questions: List of {question, field_type, options}
            context: Job context

        Returns:
            List of AIAnswer or None
        """
        results = []
        for q in questions:
            answer = await self.ask(
                question=q.get("question", ""),
                field_type=q.get("field_type", "radio"),
                options=q.get("options"),
                context=context,
            )
            results.append(answer)
        return results
