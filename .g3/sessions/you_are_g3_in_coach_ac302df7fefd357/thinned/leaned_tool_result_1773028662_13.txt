Tool result: """
IndeedApplier - Self-healing job application handler.

Phase 2 of the Indeed MTD implementation - uses SelectorEngine for all
element finding instead of hardcoded CSS selectors.

Key features:
- 4-tier resolution (cache → semantic → structural → LLM)
- Early exit on error modals (already_applied, job_expired)
- Semantic fingerprint for stuck detection
- Dynamic reCAPTCHA sitekey extraction
- Integration with IndeedScreeningHandler for questions
"""

from __future__ import annotations

import asyncio
import logging
import re
import time
from pathlib import Path
from typing import TYPE_CHECKING, Any, Dict, List, Optional, Tuple

if TYPE_CHECKING:
    from patchright.async_api import Page, Frame, Locator

from .selector_engine import SelectorEngine
from .selector_health import SelectorHealthMonitor

logger = logging.getLogger(__name__)


class IndeedApplier:
    """
    Self-healing Indeed job application handler.
    
    Uses SelectorEngine for all element finding, providing resilience
    against Indeed's Moving Target Defense (MTD) tactics.
    
    Usage:
        applier = IndeedApplier(selector_engine=engine, health_monitor=monitor)
        result = await applier.apply_on_page(job, page, resume_path)
    """
    
    # Apply iframe domains (both classic and Smart Apply)
    APPLY_DOMAINS = ("apply.indeed.com", "smartapply.indeed.com")
    
    def __init__(
        self,
        selector_engine: Optional[SelectorEngine] = None,
        health_monitor: Optional[SelectorHealthMonitor] = None,
    ) -> None:
        """
        Initialize IndeedApplier with optional dependency injection.
        
        Args:
            selector_engine: Optional pre-configured SelectorEngine.
                            If None, a new instance is created.
            health_monitor: Optional health monitor for degradation tracking.
                           If None, no health tracking is performed.
        """
        self.selector = selector_engine or SelectorEngine()
        self.health = health_monitor
        
        # Import screening handler lazily to avoid circular imports
        from .indeed_screening import IndeedScreeningHandler
        self.screening = IndeedScreeningHandler()
        
        # Stuck detection state
        self._last_fingerprint: Optional[str] = None
        self._stuck_count: int = 0
        self._max_stuck_iterations: int = 3
    
    # ═══════════════════════════════════════════════════════════════════════════
    # PUBLIC API
    # ═══════════════════════════════════════════════════════════════════════════
    
    async def apply_on_page(
        self,
        job,  # Job dataclass from base_scraper
        page: "Page",
        resume_path: str,
        **kwargs,
    ) -> Dict[str, Any]:
        """
        Apply to a job on Indeed using the self-healing selector system.
        
        Args:
            job: Job object with URL, title, company, etc.
            page: Playwright Page instance (already navigated to job page)
            resume_path: Path to resume file for upload
            **kwargs: Additional arguments (context, etc.)
            
        Returns:
            Dict with:
                - success: bool
                - error: Optional[str]
                - skip: bool (if early exit for already_applied/job_expired)
                - open_url: Optional[str] (for manual fill on failure)
                - degraded_selectors: List[str] (if degradation detected)
        """
        context = kwargs.get("context", {})
        
        # ── Phase 4: Early exit on error modals ─────────────────────────────────
        if await self.selector.find(page, "already_applied", context="apply_start"):
            logger.info(f"Already applied to job: {job.url}")
            return {"success": False, "error": "already_applied", "skip": True}
        
        if await self.selector.find(page, "job_expired", context="apply_start"):
            logger.info(f"Job expired/removed: {job.url}")
            return {"success": False, "error": "job_expired", "skip": True}
        
        # ── Check degradation ───────────────────────────────────────────────────
        degraded_selectors: List[str] = []
        if self.health:
            alerts = self.health.check_degradation()
            if alerts:
                degraded_selectors = [a.intent for a in alerts]
                logger.warning(f"⚠️ Degraded selectors: {degraded_selectors}")
        
        # ── Navigate to apply flow ──────────────────────────────────────────────
        try:
            # Click apply button on job page
            apply_btn = await self.selector.find(page, "apply_button", context="job_page")
            if not apply_btn:
                logger.error("Could not find apply button")
                return {
                    "success": False,
                    "error": "no_apply_button",
                    "open_url": job.url,
                    "degraded_selectors": degraded_selectors,
                }
            
            await apply_btn.click()
            await asyncio.sleep(1.0)  # Wait for iframe to load
            
            # Find apply iframe
            target = await self._find_apply_frame(page)
            if not target:
                logger.error("Could not find apply iframe")
                return {
                    "success": False,
                    "error": "no_apply_iframe",
                    "open_url": job.url,
                    "degraded_selectors": degraded_selectors,
                }
            
            # ── Main apply loop ──────────────────────────────────────────────────
            max_iterations = 20
            for iteration in range(max_iterations):
                # Check for stuck state
                fingerprint = await self._semantic_fingerprint(target)
                if fingerprint == self._last_fingerprint:
                    self._stuck_count += 1
                    if self._stuck_count >= self._max_stuck_iterations:
                        logger.warning("Stuck detection - same state for multiple iterations")
                        break
                else:
                    self._stuck_count = 0
                    self._last_fingerprint = fingerprint
                
                # Check for success
                if await self._is_success(page, target):
                    logger.info(f"Application submitted successfully for {job.url}")
                    return {"success": True}
                
                # Handle resume selection/upload
                resume_result = await self._handle_resume(target, resume_path)
                if resume_result == "uploaded":
                    await asyncio.sleep(0.5)
                    continue
                elif resume_result == "error":
                    logger.error("Resume handling failed")
                    return {
                        "success": False,
                        "error": "resume_error",
                        "open_url": job.url,
                        "degraded_selectors": degraded_selectors,
                    }
                
                # Fill form fields
                fill_result = await self._fill_fields(target, context)
                if fill_result == "error":
                    return {
                        "success": False,
                        "error": "fill_error",
                        "open_url": job.url,
                        "degraded_selectors": degraded_selectors,
                    }
                
                # Handle screening questions
                screening_result = await self.screening.handle_all_questions(target, context)
                if screening_result == "error":
                    logger.warning("Screening questions handling error")
                    # Continue anyway - might be optional
                
                # Handle radios (already semantic via fieldset)
                await self._handle_radios(target)
                
                # Check for visible CAPTCHA
                if await self._has_visible_captcha(target):
                    captcha_result = await self._solve_visible_captcha(page, target)
                    if captcha_result == "solved":
                        await asyncio.sleep(0.5)
                        continue
                    elif captcha_result == "manual_required":
                        return {
                            "success": False,
                            "error": "captcha_manual_required",
                            "open_url": job.url,
                            "degraded_selectors": degraded_selectors,
                        }
                
                # Click continue or submit
                if await self._has_submit_button(target):
                    click_result = await self._click_submit(target)
                else:
                    click_result = await self._click_continue(target)
                
                if click_result == "error":
                    logger.warning("Click continue/submit failed")
                    # Don't return error - might just be loading
                
                await asyncio.sleep(1.0)
            
            # Max iterations reached
            logger.warning(f"Max iterations ({max_iterations}) reached")
            return {
                "success": False,
                "error": "max_iterations",
                "open_url": job.url,
                "degraded_selectors": degraded_selectors,
            }
            
        except Exception as e:
            logger.error(f"Apply flow error: {e}", exc_info=True)
            return {
                "success": False,
                "error": str(e),
                "open_url": job.url,
                "degraded_selectors": degraded_selectors,
            }
    
    # ═══════════════════════════════════════════════════════════════════════════
    # PRIVATE HELPERS
    # ═══════════════════════════════════════════════════════════════════════════
    
    async def _find_apply_frame(self, page: "Page") -> Optional["Frame"]:
        """
        Find the apply iframe using SelectorEngine + URL fallback.
        
        Tier 1-4 via SelectorEngine first, then URL-based fallback
        covering both classic and Smart Apply domains.
        
        Args:
            page: Playwright Page instance
            
        Returns:
            Frame or None if not found
        """
        # Try SelectorEngine first
        iframe_el = await self.selector.find(page, "iframe_apply", context="apply_frame")
        if iframe_el:
            try:
                frame = await iframe_el.content_frame()
                if frame:
                    logger.debug("Found apply iframe via SelectorEngine")
                    return frame
            except Exception as e:
                logger.debug(f"SelectorEngine iframe content_frame failed: {e}")
        
        # URL-based fallback
        for frame in page.frames:
            if any(domain in frame.url for domain in self.APPLY_DOMAINS):
                logger.debug(f"Found apply iframe via URL fallback: {frame.url}")
                return frame
        
        logger.warning("Could not find apply iframe")
        return None
    
    async def _semantic_fingerprint(self, target: "Frame") -> str:
        """
        Generate a semantic fingerprint of the current form state.
        
        Used for stuck detection - if the fingerprint doesn't change
        across iterations, we're stuck.
        
        Args:
            target: Playwright Frame (iframe)
            
        Returns:
            Fingerprint string (sorted field labels joined by |)
        """
        try:
            return await target.evaluate("""
                () => {
                    const fields = [];
                    const els = document.querySelectorAll(
                        'input:not([type="hidden"]), select, textarea, fieldset, [role="radiogroup"]'
                    );
                    for (let i = 0; i < Math.min(els.length, 30); i++) {
                        const el = els[i];
                        const rect = el.getBoundingClientRect();
                        if (rect.width === 0 && rect.height === 0) continue;
                        const label = el.getAttribute('aria-label')
                            || (el.labels && el.labels[0] ? el.labels[0].textContent : '')
                            || (el.closest('fieldset')?.querySelector('legend')?.textContent || '')
                            || el.name || el.type || '';
                        fields.push(label.trim().toLowerCase().slice(0, 30));
                    }
                    return fields.sort().join('|');
                }
            """)
        except Exception as e:
            logger.debug(f"Fingerprint extraction failed: {e}")
            return ""
    
    async def _extract_sitekey(self, page: "Page", target: Optional["Frame"] = None) -> Optional[str]:
        """
        Extract reCAPTCHA sitekey from page frames or JS globals.
        
        Tries three methods:
        1. reCAPTCHA iframe URL parameter
        2. JS global window.invisibleReCaptchaKey
        3. iframe src attribute via JS query
        
        Args:
            page: Playwright Page instance
            target: Optional Frame (for checking both contexts)
            
        Returns:
            Sitekey string or None if not found
        """
        # 1. From reCAPTCHA iframe URL
        for frame in page.frames:
            if "recaptcha" in frame.url and ("anchor" in frame.url or "enterprise" in frame.url):
                match = re.search(r"[?&]k=([^&]+)", frame.url)
                if match:
                    logger.debug(f"Found sitekey from iframe URL: {match.group(1)[:20]}...")
                    return match.group(1)
        
        # 2. From JS global (Indeed sets this)
        contexts = [page]
        if target:
            contexts.append(target)
        
        for ctx in contexts:
            try:
                sitekey = await ctx.evaluate("window.invisibleReCaptchaKey")
                if sitekey and len(str(sitekey)) > 20:
                    logger.debug(f"Found sitekey from JS global: {str(sitekey)[:20]}...")
                    return str(sitekey)
            except Exception:
                pass
        
        # 3. From iframe src attribute (JS fallback)
        try:
            result = await page.evaluate("""
                () => {
                    const iframes = document.querySelectorAll('iframe[src*="recaptcha"]');
                    for (const f of iframes) {
                        const m = f.src.match(/[?&]k=([^&]+)/);
                        if (m) return m[1];
                    }
                    return null;
                }
            """)
            if result:
                logger.debug(f"Found sitekey from iframe src: {result[:20]}...")
                return result
        except Exception:
            pass
        
        logger.warning("Could not extract reCAPTCHA sitekey")
        return None
    
    async def _handle_resume(self, target: "Frame", resume_path: str) -> str:
        """
        Handle resume selection or upload.
        
        Args:
            target: Playwright Frame (iframe)
            resume_path: Path to resume file
            
        Returns:
            "selected" | "uploaded" | "error" | "not_found"
        """
        # Try to select existing resume
        resume_cards = await self.selector.find_all(target, "resume_card", context="resume_selection")
        if resume_cards:
            # Click first resume card
            try:
                await resume_cards[0].click()
                logger.debug("Selected existing resume")
                return "selected"
            except Exception as e:
                logger.debug(f"Resume card click failed: {e}")
        
        # Try upload
        upload_btn = await self.selector.find(target, "resume_upload", context="resume_upload")
        if upload_btn:
            try:
                # Use DataTransfer for file upload
                await self._upload_file_via_datatransfer(target, resume_path)
                logger.debug("Uploaded resume via DataTransfer")
                return "uploaded"
            except Exception as e:
                logger.error(f"Resume upload failed: {e}")
                return "error"
        
        return "not_found"
    
    async def _upload_file_via_datatransfer(self, target: "Frame", file_path: str) -> None:
        """
        Upload a file using DataTransfer JS injection.
        
        Args:
            target: Playwright Frame
            file_path: Path to the file to upload
        """
        path = Path(file_path)
        if not path.exists():
            raise FileNotFoundError(f"Resume file not found: {file_path}")
        
        # Find file input
        file_input = target.locator("input[type='file']").first
        if await file_input.count() == 0:
            raise RuntimeError("No file input found for resume upload")
        
        await file_input.set_input_files(str(path))
    
    async def _fill_fields(self, target: "Frame", context: Dict[str, Any]) -> str:
        """
        Fill form fields (phone, cover letter) using SelectorEngine.
        
        Args:
            target: Playwright Frame (iframe)
            context: Context dict with form data
            
        Returns:
            "success" | "error" | "no_fields"
        """
        filled_any = False
        
        # Phone input
        phone_input = await self.selector.find(target, "phone_input", context="form_field")
        if phone_input:
            phone_value = context.get("phone", "")
            if phone_value:
                try:
                    await phone_input.fill(phone_value)
                    logger.debug(f"Filled phone: {phone_value}")
                    filled_any = True
                except Exception as e:
                    logger.warning(f"Phone fill failed: {e}")
        
        # Cover letter
        cover_input = await self.selector.find(target, "cover_letter_input", context="form_field")
        if cover_input:
            cover_value = context.get("cover_letter", "")
            if cover_value:
                try:
                    await cover_input.fill(cover_value)
                    logger.debug("Filled cover letter")
                    filled_any = True
                except Exception as e:
                    logger.warning(f"Cover letter fill failed: {e}")
        
        return "success" if filled_any else "no_fields"
    
    async def _handle_radios(self, target: "Frame") -> None:
        """
        Handle radio buttons using semantic fieldset approach.
        
        This method does NOT use SelectorEngine - it's already semantic
        via fieldset/legend structure.
        
        Args:
            target: Playwright Frame (iframe)
        """
        try:
            fieldsets = await target.locator("fieldset").all()
            for fieldset in fieldsets:
                try:
                    # Get the legend text
                    legend = fieldset.locator("legend").first
                    if await legend.count() == 0:
                        continue
                    
                    legend_text = await legend.text_content()
                    if not legend_text:
                        continue
                    
                    # Find all radio buttons in this fieldset
                    radios = await fieldset.locator('input[type="radio"]').all()
                    if not radios:
                        continue
                    
                    # Click the first non-disabled radio
                    for radio in radios:
                        if await radio.is_enabled() and await radio.is_visible():
                            await radio.click()
                            logger.debug(f"Selected radio in fieldset: {legend_text.strip()[:30]}")
                            break
                except Exception as e:
                    logger.debug(f"Radio handling failed for fieldset: {e}")
                    continue
        except Exception as e:
            logger.debug(f"Radio handling failed: {e}")
    
    async def _has_visible_captcha(self, target: "Frame") -> bool:
        """
        Check if there's a visible reCAPTCHA on the page.
        
        Args:
            target: Playwright Frame (iframe)
            
        Returns:
            True if visible CAPTCHA detected
        """
        try:
            # Look for reCAPTCHA iframe or challenge
            recaptcha_iframe = target.locator('iframe[src*="recaptcha"]').first
            if await recaptcha_iframe.count() > 0:
                # Check if it's visible (not hidden)
                if await recaptcha_iframe.is_visible():
                    return True
            
            # Also check for challenge iframe
            challenge_iframe = target.locator('iframe[title*="recaptcha" i]').first
            if await challenge_iframe.count() > 0:
                if await challenge_iframe.is_visible():
                    return True
            
            return False
        except Exception:
            return False
    
    async def _solve_visible_captcha(self, page: "Page", target: "Frame") -> str:
        """
        Solve visible reCAPTCHA (if possible).
        
        Note: This requires 2Captcha or similar service integration.
        For now, returns "manual_required" as placeholder.
        
        Args:
            page: Playwright Page instance
            target: Playwright Frame (iframe) - used for submit button
            
        Returns:
            "solved" | "manual_required" | "error"
        """
        # Extract sitekey
        sitekey = await self._extract_sitekey(page, target)
        if not sitekey:
            logger.warning("Cannot solve CAPTCHA - no sitekey found")
            return "manual_required"
        
        # TODO: Integrate with 2Captcha or similar service
        # For now, return manual_required
        logger.info(f"CAPTCHA detected with sitekey: {sitekey[:20]}... - manual solve required")
        return "manual_required"
    
    async def _has_submit_button(self, target: "Frame") -> bool:
        """
        Check if submit button is present (vs continue button).
        
        Args:
            target: Playwright Frame (iframe)
            
        Returns:
            True if submit button found
        """
        submit = await self.selector.find(target, "submit_button", context="form_navigation")
        return submit is not None
    
    async def _click_submit(self, target: "Frame") -> str:
        """
        Click the submit button.
        
        Args:
            target: Playwright Frame (iframe)
            
        Returns:
            "success" | "error"
        """
        submit_btn = await self.selector.find(target, "submit_button", context="form_submit")
        if submit_btn:
            try:
                await submit_btn.click()
                logger.info("Clicked submit button")
                return "success"
            except Exception as e:
                logger.error(f"Submit click failed: {e}")
                return "error"
        return "error"
    
    async def _click_continue(self, target: "Frame") -> str:
        """
        Click the continue/next button.
        
        Args:
            target: Playwright Frame (iframe)
            
        Returns:
            "success" | "error"
        """
        continue_btn = await self.selector.find(target, "continue_button", context="form_navigation")
        if continue_btn:
            try:
                await continue_btn.click()
                logger.debug("Clicked continue button")
                return "success"
            except Exception as e:
                logger.error(f"Continue click failed: {e}")
                return "error"
        return "error"
    
    async def _is_success(self, page: "Page", target: "Frame") -> bool:
        """
        Check if application was submitted successfully.
        
        Args:
            page: Playwright Page instance
            target: Playwright Frame (iframe)
            
        Returns:
            True if success indicator found
        """
        # Check on main page for success
        success = await self.selector.find(page, "success_indicator", context="success_check")
        if success:
            return True
        
        # Also check for success text patterns (fallback)
        try:
            page_text = await page.evaluate("() => document.body.innerText.toLowerCase()")
            success_patterns = [
                "application submitted",
                "application sent",
                "successfully applied",
                "your application has been",
            ]
            for pattern in success_patterns:
                if pattern in page_text:
                    return True
        except Exception:
            pass
        
        return False
