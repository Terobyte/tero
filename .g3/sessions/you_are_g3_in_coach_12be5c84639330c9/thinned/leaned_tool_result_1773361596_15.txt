Tool result: """
Snapshot data models for capturing step state.

These models capture the full answered state of a form step for batch validation.
"""

from dataclasses import dataclass, field
from typing import List, Optional
import hashlib
import time


@dataclass
class QuestionSnapshot:
    """Snapshot of a single question's state."""
    question_id: str          # MD5 hash of question_text (first 12 chars)
    question_text: str
    intent: str               # from QuestionIntentClassifier (e.g., "requires_sponsorship")
    field_type: str           # radio, select, text, number, checkbox, textarea
    options: List[str]        # available options (empty for text/textarea)
    pre_fill_value: str       # value BEFORE our draft fill (for comparison)
    draft_answer: str         # what we filled
    is_critical: bool         # True if this is a P0 screening question
    confidence: str           # "high" (AnswerBank), "medium" (AI cache), "low" (AI live)

    @classmethod
    def create(
        cls,
        question_text: str,
        intent: str,
        field_type: str,
        options: List[str],
        pre_fill_value: str,
        draft_answer: str,
        is_critical: bool,
        confidence: str
    ) -> "QuestionSnapshot":
        """Factory method that generates question_id from question_text."""
        question_id = generate_question_id(question_text)
        return cls(
            question_id=question_id,
            question_text=question_text,
            intent=intent,
            field_type=field_type,
            options=options,
            pre_fill_value=pre_fill_value,
            draft_answer=draft_answer,
            is_critical=is_critical,
            confidence=confidence
        )

    def to_dict(self) -> dict:
        """Convert to dictionary for JSON serialization."""
        return {
            "question_id": self.question_id,
            "question_text": self.question_text,
            "intent": self.intent,
            "field_type": self.field_type,
            "options": self.options,
            "pre_fill_value": self.pre_fill_value,
            "draft_answer": self.draft_answer,
            "is_critical": self.is_critical,
            "confidence": self.confidence
        }


@dataclass
class StepSnapshot:
    """Snapshot of an entire form step's state."""
    session_id: str
    url: str
    company: str
    job_title: str
    step_index: int
    questions: List[QuestionSnapshot]
    timestamp: float = field(default_factory=time.time)
    attempt_number: int = 1   # for retry idempotency
    pass_type: str = "draft"  # "draft" | "correction" | "recheck"

    def to_dict(self) -> dict:
        """Convert to dictionary for JSON serialization."""
        return {
            "session_id": self.session_id,
            "url": self.url,
            "company": self.company,
            "job_title": self.job_title,
            "step_index": self.step_index,
            "questions": [q.to_dict() for q in self.questions],
            "timestamp": self.timestamp,
            "attempt_number": self.attempt_number,
            "pass_type": self.pass_type
        }

    def get_critical_questions(self) -> List[QuestionSnapshot]:
        """Return only questions marked as critical."""
        return [q for q in self.questions if q.is_critical]

    def get_question_by_id(self, question_id: str) -> Optional[QuestionSnapshot]:
        """Find a question by its ID."""
        for q in self.questions:
            if q.question_id == question_id:
                return q
        return None

    def has_critical_questions(self) -> bool:
        """Check if this step has any critical questions."""
        return any(q.is_critical for q in self.questions)

    def all_from_answer_bank(self) -> bool:
        """Check if all answers came from AnswerBank (high confidence)."""
        return all(q.confidence == "high" for q in self.questions)


def generate_question_id(question_text: str) -> str:
    """
    Generate a stable question ID from question text.
    
    Uses MD5 hash of normalized question text, truncated to 12 characters.
    """
    normalized = question_text.lower().strip()
    hash_hex = hashlib.md5(normalized.encode('utf-8')).hexdigest()
    return hash_hex[:12]
