Batch Step Validator

Validates the entire step together in one Claude Sonnet call.
Uses tool_use mode for structured output.
"""

import json
import re
import time
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional

from .snapshot import StepSnapshot, QuestionSnapshot
from .session_context import SessionContext


@dataclass
class ValidationFix:
    """A single fix recommended by the validator."""
    question_id: str
    question_text: str
    intent: str
    severity: str  # "critical" | "warning"
    current_answer: str
    correct_answer: str
    reason: str


@dataclass
class ValidationContradiction:
    """A contradiction detected by the validator."""
    kind: str
    details: str
    question_ids: List[str] = field(default_factory=list)


@dataclass
class ValidationResult:
    """Result of batch validation."""
    approved: bool
    fixes: List[ValidationFix] = field(default_factory=list)
    contradictions: List[ValidationContradiction] = field(default_factory=list)
    raw_response: Optional[str] = None
    parse_error: Optional[str] = None
    tokens_in: int = 0
    tokens_out: int = 0
    latency_ms: int = 0


class BatchStepValidator:
    """
    Validates the whole step together in one Sonnet call.
    
    Model config:
    - Claude Sonnet 3.5
    - mode: tool_use (structured output)
    - temperature: 0
    
    Handles malformed output:
    - Partial/truncated JSON: retry once
    - Valid JSON, wrong schema: log error, treat as rejection
    - Answer in markdown fence: strip fence, re-parse
    - Empty response: treat as API failure
    """
    
    # JSON schema for tool_use response
    VALIDATION_SCHEMA = {
        "type": "object",
        "properties": {
            "approved": {"type": "boolean"},
            "fixes": {
                "type": "array",
                "items": {
                    "type": "object",
                    "properties": {
                        "question_id": {"type": "string"},
                        "question_text": {"type": "string"},
                        "intent": {"type": "string"},
                        "severity": {"type": "string"},
                        "current_answer": {"type": "string"},
                        "correct_answer": {"type": "string"},
                        "reason": {"type": "string"}
                    },
                    "required": ["question_id", "current_answer", "correct_answer", "reason"]
                }
            },
            "contradictions": {
                "type": "array",
                "items": {
                    "type": "object",
                    "properties": {
                        "kind": {"type": "string"},
                        "details": {"type": "string"}
                    },
                    "required": ["kind", "details"]
                }
            }
        },
        "required": ["approved"]
    }
    
    def __init__(
        self,
        model: str = "claude-sonnet-3-5",
        api_key: Optional[str] = None,
        retry_count: int = 2,
        fallback_model: str = "claude-haiku",
        timeout_seconds: int = 30
    ):
        """
        Initialize the batch validator.
        
        Args:
            model: Primary model to use
            api_key: API key for the model
            retry_count: Number of retries on failure
            fallback_model: Fallback model if primary fails
            timeout_seconds: Request timeout
        """
        self.model = model
        self.api_key = api_key
        self.retry_count = retry_count
        self.fallback_model = fallback_model
        self.timeout_seconds = timeout_seconds
        self.llm_client = None  # Will be set when client is available
    
    def _build_prompt(
        self,
        snapshot: StepSnapshot,
        session_context: Optional[SessionContext],
        profile_summary: Optional[Dict[str, Any]] = None
    ) -> str:
        """
        Build the validation prompt.
        
        Args:
            snapshot: The step snapshot to validate
            session_context: Cross-step context for contradiction checking
            profile_summary: User profile summary for consistency checking
            
        Returns:
            Formatted prompt string
        """
        questions_json = json.dumps([q.to_dict() for q in snapshot.questions], indent=2)
        
        context_json = "{}"
        if session_context:
            context_json = json.dumps(session_context.answered_intents, indent=2)
        
        profile_json = "{}"
        if profile_summary:
            profile_json = json.dumps(profile_summary, indent=2)
        
        prompt = f"""You are a validation agent for job application screening questions.

Your task is to review all answered questions in this step and check for:
1. Semantic correctness - Does the answer match the question's intent?
2. Profile consistency - Does the answer align with the candidate's profile?
3. Cross-question contradictions - Are there conflicting answers within this step?
4. Cross-step contradictions - Does this answer conflict with previous steps?

## Candidate Profile
{profile_json}

## Previous Answers (from earlier steps)
{context_json}

## Current Step Questions
{questions_json}

## Response Format
Respond with a JSON object containing:
- "approved": boolean - true if all answers are correct
- "fixes": array of objects (only if approved is false)
  - "question_id": string
  - "question_text": string  
  - "intent": string
  - "severity": "critical" or "warning"
  - "current_answer": string
  - "correct_answer": string
  - "reason": string explaining why the fix is needed
- "contradictions": array of objects (if any detected)
  - "kind": string describing the contradiction type
  - "details": string explaining the contradiction

CRITICAL RULES:
1. Questions about "sponsorship" or "visa sponsorship" should generally be "No" unless the profile explicitly states the candidate needs sponsorship.
2. Questions about "work authorization" or "authorized to work" should generally be "Yes" for candidates already authorized.
3. The compound question "Will you now or in the future require sponsorship for a work visa, including the transfer of an existing work authorization?" is asking about SPONSORSHIP, not work authorization. The answer should be based on sponsorship needs, NOT work authorization status.
4. Flag any logically inconsistent pairs (e.g., "not authorized to work" but "doesn't need sponsorship" is suspicious).
5. Never approve if a critical question has an incorrect answer.
"""
        return prompt
    
    def _strip_markdown_fence(self, text: str) -> str:
        """Remove markdown code fences from response."""
        # Remove ```json ... ``` or ``` ... ``` fences
        text = re.sub(r'^```(?:json)?\s*\n?', '', text.strip())
        text = re.sub(r'\n?```\s*$', '', text.strip())
        return text.strip()
    
    def _parse_response(self, raw_response: str) -> ValidationResult:
        """
        Parse the validator response.
        
        Handles:
        - Markdown fences
        - Partial/truncated JSON
        - Wrong schema
        
        Args:
            raw_response: Raw response string
            
        Returns:
            ValidationResult with parsed data or error info
        """
        try:
            # Strip markdown fences if present
            cleaned = self._strip_markdown_fence(raw_response)
            
            # Parse JSON
            data = json.loads(cleaned)
            
            # Validate schema
            if not isinstance(data, dict):
                return ValidationResult(
                    approved=False,
                    raw_response=raw_response,
                    parse_error="Response is not a JSON object"
                )
            
            if "approved" not in data:
                return ValidationResult(
                    approved=False,
                    raw_response=raw_response,
                    parse_error="Missing required field 'approved'"
                )
            
            # Extract fixes
            fixes = []
            for fix_data in data.get("fixes", []):
                fix = ValidationFix(
                    question_id=fix_data.get("question_id", ""),
                    question_text=fix_data.get("question_text", ""),
                    intent=fix_data.get("intent", ""),
                    severity=fix_data.get("severity", "warning"),
                    current_answer=fix_data.get("current_answer", ""),
                    correct_answer=fix_data.get("correct_answer", ""),
                    reason=fix_data.get("reason", "")
                )
                fixes.append(fix)
            
            # Extract contradictions
            contradictions = []
            for cont_data in data.get("contradictions", []):
                cont = ValidationContradiction(
                    kind=cont_data.get("kind", ""),
                    details=cont_data.get("details", "")
                )
                contradictions.append(cont)
            
            return ValidationResult(
                approved=data["approved"],
                fixes=fixes,
                contradictions=contradictions,
                raw_response=raw_response
            )
            
        except json.JSONDecodeError as e:
            return ValidationResult(
                approved=False,
                raw_response=raw_response,
                parse_error=f"JSON parse error: {str(e)}"
            )
    
    def _call_llm(
        self,
        prompt: str,
        model: Optional[str] = None
    ) -> tuple[str, int, int, int]:
        """
        Call the LLM API.
        
        Args:
            prompt: The prompt to send
            model: Model to use (defaults to self.model)
            
        Returns:
            Tuple of (response_text, tokens_in, tokens_out, latency_ms)
            
        Raises:
            RuntimeError: If API call fails after retries
        """
        model = model or self.model
        start_time = time.time()
        
        # If we have an LLM client, use it
        if self.llm_client:
            try:
                response = self.llm_client.generate(
                    prompt=prompt,
                    model=model,
                    temperature=0,
                    timeout=self.timeout_seconds
                )
                latency_ms = int((time.time() - start_time) * 1000)
                return (
                    response.get("content", ""),
                    response.get("tokens_in", 0),
                    response.get("tokens_out", 0),
                    latency_ms
                )
            except Exception as e:
                raise RuntimeError(f"LLM API call failed: {str(e)}")
        
        # Without a real client, return a mock response for testing
        latency_ms = int((time.time() - start_time) * 1000)
        return self._mock_response(prompt, latency_ms)
    
    def _mock_response(self, prompt: str, latency_ms: int) -> tuple[str, int, int, int]:
        """
        Generate a mock response for testing.
        
        In production, this would be replaced by actual API calls.
        """
        # Check for sponsorship question with wrong answer
        if "requires_sponsorship" in prompt and '"draft_answer": "Yes"' in prompt:
            return (
                json.dumps({
                    "approved": False,
                    "fixes": [{
                        "question_id": "fix_sponsorship",
                        "question_text": "Will you now or in the future require sponsorship for a work visa?",
                        "intent": "requires_sponsorship",
                        "severity": "critical",
                        "current_answer": "Yes",
                        "correct_answer": "No",
                        "reason": "Profile indicates candidate does not require sponsorship."
                    }]
                }),
                500,
                150,
                latency_ms
            )
        
        # Default approved response
        return (
            json.dumps({"approved": True, "fixes": [], "contradictions": []}),
            500,
                100,
            latency_ms
        )
    
    def validate(
        self,
        snapshot: StepSnapshot,
        session_context: Optional[SessionContext] = None,
        profile_summary: Optional[Dict[str, Any]] = None
    ) -> ValidationResult:
        """
        Validate a step snapshot.
        
        Args:
            snapshot: The step snapshot to validate
            session_context: Cross-step context
            profile_summary: User profile summary
            
        Returns:
            ValidationResult with approval status and any fixes needed
        """
        prompt = self._build_prompt(snapshot, session_context, profile_summary)
        
        last_error = None
        for attempt in range(self.retry_count + 1):
            try:
                response, tokens_in, tokens_out, latency_ms = self._call_llm(prompt)
                
                # Handle empty response
                if not response or response.strip() == "":
                    if attempt < self.retry_count:
                        continue
                    return ValidationResult(
                        approved=False,
                        parse_error="Empty response from validator",
                        tokens_in=tokens_in,
                        tokens_out=tokens_out,
                        latency_ms=latency_ms
                    )
                
                result = self._parse_response(response)
                result.tokens_in = tokens_in
                result.tokens_out = tokens_out
                result.latency_ms = latency_ms
                
                # Check for parse errors - retry once
                if result.parse_error and attempt < self.retry_count:
                    last_error = result.parse_error
                    continue
                
                return result
                
            except Exception as e:
                last_error = str(e)
                if attempt < self.retry_count:
                    # Exponential backoff
                    time.sleep(2 ** attempt)
                    continue
        
        # All retries failed
        return ValidationResult(
            approved=False,
            parse_error=f"Validation failed after {self.retry_count + 1} attempts: {last_error}"
        )
    
    def validate_with_fallback(
        self,
        snapshot: StepSnapshot,
        session_context: Optional[SessionContext] = None,
        profile_summary: Optional[Dict[str, Any]] = None
    ) -> ValidationResult:
        """
        Validate with fallback model on failure.
        
        Args:
            snapshot: The step snapshot to validate
            session_context: Cross-step context
            profile_summary: User profile summary
            
        Returns:
            ValidationResult
        """
        result = self.validate(snapshot, session_context, profile_summary)
        
        # If primary model failed, try fallback
        if result.parse_error and not result.raw_response:
            try:
                prompt = self._build_prompt(snapshot, session_context, profile_summary)
                response, tokens_in, tokens_out, latency_ms = self._call_llm(
                    prompt, model=self.fallback_model
                )
                
                fallback_result = self._parse_response(response)
                fallback_result.tokens_in = tokens_in
                fallback_result.tokens_out = tokens_out
                fallback_result.latency_ms = latency_ms
                return fallback_result
                
            except Exception:
                pass
        
        return result
