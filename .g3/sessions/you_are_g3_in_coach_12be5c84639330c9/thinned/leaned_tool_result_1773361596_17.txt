SessionContext - Tracks answered intents across ALL steps in one application
to detect cross-step contradictions.
"""

from dataclasses import dataclass, field
from typing import Dict, Optional


@dataclass
class SessionContext:
    """
    Tracks answered intents across ALL steps in one application to detect
    cross-step contradictions.
    """
    session_id: str
    company: str
    job_title: str
    answered_intents: Dict[str, str] = field(default_factory=dict)

    def record(self, intent: str, answer: str) -> None:
        """
        Record an intent-answer pair.
        
        Args:
            intent: The canonical intent (e.g., "requires_sponsorship")
            answer: The answer value (e.g., "No")
        """
        self.answered_intents[intent] = answer

    def check_cross_step_contradiction(
        self, intent: str, new_answer: str
    ) -> Optional[str]:
        """
        Check if the new answer contradicts a previously recorded answer
        for the same intent.
        
        Args:
            intent: The canonical intent being checked
            new_answer: The proposed new answer
            
        Returns:
            Contradiction description if there's a conflict, None if clean.
        """
        if intent not in self.answered_intents:
            return None
        
        existing_answer = self.answered_intents[intent]
        if existing_answer != new_answer:
            return (
                f"Cross-step contradiction: '{intent}' was previously answered as "
                f"'{existing_answer}' but new answer is '{new_answer}'"
            )
        return None

    def get_answer(self, intent: str) -> Optional[str]:
        """
        Get the previously recorded answer for an intent.
        
        Args:
            intent: The canonical intent
            
        Returns:
            The recorded answer or None if not found
        """
        return self.answered_intents.get(intent)

    def clear(self) -> None:
        """Clear all recorded answers (for new session)."""
        self.answered_intents.clear()

    def to_dict(self) -> Dict:
        """Export to dictionary for serialization."""
        return {
            "session_id": self.session_id,
            "company": self.company,
            "job_title": self.job_title,
            "answered_intents": dict(self.answered_intents),
        }

    @classmethod
    def from_dict(cls, data: Dict) -> "SessionContext":
        """Create from dictionary."""
        return cls(
            session_id=data["session_id"],
            company=data["company"],
            job_title=data["job_title"],
            answered_intents=data.get("answered_intents", {}),
        )
