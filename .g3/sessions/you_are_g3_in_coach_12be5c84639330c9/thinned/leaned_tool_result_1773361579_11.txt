Tool result: """
Critical Question Policy

Hard rules for P0 screening topics. Config-driven via config/critical_intents.json

Rules:
- no optimistic default "Yes"
- no silent fallback for unknown critical questions
- no automatic continue if critical intent confidence is low
- no contradictory pair allowed on same step
"""

import json
from dataclasses import dataclass, field
from enum import Enum
from pathlib import Path
from typing import Optional



class Severity(str, Enum):
    """Severity level for policy violations."""
    CRITICAL = "critical"
    WARNING = "warning"
    INFO = "info"


@dataclass
class PolicyResult:
    """Result of policy evaluation."""
    allowed: bool
    hard_stop: bool
    severity: Severity
    reason: Optional[str] = None
    contradiction_details: Optional[str] = None


@dataclass
class ContradictionPair:
    """Defines a pair of intents that can contradict each other."""
    intent_a: str
    intent_b: str
    valid_combinations: list[tuple[str, str]]  # List of (answer_a, answer_b) that are valid
    suspicious_combinations: list[tuple[str, str]]  # Flagged but not blocked


@dataclass
class CriticalQuestionPolicy:
    """
    Policy engine for critical screening questions.
    
    Loads critical intent definitions from config and enforces:
    - No auto-answer with unsafe defaults
    - No silent fallback for unknown critical questions
    - Hard stop for low-confidence critical answers
    - Contradiction detection for related questions
    """
    
    config_path: Optional[Path] = None
    critical_intents: set[str] = field(default_factory=set)
    unsafe_defaults: dict[str, str] = field(default_factory=dict)
    contradiction_pairs: list[ContradictionPair] = field(default_factory=list)
    
    def __post_init__(self):
        """Load config after initialization."""
        if self.config_path is None:
            self.config_path = Path(__file__).parent.parent.parent.parent / "config" / "critical_intents.json"
        
        self._load_config()
        self._build_contradiction_matrix()
    
    def _load_config(self) -> None:
        """Load critical intent configuration from JSON."""
        if not self.config_path.exists():
            # Use defaults if config doesn't exist
            self.critical_intents = {"requires_sponsorship"}
            self.min_confidences = {"requires_sponsorship": 0.9}
            self.unsafe_defaults = {"requires_sponsorship": "Yes"}
            return
        
        with open(self.config_path, "r") as f:
            config = json.load(f)
        
        # Parse critical intents list
        critical_intent_list = config.get("critical_intents", [])
        self.critical_intents = {item["intent"] for item in critical_intent_list}
        
        # Load min confidences per intent
        self.min_confidences = {
            item["intent"]: item.get("min_confidence", 0.7)
            for item in critical_intent_list
        }
        
        # Build unsafe defaults - any intent with allow_optimistic_default=False
        # should never have an optimistic "Yes" answer
        self.unsafe_defaults = {}
        for item in critical_intent_list:
            if not item.get("allow_optimistic_default", True):
                # The unsafe default for most critical questions is "Yes"
                # (assuming you need something when you don't)
                self.unsafe_defaults[item["intent"]] = "Yes"
        
        # Store contradiction rules from config
        self._load_contradiction_rules(config.get("contradiction_rules", []))
    
    def _load_contradiction_rules(self, rules: list) -> None:
        """Load contradiction rules from config."""
        self.contradiction_rules_config = rules
    
    def _build_contradiction_matrix(self) -> None:
        """Build contradiction pairs for validation."""
        self.contradiction_pairs = [
            ContradictionPair(
                intent_a="authorized_to_work",
                intent_b="requires_sponsorship",
                # Valid combinations
                valid_combinations=[
                    ("Yes", "No"),   # Authorized, doesn't need sponsorship
                    ("Yes", "Yes"),  # Authorized but will need future sponsorship
                    ("No", "Yes"),   # Not authorized, needs visa
                ],
                # Suspicious but not blocked
                suspicious_combinations=[
                    ("No", "No"),    # Not authorized but doesn't need sponsorship? Suspicious.
                ],
            ),
            ContradictionPair(
                intent_a="has_drivers_license",
                intent_b="willing_to_relocate",
                valid_combinations=[
                    ("Yes", "Yes"),
                    ("Yes", "No"),
                    ("No", "Yes"),
                    ("No", "No"),
                ],
                suspicious_combinations=[],
            ),
        ]
    
    def is_critical(self, intent: str) -> bool:
        """Check if an intent is classified as critical."""
        return intent in self.critical_intents
    
    def is_unsafe_default(self, intent: str, answer: str) -> bool:
        """Check if an answer is an unsafe default for a critical intent."""
        return intent in self.unsafe_defaults and answer == self.unsafe_defaults[intent]
    
    def evaluate(
        self,
        intent: str,
        answer: str,
        confidence: float,
    ) -> PolicyResult:
        """
        Evaluate a single answer against policy rules.
        
        Args:
            intent: The classified intent
            answer: The proposed answer
            confidence: Confidence level (0.0-1.0)
            
        Returns:
            PolicyResult with allowed, hard_stop, severity, and reason
        """
        # Non-critical intents are always allowed
        if not self.is_critical(intent):
            return PolicyResult(
                allowed=True,
                hard_stop=False,
                severity=Severity.INFO,
            )
        
        # Check for unsafe defaults on critical intents
        if self.is_unsafe_default(intent, answer):
            return PolicyResult(
                allowed=False,
                hard_stop=True,
                severity=Severity.CRITICAL,
                reason=f"Answer '{answer}' is an unsafe default for critical intent '{intent}'. "
                       f"This requires explicit profile confirmation.",
            )
        
        # Low confidence on critical intent = hard stop
        if confidence < 0.7:
            return PolicyResult(
                allowed=False,
                hard_stop=True,
                severity=Severity.CRITICAL,
                reason=f"Confidence {confidence:.2f} below threshold (0.7) for critical intent '{intent}'. "
                       f"Cannot auto-answer without higher confidence.",
            )
        
        # Unknown intent that might be critical
        if intent == "unknown":
            # If we can't classify, but it contains critical keywords, block
            return PolicyResult(
                allowed=False,
                hard_stop=True,
                severity=Severity.WARNING,
                reason="Cannot classify intent. Critical questions must have clear intent classification.",
            )
        
        return PolicyResult(
            allowed=True,
            hard_stop=False,
            severity=Severity.INFO,
        )
    
    def check_pair_contradiction(
        self,
        intent_a: str,
        answer_a: str,
        intent_b: str,
        answer_b: str,
    ) -> PolicyResult:
        """
        Check if a pair of answers form a contradiction.
        
        Args:
            intent_a: First intent
            answer_a: First answer
            intent_b: Second intent
            answer_b: Second answer
            
        Returns:
            PolicyResult with contradiction details if found
        """
        for pair in self.contradiction_pairs:
            # Check both orderings
            if (intent_a == pair.intent_a and intent_b == pair.intent_b):
                combo = (answer_a, answer_b)
            elif (intent_a == pair.intent_b and intent_b == pair.intent_a):
                combo = (answer_b, answer_a)
            else:
                continue
            
            # Check for valid combinations
            if combo in pair.valid_combinations:
                return PolicyResult(
                    allowed=True,
                    hard_stop=False,
                    severity=Severity.INFO,
                )
            
            # Check for suspicious combinations
            if combo in pair.suspicious_combinations:
                return PolicyResult(
                    allowed=True,  # Allowed but flagged
                    hard_stop=False,
                    severity=Severity.WARNING,
                    contradiction_details=f"Suspicious combination: {intent_a}={answer_a}, {intent_b}={answer_b}. "
                                        f"This combination is unusual and should be reviewed.",
                )
            
            # Invalid combination - hard stop
            return PolicyResult(
                allowed=False,
                hard_stop=True,
                severity=Severity.CRITICAL,
                contradiction_details=f"Contradiction detected: {intent_a}={answer_a}, {intent_b}={answer_b}. "
                                     f"This combination is logically inconsistent.",
            )
        
        # No contradiction pair found
        return PolicyResult(
            allowed=True,
            hard_stop=False,
            severity=Severity.INFO,
        )
    
    def validate_step_intents(
        self,
        intent_answers: list[tuple[str, str, float]],
    ) -> list[PolicyResult]:
        """
        Validate all intents in a step, including pair-wise contradictions.
        
        Args:
            intent_answers: List of (intent, answer, confidence) tuples
            
        Returns:
            List of PolicyResult for each answer, plus any pair contradictions
        """
        results = []
        
        # First pass: evaluate each answer individually
        for intent, answer, confidence in intent_answers:
            result = self.evaluate(intent, answer, confidence)
            results.append(result)
        
        # Second pass: check pairs for contradictions
        for i, (intent_a, answer_a, _) in enumerate(intent_answers):
            for j, (intent_b, answer_b, _) in enumerate(intent_answers):
                if i >= j:
                    continue
                
                pair_result = self.check_pair_contradiction(
                    intent_a, answer_a, intent_b, answer_b
                )
                
                if not pair_result.allowed or pair_result.severity == Severity.WARNING:
                    # Add pair result to the first answer's result
                    results[i] = PolicyResult(
                        allowed=results[i].allowed and pair_result.allowed,
                        hard_stop=results[i].hard_stop or pair_result.hard_stop,
                        severity=max(results[i].severity, pair_result.severity, key=lambda s: s.value),
                        reason=results[i].reason,
                        contradiction_details=pair_result.contradiction_details,
                    )
        
        return results
    
    def should_skip_validation(
        self,
        intents: list[str],
        confidences: list[float],
        field_types: list[str],
        question_count: int,
    ) -> bool:
        """
        Determine if batch validation can be skipped for this step.
        
        A step is skip-able when ALL of the following are true:
        - zero questions classified as critical intent
        - all answers came from AnswerBank with confidence=high (>= 0.9)
        - step has 3 or fewer questions
        - step contains only text/number fields (no radio, no select)
        
        Args:
            intents: List of classified intents
            confidences: List of confidence values
            field_types: List of field types
            question_count: Number of questions in step
            
        Returns:
            True if validation can be skipped
        """
        # Check for critical intents
        if any(self.is_critical(intent) for intent in intents):
            return False
        
        # Check confidence levels
        if any(conf < 0.9 for conf in confidences):
            return False
        
        # Check question count
        if question_count > 3:
            return False
        
        # Check field types
        safe_types = {"text", "number", "textarea"}
        if any(ft not in safe_types for ft in field_types):
            return False
        
        return True
