Tool result: """Health monitoring and degradation detection for selector resolution.

Phase 3 of the Indeed MTD implementation - tracks selector resolution stats
and alerts when hit rate drops below threshold.
"""

from __future__ import annotations

import json
import logging
import os
import time
from dataclasses import dataclass, field, asdict
from pathlib import Path
from typing import Dict, List, Optional

logger = logging.getLogger(__name__)


@dataclass
class SelectorStats:
    """Statistics for a single intent's selector resolution."""
    intent: str
    total_attempts: int = 0
    tier1_hits: int = 0  # cached
    tier2_hits: int = 0  # semantic
    tier3_hits: int = 0  # structural
    tier4_hits: int = 0  # LLM
    misses: int = 0


@dataclass
class DegradationAlert:
    """Alert when an intent's hit rate falls below threshold."""
    intent: str
    hit_rate: float
    total_attempts: int
    misses: int


class SelectorHealthMonitor:
    """Monitors selector resolution health and detects degradation.
    
    Persists stats to disk for cross-session tracking.
    Debounces disk writes to prevent thrashing.
    Enforces LLM budget caps per session.
    """
    
    DEGRADATION_THRESHOLD = 0.3  # Alert if hit rate < 30%
    MAX_LLM_CALLS_PER_SESSION = 20  # Max Tier-4 calls per bot run
    STATS_FILE = Path(__file__).parent.parent.parent / "data" / "selector_stats.json"
    
    def __init__(self) -> None:
        self._stats: Dict[str, SelectorStats] = {}
        self._llm_calls_this_session: int = 0  # Instance-level - safe for parallel appliers
        self._last_save: float = 0.0  # Timestamp of last disk write (debounce)
        
        # Ensure data directory exists
        self.STATS_FILE.parent.mkdir(parents=True, exist_ok=True)
        
        # Restore from previous runs
        self.load_stats()
    
    def record(
        self, 
        intent: str, 
        tier: int, 
        selector: str = "", 
        miss: bool = False
    ) -> None:
        """Record a selector resolution attempt.
        
        Args:
            intent: e.g. "apply_button"
            tier: 1-4 on success (which tier resolved it), ignored if miss=True
            selector: the winning selector string (empty string on miss)
            miss: True if all tiers failed to find a visible element
                  
        Example success call: record("apply_button", 2, "button:has-text('Apply now')")
        Example miss call:    record("apply_button", 0, "", miss=True)
        """
        stats = self._stats.setdefault(intent, SelectorStats(intent=intent))
        stats.total_attempts += 1
        
        if miss:
            stats.misses += 1
        elif tier == 1:
            stats.tier1_hits += 1
        elif tier == 2:
            stats.tier2_hits += 1
        elif tier == 3:
            stats.tier3_hits += 1
        elif tier == 4:
            stats.tier4_hits += 1
        
        # Debounced disk write (max once per 30s)
        now = time.monotonic()
        if now - self._last_save >= 30.0:
            self.save_stats()
            self._last_save = now
    
    def check_degradation(self) -> List[DegradationAlert]:
        """Check for degraded selectors (hit rate < threshold).
        
        Returns:
            List of DegradationAlert for intents with low hit rates.
        """
        alerts: List[DegradationAlert] = []
        
        for intent, stats in self._stats.items():
            if stats.total_attempts < 3:
                # Need minimum attempts before alerting
                continue
            
            hits = (
                stats.tier1_hits + 
                stats.tier2_hits + 
                stats.tier3_hits + 
                stats.tier4_hits
            )
            hit_rate = hits / stats.total_attempts if stats.total_attempts > 0 else 0.0
            
            if hit_rate < self.DEGRADATION_THRESHOLD:
                alerts.append(DegradationAlert(
                    intent=intent,
                    hit_rate=hit_rate,
                    total_attempts=stats.total_attempts,
                    misses=stats.misses
                ))
        
        return alerts
    
    def get_report(self) -> Dict:
        """Get a comprehensive health report.
        
        Returns:
            Dict with per-intent stats and overall health metrics.
        """
        report = {
            "intents": {},
            "summary": {
                "total_intents": len(self._stats),
                "total_attempts": sum(s.total_attempts for s in self._stats.values()),
                "total_misses": sum(s.misses for s in self._stats.values()),
                "degraded_intents": [],
            }
        }
        
        for intent, stats in self._stats.items():
            hits = (
                stats.tier1_hits + 
                stats.tier2_hits + 
                stats.tier3_hits + 
                stats.tier4_hits
            )
            hit_rate = hits / stats.total_attempts if stats.total_attempts > 0 else 0.0
            
            report["intents"][intent] = {
                "total_attempts": stats.total_attempts,
                "tier1_hits": stats.tier1_hits,
                "tier2_hits": stats.tier2_hits,
                "tier3_hits": stats.tier3_hits,
                "tier4_hits": stats.tier4_hits,
                "misses": stats.misses,
                "hit_rate": hit_rate,
            }
            
            if hit_rate < self.DEGRADATION_THRESHOLD:
                report["summary"]["degraded_intents"].append(intent)
        
        return report
    
    def save_stats(self) -> None:
        """Persist stats to data/selector_stats.json."""
        try:
            data = {
                "stats": {
                    intent: asdict(stats) 
                    for intent, stats in self._stats.items()
                },
                "llm_calls_this_session": self._llm_calls_this_session,
                "saved_at": time.time(),
            }
            
            with open(self.STATS_FILE, "w") as f:
                json.dump(data, f, indent=2)
                
        except Exception as e:
            logger.warning(f"Failed to save selector stats: {e}")
    
    def load_stats(self) -> None:
        """Restore stats from data/selector_stats.json."""
        if not self.STATS_FILE.exists():
            return
        
        try:
            with open(self.STATS_FILE, "r") as f:
                data = json.load(f)
            
            for intent, stats_dict in data.get("stats", {}).items():
                self._stats[intent] = SelectorStats(
                    intent=stats_dict.get("intent", intent),
                    total_attempts=stats_dict.get("total_attempts", 0),
                    tier1_hits=stats_dict.get("tier1_hits", 0),
                    tier2_hits=stats_dict.get("tier2_hits", 0),
                    tier3_hits=stats_dict.get("tier3_hits", 0),
                    tier4_hits=stats_dict.get("tier4_hits", 0),
                    misses=stats_dict.get("misses", 0),
                )
            
            # Note: We don't restore llm_calls_this_session across restarts
            # because a new session means a fresh budget
            
            logger.debug(f"Loaded stats for {len(self._stats)} intents")
            
        except Exception as e:
            logger.warning(f"Failed to load selector stats: {e}")
    
    def llm_budget_ok(self) -> bool:
        """Check if LLM budget is available for Tier 4 calls.
        
        Returns:
            True if budget available, False if cap reached.
        """
        return self._llm_calls_this_session < self.MAX_LLM_CALLS_PER_SESSION
    
    def increment_llm_call(self) -> None:
        """Increment the LLM call counter (called by SelectorEngine on Tier 4 use)."""
        self._llm_calls_this_session += 1
        
        if self._llm_calls_this_session >= self.MAX_LLM_CALLS_PER_SESSION:
            logger.warning(
                f"LLM budget exhausted: {self._llm_calls_this_session}/{self.MAX_LLM_CALLS_PER_SESSION} calls"
            )
