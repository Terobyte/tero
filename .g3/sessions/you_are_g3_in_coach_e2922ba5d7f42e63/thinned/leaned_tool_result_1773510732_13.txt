"""
NaturalLanguageGenerator — генерация уникальных ответов на screening вопросы.

Ключевые отличия от AIAnswerer:
1. Job-specific: учитывает конкретную JD и компанию
2. PKB-aware: извлекает релевантные примеры из Knowledge Base
3. Variability: каждый ответ уникален (нет шаблонных фраз)
4. ATS-optimized: вставляет ключевые слова из JD
"""
from __future__ import annotations

import json
import random
from dataclasses import dataclass
from typing import Optional

from loguru import logger


@dataclass
class NLGResult:
    """Результат генерации."""
    answer: str
    keywords_used: list[str]    # JD-ключевые слова в ответе
    confidence: str             # high/medium/low
    word_count: int


_SYSTEM = """You are an expert at writing compelling job application answers.
Write a direct, authentic response as the candidate. No preambles, no "Based on".

Candidate profile context:
{profile_context}

Relevant experience (from knowledge base):
{kb_context}

Job: {job_title} at {company}
Key requirements: {requirements}

Rules:
1. Start directly with the answer (no "I would like to", "Based on", etc.)
2. Use 2-4 sentences for textarea, 1 sentence for text
3. Naturally include these JD keywords: {keywords}
4. Speak authentically in first person
5. Quantify when possible (years, percentages, project counts)

Question: {question}
Field type: {field_type}

Answer (text only, no JSON):"""


class NaturalLanguageGenerator:
    """
    Генератор уникальных ответов на вопросы форм.

    Usage:
        nlg = NaturalLanguageGenerator(llm_client, kb)
        result = await nlg.generate(
            question="Why do you want to join us?",
            job={"title": "SRE", "company": "Stripe"},
            field_type="textarea",
        )
    """

    def __init__(self, llm_client, kb=None):
        """
        Args:
            llm_client: LLM клиент (KimiClient, GLM5Client, etc.)
            kb: VectorKnowledgeBase (опционально, для RAG)
        """
        self._llm = llm_client
        self._kb = kb

    async def generate(
        self,
        question: str,
        job: dict,
        field_type: str = "textarea",
        profile: Optional[dict] = None,
        requirements: Optional[list[str]] = None,
        max_words: int = 80,
    ) -> NLGResult:
        """
        Сгенерировать уникальный ответ на вопрос формы.

        Args:
            question: Текст вопроса
            job: Словарь вакансии {title, company, description}
            field_type: textarea / text
            profile: Профиль кандидата (personal, experience, etc.)
            requirements: Требования из JD (для keyword injection)
            max_words: Максимум слов в ответе
        """
        # Получить контекст из KB
        kb_context = ""
        if self._kb:
            try:
                results = self._kb.search_pkb(question, k=3)
                if results:
                    kb_context = "\n".join(r["text"][:200] for r in results)
            except Exception as e:
                logger.debug(f"NLG KB lookup failed: {e}")

        # Профиль кандидата
        profile_context = self._format_profile(profile or {})

        # Ключевые слова из JD
        keywords = requirements[:5] if requirements else []
        keywords_str = ", ".join(keywords) if keywords else "N/A"

        prompt = _SYSTEM.format(
            profile_context=profile_context[:1000],
            kb_context=kb_context[:800],
            job_title=job.get("title", "the role"),
            company=job.get("company", "your company"),
            requirements=", ".join((requirements or [])[:8]),
            keywords=keywords_str,
            question=question,
            field_type=field_type,
        )

        try:
            raw = await self._llm.generate(prompt)
            answer = raw.strip()

            # Обрезать до max_words
            words = answer.split()
            if len(words) > max_words:
                answer = " ".join(words[:max_words]).rstrip(",;") + "."

            # Посчитать сколько ключевых слов попало в ответ
            answer_lower = answer.lower()
            keywords_used = [kw for kw in keywords if kw.lower() in answer_lower]

            return NLGResult(
                answer=answer,
                keywords_used=keywords_used,
                confidence="high" if len(keywords_used) >= 2 else "medium",
                word_count=len(answer.split()),
            )
        except Exception as e:
            logger.error(f"NLG generation failed: {e}")
            return NLGResult(
                answer="Please see my resume for details.",
                keywords_used=[],
                confidence="low",
                word_count=6,
            )

    def _format_profile(self, profile: dict) -> str:
        """Краткий текст профиля для промпта."""
        lines = []
        p = profile.get("personal", {})
        if p.get("first_name"):
            lines.append(f"Name: {p['first_name']} {p.get('last_name', '')}")

        exp = profile.get("experience", {})
        if exp.get("total_years"):
            lines.append(f"Total experience: {exp['total_years']} years")

        skills_keys = [k for k in exp if k.startswith("years_") and int(exp[k] or 0) > 0]
        if skills_keys:
            skill_list = [f"{k.replace('years_', '')} ({exp[k]}y)" for k in skills_keys[:6]]
            lines.append(f"Skills: {', '.join(skill_list)}")

        return "\n".join(lines) or "Experienced professional"
