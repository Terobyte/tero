class ATSDetector:
    """
    Detects known ATS platforms from URL patterns.
    """
    
    ATS_PATTERNS = {
        'greenhouse': [
            r'greenhouse\.io',
            r'greenhouse\.com',
            r'/jobs/[^/]+/apply',
            r'boards\.greenhouse\.io',
        ],
        'lever': [
            r'lever\.co',
            r'jobs\.lever\.co',
        ],
        'workday': [
            r'workday\.',
            r'myworkdayjobs\.com',
            r'wd5\.myworkdayjobs\.com',
        ],
        'ashby': [
            r'ashby\.hq',
            r'jobs\.ashbyhq\.com',
        ],
        'bamboohr': [
            r'bamboohr\.com',
            r'bamboohr\.',
        ],
        'smartrecruiters': [
            r'smartrecruiters\.com',
        ],
        'indeed': [
            r'indeed\.com',
        ],
        'linkedin': [
            r'linkedin\.com/jobs',
            r'linkedin\.com/company',
        ],
        'icims': [
            r'icims\.com',
        ],
        'taleo': [
            r'taleo\.net',
            r'taleo\.com',
        ],
        'brassring': [
            r'brassring\.com',
        ],
        'jobvite': [
            r'jobvite\.com',
        ],
        'myapplication144': [
            r'myapplication144\.',
        ],
        'ultipro': [
            r'ultipro\.com',
        ],
        'cornerstone': [
            r'cornerstone',
        ],
    }
    
    @classmethod
    def detect(cls, url: str) -> str:
        """
        Detect ATS platform from URL.
        
        Returns:
            ATS name or 'unknown'
        """
        url_lower = url.lower()
        
        for ats_name, patterns in cls.ATS_PATTERNS.items():
            for pattern in patterns:
                if re.search(pattern, url_lower):
                    return ats_name
        
        return 'unknown'
    
    @classmethod
    def is_likely_custom_careers_page(cls, url: str) -> bool:
        """
        Heuristic: URL likely leads to a custom career site.
        """
        customs_patterns = [
            r"careers?\.",
            r"/careers",
            r"/jobs?(/|$)",
            r"/join-us",
            r"/opportunities",
            r"/apply",
            r"/openings",
            r"/vacancies",
            r"/talent",
            r"/recruitment",
        ]
        return any(re.search(p, url.lower()) for p in customs_patterns)

class ReconAgent:
    """
    Site reconnaissance agent.
    
    Analyzes unknown career sites to understand their structure.
    Uses:
    1. DOM extraction (JavaScript) - fast, cheap
    2. Vision analysis (Kimi 2.5) - understands layout, UI
    3. ATS detection - redirect if known ATS
    """
    
    def __init__(
        self,
        kimi_vision_client: Any = None,
        screenshot_dir: Path | None = None,
        use_vision: bool = True,
    ):
        """
        Initialize ReconAgent.
        
        Args:
            kimi_vision_client: KimiVisionClient instance
            screenshot_dir: Directory to save screenshots
            use_vision: Enable vision analysis (default True)
        """
        self.kimi_vision = kimi_vision_client
        self.screenshot_dir = screenshot_dir or Path(tempfile.mkdtemp(prefix="recon_"))
        self.use_vision = use_vision
        
        if self.screenshot_dir:
            self.screenshot_dir.mkdir(parents=True, exist_ok=True)
    
    async def analyze(self, page) -> SiteSchema:
        """
        Analyze a page and return a SiteSchema.
        
        Args:
            page: Playwright page object
            
        Returns:
            SiteSchema with analysis results
        """
        url = page.url
        domain = urlparse(url).netloc
        
        logger.info(f"🔍 Recon: analyzing {url}")
        
        # Step 1: DOM extraction
        dom_snapshot = await self._extract_dom(page)
        dom_hash = self._hash_dom(dom_snapshot)
        
        # Step 2: Screenshot
        screenshot_path = await self._take_screenshot(page, domain)
        
        # Step 3: ATS detection
        detected_ats = ATSDetector.detect(url)
        if detected_ats == 'unknown':
            # Also check iframes (embedded ATS)
            detected_ats = self._detect_ats_in_iframes(dom_snapshot)
        
        # Step 4: Quick analysis from DOM
        page_type = self._guess_page_type_from_dom(dom_snapshot)
        requires_account = dom_snapshot.get('has_login_form', False)
        form_fields_preview = self._extract_fields_preview(dom_snapshot)
        
        # Step 5: Vision analysis (if enabled and available)
        vision_result = None
        if self.use_vision and self.kimi_vision and screenshot_path:
            vision_result = await self._vision_analysis(screenshot_path)
        
        # Step 6: Build SiteSchema
        schema = self._build_schema(
            url=url,
            domain=domain,
            dom_snapshot=dom_snapshot,
            dom_hash=dom_hash,
            screenshot_path=screenshot_path,
            detected_ats=detected_ats,
            page_type=page_type,
            requires_account=requires_account,
            form_fields_preview=form_fields_preview,
            vision_result=vision_result,
        )
        
        logger.info(f"🔍 Recon: page_type={schema.page_type}, ats={schema.detected_ats}, confidence={schema.confidence}")
        
        return schema
    
    async def _extract_dom(self, page) -> dict:
        """Extract DOM snapshot using JavaScript."""
        try:
            result = await page.evaluate(RECON_DOM_JS)
            return result or {}
        except Exception as e:
            logger.warning(f"DOM extraction failed: {e}")
            return {}
    
    def _hash_dom(self, dom_snapshot: dict) -> str:
        """Create hash of DOM snapshot for caching."""
        # Hash key elements only (not full snapshot)
        key_data = {
            'buttons': len(dom_snapshot.get('buttons', [])),
            'inputs': [(i.get('label', ''), i.get('type', '')) for i in dom_snapshot.get('inputs', [])],
            'forms': len(dom_snapshot.get('forms', [])),
            'url': dom_snapshot.get('url', ''),
        }
        data_str = json.dumps(key_data, sort_keys=True)
        return hashlib.md5(data_str.encode()).hexdigest()[:16]
    
    async def _take_screenshot(self, page, domain: str) -> str:
        """Take screenshot of current page."""
        try:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            filename = f"recon_{domain}_{timestamp}.png"
            filepath = self.screenshot_dir / filename
            
            # Full page screenshot
            await page.screenshot(path=str(filepath), full_page=True)
            logger.debug(f"Screenshot saved: {filepath}")
            return str(filepath)
        except Exception as e:
            logger.warning(f"Screenshot failed: {e}")
            return ""
    
    def _detect_ats_in_iframes(self, dom_snapshot: dict) -> str:
        """Check iframes for known ATS."""
        for iframe in dom_snapshot.get('iframes', []):
            src = iframe.get('src', '')
            if src:
                detected = ATSDetector.detect(src)
                if detected != 'unknown':
                    return detected
        return 'unknown'
    
    def _guess_page_type_from_dom(self, dom_snapshot: dict) -> str:
        """
        Guess page type from DOM snapshot.
        
        Returns:
            page_type string
        """
        # Check for error indicators
        if not dom_snapshot:
            return 'error'
        
        # Check for login requirement
        if dom_snapshot.get('has_login_form'):
            return 'login_required'
        
        # Check for captcha
        if dom_snapshot.get('has_captcha'):
            return 'error'  # Can't proceed with captcha
        
        # Count form inputs
        inputs = dom_snapshot.get('inputs', [])
        forms = dom_snapshot.get('forms', [])
        
        # If multiple inputs and forms → likely application form
        if len(inputs) >= 3 and forms:
            # Check for wizard indicators
            if dom_snapshot.get('has_progress_indicator'):
                return 'form_wizard'
            return 'form_single'
        
        # Check for job listing indicators
        buttons = dom_snapshot.get('buttons', [])
        links = dom_snapshot.get('links', [])
        
        has_apply = any(
            'apply' in (b.get('text', '')).lower()
            for b in buttons + links
        )
        
        if has_apply and len(inputs) < 3:
            return 'job_listing'
        
        # Check for confirmation indicators
        page_title = dom_snapshot.get('page_title', '').lower()
        if any(word in page_title for word in ['thank', 'confirm', 'success', 'submitted']):
            return 'confirmation'
        
        return 'unknown'
    
    def _extract_fields_preview(self, dom_snapshot: dict) -> list[dict]:
        """Extract preview of visible form fields."""
        preview = []
        for inp in dom_snapshot.get('inputs', [])[:10]:  # First 10 fields
            preview.append({
                'label': inp.get('label', ''),
                'type_guess': inp.get('type', 'text'),
                'required': inp.get('required', False),
            })
        return preview
    
    async def _vision_analysis(self, screenshot_path: str) -> dict | None:
        """
        Perform vision analysis using Kimi 2.5.
        
        Args:
            screenshot_path: Path to screenshot
            
        Returns:
            Vision analysis result dict or None
        """
        if not self.kimi_vision:
            return None
        
        try:
            # Read screenshot
            with open(screenshot_path, 'rb') as f:
                image_bytes = f.read()
            
            # Call Kimi Vision
            response = await self.kimi_vision.generate_with_image(
                prompt=RECON_VISION_PROMPT,
                image_bytes=image_bytes,
            )
            
            # Parse JSON response
            # Handle potential markdown code blocks
            response = response.strip()
            if response.startswith('```'):
                # Remove markdown code blocks
                response = re.sub(r'^```(?:json)?\n?', '', response)
                response = re.sub(r'\n?```$', '', response)
            
            result = json.loads(response)
            logger.debug(f"Vision analysis: {result}")
            return result
            
        except json.JSONDecodeError as e:
            logger.warning(f"Vision response not valid JSON: {e}")
            return None
        except Exception as e:
            logger.warning(f"Vision analysis failed: {e}")
            return None
    
    def _build_schema(
        self,
        url: str,
        domain: str,
        dom_snapshot: dict,
        dom_hash: str,
        screenshot_path: str,
        detected_ats: str,
        page_type: str,
        requires_account: bool,
        form_fields_preview: list[dict],
        vision_result: dict | None,
    ) -> SiteSchema:
        """Build SiteSchema from all analysis results."""
        
        # Start with DOM-based values
        has_apply_button = False
        apply_button_hint = ""
        estimated_steps = 1
        language = "en"
        blocked_reason = None
        confidence = 0.5  # DOM-only confidence
        
        # Override with vision results if available
        if vision_result:
            vision_page_type = vision_result.get('page_type', '').lower()
            if vision_page_type in ('job_listing', 'login_required', 'form_single', 'form_wizard', 'confirmation', 'error'):
                page_type = vision_page_type
            
            apply_info = vision_result.get('apply_button', {})
            has_apply_button = apply_info.get('visible', False) if isinstance(apply_info, dict) else False
            if has_apply_button:
                apply_button_hint = f"{apply_info.get('text', 'Apply')} button at {apply_info.get('location', 'unknown')}"
            
            requires_account = vision_result.get('account_required', requires_account)
            estimated_steps = vision_result.get('estimated_steps', estimated_steps)
            language = vision_result.get('language', language)
            blocker = vision_result.get('blocker')
            if blocker:
                blocked_reason = blocker
            confidence = max(confidence, vision_result.get('confidence', 0.5))
        
        # Also check DOM for apply button
        if not has_apply_button:
            buttons = dom_snapshot.get('buttons', [])
            links = dom_snapshot.get('links', [])
            for btn in buttons + links:
                text = btn.get('text', '').lower()
                if 'apply' in text:
                    has_apply_button = True
                    apply_button_hint = f"{btn.get('text', 'Apply')} button"
                    break
        
        # Detect blockers
        if dom_snapshot.get('has_captcha') and not blocked_reason:
            blocked_reason = "captcha"
        
        return SiteSchema(
            url=url,
            domain=domain,
            page_type=page_type,
            requires_account=requires_account,
            has_apply_button=has_apply_button,
            apply_button_hint=apply_button_hint,
            estimated_steps=estimated_steps,
            language=language,
            detected_ats=detected_ats,
            form_fields_preview=form_fields_preview,
            blocked_reason=blocked_reason,
            screenshot_path=screenshot_path,
            raw_dom_hash=dom_hash,
            confidence=confidence,
            dom_snapshot=dom_snapshot,
        )
    
    async def find_apply_button(self, page, screenshot_path: str | None = None) -> dict | None:
        """
        Find Apply button using vision analysis.
        
        Args:
            page: Playwright page object
            screenshot_path: Optional path to existing screenshot
            
        Returns:
            Bounding box dict or None
        """
        if not self.kimi_vision:
            return None
        
        # Take screenshot if not provided
        if not screenshot_path:
            screenshot_path = await self._take_screenshot(page, urlparse(page.url).netloc)
        
        if not screenshot_path:
            return None
        
        try:
            with open(screenshot_path, 'rb') as f:
                image_bytes = f.read()
            
            response = await self.kimi_vision.generate_with_image(
                prompt=APPLY_BUTTON_VISION_PROMPT,
                image_bytes=image_bytes,
            )
            
            # Parse response
            response = response.strip()
            if response.startswith('```'):
                response = re.sub(r'^```(?:json)?\n?', '', response)
                response = re.sub(r'\n?```$', '', response)
            
            result = json.loads(response)
            
            if result.get('not_found'):
                return None
            
            return result
            
        except Exception as e:
            logger.warning(f"Failed to find apply button via vision: {e}")
            return None


# ============================================================================
# Convenience function
# ============================================================================

async def recon_page(page, kimi_vision_client=None, use_vision: bool = True) -> SiteSchema:
    """
    Convenience function to analyze a page.
    
    Args:
        page: Playwright page object
        kimi_vision_client: Optional KimiVisionClient
        use_vision: Enable vision analysis
        
    Returns:
        SiteSchema
    """
    agent = ReconAgent(
        kimi_vision_client=kimi_vision_client,
        use_vision=use_vision,
    )
    return await agent.analyze(page)
