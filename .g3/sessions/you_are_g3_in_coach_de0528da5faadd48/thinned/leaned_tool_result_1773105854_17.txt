Tool result: """
Radio/Select/Checkbox filling mechanics for Universal Screening.

Handles the actual DOM interactions for filling form elements.
No dependencies on AnswerBank/AIAnswerer — pure Playwright operations.
"""

from typing import Optional, List
from loguru import logger


async def click_radio_option(radio) -> bool:
    """
    Click a single radio/ARIA-radio element using escalating strategies.

    Handles both:
    - <input type='radio'> (hidden with CSS, click via label)
    - <div role='radio'> (Indeed's custom styled buttons for consent questions)

    Args:
        radio: Playwright element handle for a radio button

    Returns:
        True if successfully clicked, False otherwise
    """
    # Detect element type first
    tag_and_role = await radio.evaluate(
        "el => el.tagName.toLowerCase() + '|' + (el.getAttribute('role') || '')"
    )
    is_aria_radio = "role='radio'" in tag_and_role or "|radio" in tag_and_role
    is_input = tag_and_role.startswith("input")

    # For div[role='radio'] — just click it directly (it IS the visual element)
    if is_aria_radio and not is_input:
        try:
            await radio.click(timeout=3000)
            logger.debug("🔘 aria-radio: direct click ✓")
            return True
        except Exception:
            pass
        try:
            await radio.evaluate("el => el.click()")
            logger.debug("🔘 aria-radio: JS el.click() ✓")
            return True
        except Exception:
            pass
        try:
            # Dispatch click + aria-checked change
            await radio.evaluate("""el => {
                el.click();
                el.setAttribute('aria-checked', 'true');
                el.dispatchEvent(new MouseEvent('click', {bubbles: true, cancelable: true}));
            }""")
            logger.debug("🔘 aria-radio: JS aria-checked ✓")
            return True
        except Exception:
            pass
        return False

    # For <input type='radio'> — try label-first strategy
    # Pass 1: click the associated <label> via JS (catches hidden inputs)
    try:
        clicked = await radio.evaluate("""el => {
            const label = el.labels?.[0]
                || document.querySelector(`label[for="${el.id}"]`);
            if (label) { label.click(); return true; }
            return false;
        }""")
        if clicked:
            logger.debug("🔘 radio: JS label.click() ✓")
            return True
    except Exception:
        pass

    # Pass 2: Playwright click on the label element
    try:
        label_handle = await radio.evaluate_handle(
            "el => el.labels?.[0] || document.querySelector(`label[for='${el.id}']`) || null"
        )
        if label_handle:
            label_elem = label_handle.as_element()
            if label_elem:
                await label_elem.click(timeout=3000)
                logger.debug("🔘 radio: evaluate_handle label.click() ✓")
                return True
    except Exception:
        pass

    # Pass 3: force-click the hidden input itself
    try:
        await radio.click(force=True, timeout=3000)
        logger.debug("🔘 radio: force-click input ✓")
        return True
    except Exception:
        pass

    # Pass 4: JS el.click()
    try:
        await radio.evaluate("el => el.click()")
        logger.debug("🔘 radio: Pass 4 JS el.click() ✓")
        return True
    except Exception:
        pass

    # Pass 5 (last resort): set checked + dispatch events
    try:
        await radio.evaluate("""el => {
            el.checked = true;
            el.dispatchEvent(new Event('change', {bubbles: true}));
            el.dispatchEvent(new Event('input',  {bubbles: true}));
        }""")
        logger.debug("🔘 radio: Pass 5 JS .checked=true + events ✓")
        return True
    except Exception:
        pass

    return False


async def _get_option_label(radio) -> str:
    """Get display text for a radio option (works for input & aria)."""
    try:
        return await radio.evaluate("""
            el => {
                // aria-label on element itself
                const al = el.getAttribute('aria-label');
                if (al) return al;
                // associated <label> (for input[type=radio])
                const lbl = el.labels?.[0]
                    || document.querySelector(`label[for='${el.id}']`);
                if (lbl) return lbl.textContent;
                // aria-labelledby
                const lblId = el.getAttribute('aria-labelledby');
                if (lblId) {
                    const ref = document.getElementById(lblId);
                    if (ref) return ref.textContent;
                }
                // textContent of the element itself (for role=radio divs)
                return el.textContent || '';
            }
        """)
    except Exception:
        return ""


async def _get_option_value(radio) -> str:
    """Get value attribute for a radio option."""
    try:
        return await radio.evaluate("el => el.value || el.getAttribute('data-value') || ''")
    except Exception:
        return ""


async def fill_radio(element, answer: str, page=None) -> bool:
    """Fill a radio button group. Returns True if an option was clicked.

    Handles:
    - Standard <input type='radio'> in a container element
    - ARIA <div role='radio'> in a container element
    - Indeed name-grouped radios (element is '__name__q_xxx' marker string)

    Matching priority:
    1. Exact label/text match
    2. Exact value attr match
    3. Partial label/text match  (avoids "Strong Yes" matching "Yes")
    4. File-extension fallback for resume selection

    Args:
        element: Playwright element handle for radio group container,
                 or string marker "__name__q_xxx" for name-grouped radios
        answer: Answer text to match against radio options
        page: Playwright page object (required for name-group mode)

    Returns:
        True if an option was clicked, False otherwise
    """
    answer_lower = answer.lower().strip()

    # ── Name-group mode ────────────────────────────────────────────────────
    # Indeed consent questions: no fieldset/ARIA, plain inputs grouped by name.
    # element is the string marker "__name__q_xxx".
    if isinstance(element, str) and element.startswith("__name__"):
        radio_name = element[len("__name__"):]
        if not page:
            logger.warning("⚠️ _fill_radio name-group: page=None, cannot locate radios")
            return False
        radios = await page.locator(f'input[name="{radio_name}"]').all()
    else:
        # Standard container mode
        radios = await element.locator("input[type='radio'], [role='radio']").all()

    if not radios:
        logger.warning(f"⚠️ _fill_radio: no radio elements found in group for answer='{answer}'")
        return False

    # ── Pass 1: exact label or value match ──────────────────────────────
    for radio in radios:
        try:
            label = (await _get_option_label(radio)).strip()
            if label.lower() == answer_lower:
                if await click_radio_option(radio):
                    logger.debug(f"🎯 radio exact label match: '{label}'")
                    return True

            value = (await _get_option_value(radio)).strip()
            if value.lower() == answer_lower:
                if await click_radio_option(radio):
                    logger.debug(f"🎯 radio exact value match: '{value}'")
                    return True
        except Exception:
            pass

    # ── Pass 2: partial / substring match ───────────────────────────────
    for radio in radios:
        try:
            label = (await _get_option_label(radio)).strip()
            label_lower = label.lower()

            if answer_lower in label_lower or label_lower in answer_lower:
                if await click_radio_option(radio):
                    logger.debug(f"🎯 radio partial label match: '{label}'")
                    return True

            value = (await _get_option_value(radio)).strip()
            if answer_lower in value.lower():
                if await click_radio_option(radio):
                    logger.debug(f"🎯 radio partial value match: '{value}'")
                    return True
        except Exception:
            pass

    # ── Pass 3: collect all labels for diagnostic warning ───────────────
    labels: List[str] = []
    for radio in radios:
        labels.append((await _get_option_label(radio)).strip())

    # ── Pass 4: file-selection fallback ─────────────────────────────────
    # Indeed's "Add a resume" question has options like:
    #   ['Use your Indeed Resume', 'Temirlan_Resume_Master_Fixed.pdf']
    # The AnswerBank returns 'Yes' which never matches file names.
    # Pick the option that looks like the uploaded file (has extension),
    # or fall back to the first non-empty option.
    file_extensions = (".pdf", ".doc", ".docx", ".txt", ".rtf")
    for i, lbl in enumerate(labels):
        if any(lbl.lower().endswith(ext) for ext in file_extensions):
            if await click_radio_option(radios[i]):
                logger.debug(f"🎯 radio file-selection fallback: '{lbl}'")
                return True

    logger.warning(
        f"⚠️ _fill_radio: no match for answer='{answer}' "
        f"in options={labels}"
    )
    return False


async def fill_select(element, answer: str) -> None:
    """
    Fill a select dropdown by label, value, or partial match.

    Args:
        element: Playwright element handle for <select>
        answer: Option text or value to select
    """
    try:
        await element.select_option(label=answer)
    except Exception:
        # Try by value
        try:
            await element.select_option(value=answer)
        except Exception:
            # Try partial match
            options = await element.locator("option").all_text_contents()
            for i, opt in enumerate(options):
                if answer.lower() in opt.lower():
                    await element.select_option(index=i)
                    return


async def fill_checkbox(element, answer: str) -> None:
    """
    Fill a checkbox based on yes/no answer.

    Args:
        element: Playwright element handle for checkbox
        answer: "yes", "true", "1" to check; anything else to uncheck
    """
    should_check = answer.lower() in ["yes", "true", "1"]
    is_checked = await element.is_checked()

    if should_check != is_checked:
        await element.click()
