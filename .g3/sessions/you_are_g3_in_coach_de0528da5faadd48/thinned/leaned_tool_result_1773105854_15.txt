"""
AnswerRouter — 3-tier pipeline for answering screening questions.

Level 1: AnswerBank (fast, free)
Level 2: CustomAnswers cache
Level 3: AI Answerer + HITL

Pure business logic — no Playwright page dependency.
"""

from pathlib import Path
from typing import TYPE_CHECKING, Any, Dict, List, Optional

from loguru import logger

from src.applier.answer_bank import AnswerBank
from src.applier.ai_answerer import AIAnswerer

if TYPE_CHECKING:
    from src.applier.human_review import HumanReviewPrompter
    from .models import Question


from .models import is_hitl_needed


class AnswerRouter:
    """
    Routes question answering through 3-tier pipeline.
    
    AnswerBank → cache → AI + HITL.
    Returns str — no DOM manipulation.
    """

    def __init__(
        self,
        answer_bank: AnswerBank,
        ai_answerer: AIAnswerer,
        hitl: "HumanReviewPrompter",
    ) -> None:
        self.answer_bank = answer_bank
        self.ai_answerer = ai_answerer
        self._hitl = hitl

    async def answer_question(
        self,
        question: "Question",
        context: Optional[Dict[str, Any]] = None
    ) -> Optional[str]:
        """
        Answer a single question using three-tier system.

        Args:
            question: Question object
            context: Additional context (job title, company, etc.)

        Returns:
            Answer string or None if failed
        """
        answer = None

        # Level 1: AnswerBank
        if question.field_type == "radio":
            # Use option-aware lookup so answer matches an actual label on the page.
            if question.options:
                answer = self.answer_bank.get_answer_for_options(
                    question.text, question.options
                )
            else:
                answer = self.answer_bank.get_yes_no(question.text)
        elif question.field_type == "checkbox":
            answer = self.answer_bank.get_yes_no(question.text)
        elif question.field_type == "select":
            # Check if it's a years-of-experience dropdown
            if "year" in question.text.lower():
                answer = self.answer_bank.get_years_for_dropdown(question.text, question.options)
            else:
                answer = self.answer_bank.get_answer(question.text)
        elif question.field_type in ["text", "number"]:
            answer = self.answer_bank.get_answer(question.text)
        elif question.field_type == "textarea":
            # For open-ended, use AI directly (skip AnswerBank)
            pass

        # NOTE: check `is not None` (not `if answer`) so that empty string ""
        # (intentionally blank like current_company) gets filled WITHOUT calling AI.
        if answer is not None and question.field_type != "textarea":
            logger.info(f"✅ AnswerBank: '{question.text[:40]}...' → '{answer}'")
            return answer

        # Level 2: Check AI cache
        cached = self.ai_answerer.get_cached_answer(question.text, question.options)
        if cached:
            logger.info(f"🎯 Cache: '{question.text[:40]}...' → '{cached}'")
            return cached

        # Level 3: Gemini API + HITL
        # For textarea, load skills context first
        ai_context = dict(context) if context else {}
        if question.field_type == "textarea":
            skills = await self._load_skills_context(question.text)
            if skills:
                ai_context["skills_content"] = skills

        variants, best = await self.ai_answerer.ask_variants(
            question=question.text,
            field_type=question.field_type,
            options=question.options,
            context=ai_context,
        )

        if not variants:
            logger.warning(f"❌ No answer for: '{question.text[:60]}...'")
            return None

        if is_hitl_needed(question.field_type, question.text):
            chosen = await self._hitl.prompt(
                question=question.text,
                field_type=question.field_type,
                variants=variants,
                company=ai_context.get("company", ""),
                job_title=ai_context.get("job_title", ""),
                default_idx=best,
            )
        else:
            chosen = variants[best] if variants else None

        # For radio with known options, try to match the AI answer to an actual option label.
        if question.field_type == "radio" and question.options and chosen:
            resolved = self.answer_bank.get_answer_for_options(
                question.text, question.options, default=chosen
            )
            chosen = resolved

        if chosen:
            self.ai_answerer.set_cached_answer(question.text, chosen, question.options)
            logger.info(f"🤖 AI: '{question.text[:40]}...' → '{chosen}'")
            return chosen

        logger.warning(f"❌ No answer for: '{question.text[:60]}...'")
        return None

    async def _load_skills_context(self, question_text: str) -> str:
        """Load relevant skills/*.md content for open-ended question context."""
        skills_dir = Path("data/candidate/skills")
        if not skills_dir.exists():
            return ""

        skills_content = ""
        question_lower = question_text.lower()
        generic_keywords = ["experience", "describe", "tell us", "background", "yourself", "relevant"]
        is_generic = any(kw in question_lower for kw in generic_keywords)

        for skill_file in skills_dir.glob("*.md"):
            skill_name = skill_file.stem.lower()
            if skill_name in question_lower or is_generic:
                try:
                    content = skill_file.read_text(encoding="utf-8")
                    skills_content += f"\n--- {skill_file.stem.upper()} ---\n{content[:400]}\n"
                except Exception:
                    pass

        return skills_content
