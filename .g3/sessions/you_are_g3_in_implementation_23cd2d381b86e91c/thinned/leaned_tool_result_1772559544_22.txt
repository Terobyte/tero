Tool result: """
Indeed Smart Router - Routes Indeed job applications to correct ATS applier.

For Indeed jobs that redirect to company career pages, this module:
1. Opens the apply_url in a browser
2. Waits for redirects
3. Detects the actual ATS from HTML
4. Routes to the appropriate applier
"""

import asyncio
import re
from typing import Dict, Any, Optional
from pathlib import Path
import sys

sys.path.insert(0, str(Path(__file__).parent.parent.parent))

from src.applier.ats_detector import ATSDetector


class IndeedRouter:
    """
    Routes Indeed job applications to correct ATS handler.
    
    Indeed jobs often redirect to company career pages (Greenhouse, Lever, etc.)
    This router detects the actual ATS and dispatches to the right applier.
    """
    
    def __init__(self, headless: bool = True):
        self.headless = headless
        self.detector = ATSDetector()
    
    async def route_and_apply(self, job: Dict[str, Any], user_profile: Dict[str, Any]) -> Dict[str, Any]:
        """
        Route Indeed job to correct ATS applier.
        
        Args:
            job: Job dict with keys: id, title, company, url, apply_url
            user_profile: User profile dict with personal info
            
        Returns:
            dict with keys: success (bool), screenshot_path (str|None), error (str|None), ats_type (str)
        """
        result = {
            "success": False,
            "screenshot_path": None,
            "error": None,
            "ats_type": None
        }
        
        try:
            from patchright.async_api import async_playwright
            
            async with async_playwright() as p:
                browser = await p.chromium.launch(headless=self.headless)
                context = await browser.new_context()
                page = await context.new_page()
                
                # Navigate to apply URL
                apply_url = job.get("apply_url") or job.get("url")
                await page.goto(apply_url, wait_until="domcontentloaded", timeout=60000)
                
                # Wait for redirects
                await asyncio.sleep(2.0)
                
                # Detect ATS from page
                ats_type = await self.detector.detect_from_page(page)
                result["ats_type"] = ats_type
                
                await browser.close()
                
                # Route to appropriate applier
                if ats_type == "greenhouse":
                    from src.applier.greenhouse_apply import apply_to_greenhouse
                    return await apply_to_greenhouse(job, user_profile)
                    
                elif ats_type == "lever":
                    from src.applier.lever_apply import LeverApplier
                    applier = LeverApplier(headless=self.headless)
                    success, screenshot = await applier.apply(
                        job_url=job.get("url", ""),
                        job_id=job.get("id", ""),
                        company=job.get("company", ""),
                        position=job.get("title", ""),
                        user_data=user_profile
                    )
                    result["success"] = success
                    result["screenshot_path"] = screenshot
                    return result
                    
                elif ats_type == "ashby":
                    from src.applier.ashby_apply import AshbyApplier
                    applier = AshbyApplier(headless=self.headless)
                    return await applier.apply(
                        job_url=job.get("url", ""),
                        job_id=job.get("id", ""),
                        company=job.get("company", "")
                    )
                    
                elif ats_type == "bamboohr":
                    from src.applier.bamboohr_apply import BambooHRApply
                    applier = BambooHRApply(headless=self.headless)
                    return await applier.apply(
                        job_id=job.get("id", ""),
                        job_url=job.get("url", ""),
                        company=job.get("company", ""),
                        title=job.get("title", ""),
                        applicant_data=user_profile
                    )
                    
                elif ats_type == "workday":
                    from src.applier.workday_apply import WorkdayApplier
                    applier = WorkdayApplier(headless=self.headless)
                    # Merge user_profile into job dict for WorkdayApplier
                    job_with_user = {**job, "user_data": user_profile}
                    return await applier.apply(job_with_user)
                    
                elif ats_type == "smartrecruiters":
                    # TODO: Implement SmartRecruiters applier
                    result["error"] = "SmartRecruiters not yet implemented"
                    return result
                    
                elif ats_type == "icims":
                    # TODO: Implement iCIMS applier
                    result["error"] = "iCIMS not yet implemented"
                    return result
                    
                elif ats_type == "taleo":
                    # TODO: Implement Taleo applier
                    result["error"] = "Taleo not yet implemented"
                    return result
                    
                else:
                    result["error"] = f"Unknown ATS: {ats_type}"
                    return result
                    
        except Exception as e:
            result["error"] = str(e)
            return result
    
    def detect_ats_from_url(self, url: str) -> Optional[str]:
        """
        Quick ATS detection from URL only (without loading page).
        
        Args:
            url: Job URL to check
            
        Returns:
            ATS type string or None
        """
        return self.detector.detect_from_url(url)


async def route_indeed_job(job: Dict[str, Any], user_profile: Dict[str, Any], headless: bool = True) -> Dict[str, Any]:
    """
    Convenience function to route Indeed job.
    
    Args:
        job: Job dict
        user_profile: User profile dict
        headless: Run browser in headless mode
        
    Returns:
        Application result dict
    """
    router = IndeedRouter(headless=headless)
    return await router.route_and_apply(job, user_profile)
