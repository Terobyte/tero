# scripts/collect_cdx_slugs.py
"""
Собирает ATS-слаги из Common Crawl CDX API.

Запуск:
    python scripts/collect_cdx_slugs.py --ats greenhouse --output data/companies/greenhouse_slugs.json
    python scripts/collect_cdx_slugs.py --ats lever --output data/companies/lever_slugs.json
    python scripts/collect_cdx_slugs.py --ats ashby --output data/companies/ashby_slugs.json

Бесплатно. Без регистрации. ~30-80K slugs за 1-2 часа.
"""
import json
import re
import time
import argparse
import urllib.request
from pathlib import Path
from urllib.parse import urlencode


# Конфиг ATS: какой домен искать и как извлечь slug из URL
ATS_CONFIG = {
    "greenhouse": {
        "search_url": "boards.greenhouse.io/*",
        "slug_pattern": r"boards\.greenhouse\.io/([a-zA-Z0-9_-]+)(?:/jobs)?",
        "min_slug_len": 2,
        "max_slug_len": 60,
    },
    "lever": {
        "search_url": "jobs.lever.co/*",
        "slug_pattern": r"jobs\.lever\.co/([a-zA-Z0-9_-]+)",
        "min_slug_len": 2,
        "max_slug_len": 60,
    },
    "ashby": {
        "search_url": "jobs.ashbyhq.com/*",
        "slug_pattern": r"jobs\.ashbyhq\.com/([a-zA-Z0-9_-]+)",
        "min_slug_len": 2,
        "max_slug_len": 60,
    },
}

# Используем последние 4 краула Common Crawl (обновляются каждые 2 месяца)
CDX_INDEXES = [
    "CC-MAIN-2024-10",
    "CC-MAIN-2024-18",
    "CC-MAIN-2024-26",
    "CC-MAIN-2024-33",
]
CDX_BASE = "https://index.commoncrawl.org/{index}-index"


def fetch_cdx_page(index: str, url_pattern: str, page: int = 0) -> list[str]:
    """Запрашивает одну страницу CDX results."""
    params = urlencode({
        "url": url_pattern,
        "output": "json",
        "fl": "url,status",
        "filter": "status:200",
        "limit": 10000,
        "page": page,
    })
    api_url = f"{CDX_BASE.format(index=index)}?{params}"

    try:
        with urllib.request.urlopen(api_url, timeout=30) as resp:
            lines = resp.read().decode("utf-8").strip().split("\n")
            return [l for l in lines if l.strip()]
    except Exception as e:
        print(f"  CDX error page={page}: {e}")
        return []


def extract_slug(url: str, pattern: str, min_len: int, max_len: int) -> str | None:
    """Извлекает slug из URL."""
    match = re.search(pattern, url)
    if match:
        slug = match.group(1).lower().strip("/")
        # Фильтруем мусор
        if slug in {"jobs", "apply", "career", "careers", "embed", "api"}:
            return None
        if min_len <= len(slug) <= max_len:
            return slug
    return None


def collect_slugs(ats: str) -> set[str]:
    """Собирает все уникальные slugs для заданного ATS."""
    cfg = ATS_CONFIG[ats]
    slugs: set[str] = set()
    pattern = cfg["slug_pattern"]

    for index in CDX_INDEXES:
        print(f"\n[{ats}] Crawling index: {index}")
        page = 0
        while True:
            rows = fetch_cdx_page(index, cfg["search_url"], page)
            if not rows:
                break

            for row in rows:
                try:
                    data = json.loads(row)
                    slug = extract_slug(
                        data.get("url", ""),
                        pattern,
                        cfg["min_slug_len"],
                        cfg["max_slug_len"]
                    )
                    if slug:
                        slugs.add(slug)
                except (json.JSONDecodeError, KeyError):
                    continue

            print(f"  page={page} rows={len(rows)} total_slugs={len(slugs)}")
            page += 1
            time.sleep(1)  # вежливый rate limiting

    return slugs


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--ats", required=True, choices=list(ATS_CONFIG.keys()))
    parser.add_argument("--output", required=True)
    parser.add_argument("--merge", action="store_true",
                        help="Объединить с существующим файлом")
    args = parser.parse_args()

    print(f"Collecting {args.ats} slugs from Common Crawl CDX...")
    new_slugs = collect_slugs(args.ats)

    output_path = Path(args.output)
    existing: list[str] = []

    if args.merge and output_path.exists():
        existing = json.loads(output_path.read_text())
        print(f"Merging with {len(existing)} existing slugs")

    merged = sorted(set(existing) | new_slugs)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(json.dumps(merged, indent=2))

    print(f"\nDone: {len(new_slugs)} new slugs → {len(merged)} total → {args.output}")


if __name__ == "__main__":
    main()
