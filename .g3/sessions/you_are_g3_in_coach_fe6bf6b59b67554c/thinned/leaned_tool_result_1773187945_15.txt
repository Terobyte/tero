from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Optional
import json


class JobStatus(str, Enum):
    """Status of a job application."""
    PENDING = "pending"
    IN_PROGRESS = "in_progress"
    APPLIED = "applied"
    ERROR = "error"
    RETRY_SCHEDULED = "retry_scheduled"
    ALREADY_APPLIED = "already_applied"
    NOT_ELIGIBLE = "not_eligible"
    JOB_EXPIRED = "job_expired"
    SKIPPED = "skipped"
    HITL_REQUIRED = "hitl_required"  # Human-in-the-loop needed


class ErrorType(str, Enum):
    """Types of errors that can occur during application."""
    # Retryable errors
    NETWORK_ERROR = "network_error"
    TIMEOUT = "timeout"
    STALE_ELEMENT = "stale_element"
    RATE_LIMITED = "rate_limited"
    CAPTCHA = "captcha"
    PAGE_CRASH = "page_crash"
    UNKNOWN_ERROR = "unknown_error"
    
    # Non-retryable errors
    ALREADY_APPLIED = "already_applied"
    JOB_EXPIRED = "job_expired"
    NOT_ELIGIBLE = "not_eligible"
    MISSING_REQUIRED_FIELD = "missing_required_field"
    BUDGET_EXHAUSTED = "budget_exhausted"
    SITE_BROKEN = "site_broken"


@dataclass
class Job:
    """Represents a job to apply to."""
    id: str
    title: str
    company: str
    url: str
    location: Optional[str] = None
    salary_range: Optional[str] = None
    description: Optional[str] = None
    source: str = "indeed"
    
    # Retry fields
    retry_count: int = 0
    retry_after: Optional[datetime] = None
    last_error_type: Optional[str] = None
    last_error_message: Optional[str] = None
    
    # Status
    status: JobStatus = JobStatus.PENDING
    created_at: datetime = field(default_factory=datetime.utcnow)
    updated_at: datetime = field(default_factory=datetime.utcnow)
    applied_at: Optional[datetime] = None
    
    def to_dict(self) -> dict:
        """Convert to dictionary for storage."""
        return {
            "id": self.id,
            "title": self.title,
            "company": self.company,
            "url": self.url,
            "location": self.location,
            "salary_range": self.salary_range,
            "description": self.description,
            "source": self.source,
            "retry_count": self.retry_count,
            "retry_after": self.retry_after.isoformat() if self.retry_after else None,
            "last_error_type": self.last_error_type,
            "last_error_message": self.last_error_message,
            "status": self.status.value,
            "created_at": self.created_at.isoformat(),
            "updated_at": self.updated_at.isoformat(),
            "applied_at": self.applied_at.isoformat() if self.applied_at else None,
        }
    
    @classmethod
    def from_dict(cls, data: dict) -> "Job":
        """Create from dictionary."""
        return cls(
            id=data["id"],
            title=data["title"],
            company=data["company"],
            url=data["url"],
            location=data.get("location"),
            salary_range=data.get("salary_range"),
            description=data.get("description"),
            source=data.get("source", "indeed"),
            retry_count=data.get("retry_count", 0),
            retry_after=datetime.fromisoformat(data["retry_after"]) if data.get("retry_after") else None,
            last_error_type=data.get("last_error_type"),
            last_error_message=data.get("last_error_message"),
            status=JobStatus(data.get("status", "pending")),
            created_at=datetime.fromisoformat(data["created_at"]) if data.get("created_at") else datetime.utcnow(),
            updated_at=datetime.fromisoformat(data["updated_at"]) if data.get("updated_at") else datetime.utcnow(),
            applied_at=datetime.fromisoformat(data["applied_at"]) if data.get("applied_at") else None,
        )


@dataclass
class ApplicationResult:
    """Result of an application attempt."""
    success: bool
    error_type: Optional[str] = None
    error_message: Optional[str] = None
    screenshot_path: Optional[str] = None
    filled_fields: list = field(default_factory=list)
    failed_fields: list = field(default_factory=list)
    llm_calls: int = 0
    elapsed_seconds: float = 0.0


@dataclass
class StrategyRecord:
    """Record of a successful strategy for learning."""
    domain: str
    element_type: str  # "checkbox", "radio", "select", "input"
    strategy_name: str
    success_count: int = 1
    failure_count: int = 0
    last_success: datetime = field(default_factory=datetime.utcnow)
    created_at: datetime = field(default_factory=datetime.utcnow)
    
    def to_dict(self) -> dict:
        return {
            "domain": self.domain,
            "element_type": self.element_type,
            "strategy_name": self.strategy_name,
            "success_count": self.success_count,
            "failure_count": self.failure_count,
            "last_success": self.last_success.isoformat(),
            "created_at": self.created_at.isoformat(),
        }
    
    @property
    def success_rate(self) -> float:
        total = self.success_count + self.failure_count
        return self.success_count / total if total > 0 else 0.0
