"""
CorrectionExecutor — выполняет коррекцию ответов в DOM после BatchValidator.
Находит элементы по сохранённым локаторам и перезаписывает ответы.
"""
from __future__ import annotations
from loguru import logger
from typing import Any, TYPE_CHECKING

if TYPE_CHECKING:
    from src.applier.universal_screening.batch_validator import Correction
from src.applier.universal_screening.snapshot import StepSnapshot


class CorrectionExecutor:
    """
    Выполняет физическую коррекцию заполненных полей в браузере.

    Usage:
        executor = CorrectionExecutor(page)
        await executor.execute(corrections, snapshot)
    """

    def __init__(self, page):
        self.page = page

    async def execute(
        self,
        corrections: list["Correction"],
        snapshot: StepSnapshot,
    ) -> list[str]:
        """
        Применить коррекции к DOM.
        Возвращает список question_text успешно исправленных вопросов.
        """
        fixed = []
        for correction in corrections:
            snap_q = snapshot.get(correction.question_text)
            if not snap_q or not snap_q.locator_info:
                logger.warning(f"⚠️ CorrectionExecutor: no locator for '{correction.question_text[:50]}'")
                continue

            try:
                ok = await self._refill(
                    snap_q.field_type,
                    snap_q.locator_info,
                    correction.corrected_answer,
                )
                if ok:
                    snap_q.corrected_answer = correction.corrected_answer
                    fixed.append(correction.question_text)
                    logger.info(
                        f"✅ Corrected '{correction.question_text[:40]}': "
                        f"'{correction.current_answer}' → '{correction.corrected_answer}'"
                    )
            except Exception as e:
                logger.error(f"❌ CorrectionExecutor failed for '{correction.question_text[:40]}': {e}")

        return fixed

    async def _refill(self, field_type: str, locator_info: dict, answer: str) -> bool:
        """Re-fill элемент используя сохранённый selector."""
        selector = locator_info.get("selector")
        if not selector:
            return False

        try:
            el = self.page.locator(selector).nth(locator_info.get("nth", 0))

            if field_type == "radio":
                # Import locally to avoid circular deps
                try:
                    from src.applier.universal_screening.radio_filler import fill_radio
                    return await fill_radio(el, answer, page=self.page)
                except ImportError:
                    # Fallback: click radio with matching label
                    await el.click()
                    return True

            elif field_type == "select":
                try:
                    from src.applier.form_utils import fill_select_safe
                    return await fill_select_safe(el, answer, page=self.page)
                except ImportError:
                    # Fallback: use select_option
                    await el.select_option(label=answer)
                    return True

            elif field_type in ("text", "number", "textarea"):
                await el.fill(answer)
                return True

            elif field_type == "checkbox":
                try:
                    from src.applier.universal_screening.radio_filler import fill_checkbox_verified
                    return await fill_checkbox_verified(el, answer)
                except ImportError:
                    # Fallback: check/uncheck based on answer
                    if answer.lower() in ("yes", "true", "1"):
                        await el.check()
                    else:
                        await el.uncheck()
                    return True

        except Exception as e:
            logger.debug(f"_refill error: {e}")
        return False
