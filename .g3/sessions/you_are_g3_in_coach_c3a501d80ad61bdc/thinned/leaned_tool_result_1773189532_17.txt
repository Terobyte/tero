Tool result: """
FormStateMachine - Backtracking state machine for form filling.

Phase 4: Handles "wrong answer → try different answer" scenario.
Saves DOM state before each step, can rollback and retry with variants.
"""

import asyncio
import json
import logging
from dataclasses import dataclass, field
from enum import Enum
from typing import Any, Optional

logger = logging.getLogger(__name__)


class StepResult(Enum):
    """Result of a form step attempt."""
    SUCCESS = "success"
    FAILURE_RETRYABLE = "failure_retryable"  # Try again with same answer
    FAILURE_ALTERNATE = "failure_alternate"  # Try with different answer
    FAILURE_FINAL = "failure_final"  # Escalate to AI Full Form Mode


@dataclass
class DOMSnapshot:
    """Snapshot of form state for rollback."""
    html: str
    values: dict[str, str] = field(default_factory=dict)
    checkboxes: dict[str, bool] = field(default_factory=dict)
    radios: dict[str, str] = field(default_factory=dict)
    selects: dict[str, str] = field(default_factory=dict)
    
    def to_json(self) -> str:
        """Serialize snapshot to JSON."""
        return json.dumps({
            "html": self.html[:1000],  # Truncate for storage
            "values": self.values,
            "checkboxes": self.checkboxes,
            "radios": self.radios,
            "selects": self.selects,
        })
    
    @classmethod
    def from_json(cls, data: str) -> "DOMSnapshot":
        """Deserialize from JSON."""
        obj = json.loads(data)
        return cls(
            html=obj.get("html", ""),
            values=obj.get("values", {}),
            checkboxes=obj.get("checkboxes", {}),
            radios=obj.get("radios", {}),
            selects=obj.get("selects", {}),
        )


@dataclass
class Question:
    """A form question with answer variants."""
    field_id: str
    field_type: str  # "text", "checkbox", "radio", "select"
    label: str
    answer: str
    answer_variants: list[str] = field(default_factory=list)  # Alternative answers
    element: Any = None  # Playwright element
    
    def get_answer_variant(self, attempt: int) -> str:
        """Get answer for given attempt number."""
        if attempt == 0:
            return self.answer
        
        # Try variants in order
        variant_idx = attempt - 1
        if variant_idx < len(self.answer_variants):
            return self.answer_variants[variant_idx]
        
        # Return original if no more variants
        return self.answer


@dataclass
class StepAttempt:
    """Record of a step attempt."""
    step_name: str
    attempt: int
    answer_used: str
    snapshot_before: Optional[DOMSnapshot] = None
    success: bool = False
    error: Optional[str] = None


class FormStateMachine:
    """
    State machine for form filling with backtracking.
    
    Flow:
    1. save_dom_state() before each step
    2. fill_step() with questions and answers
    3. If step fails:
       a. restore_dom_state() to rollback
       b. Try alternative answer variants
       c. After 3 failures, escalate to AI Full Form Mode
    
    Attributes:
        page: Playwright page object
        max_attempts: Max attempts per step (default 3)
        budget_tracker: Optional ApplicationBudget for circuit breaker
    """
    
    MAX_ATTEMPTS_PER_STEP = 3
    BACKOFF_DELAY = 0.5  # Seconds between attempts
    
    def __init__(
        self,
        page: Any,
        budget_tracker: Any = None,
        answer_bank: Any = None,
        ai_answerer: Any = None,
    ):
        self.page = page
        self.budget_tracker = budget_tracker
        self.answer_bank = answer_bank
        self.ai_answerer = ai_answerer
        self._snapshots: list[DOMSnapshot] = []
        self._attempt_history: list[StepAttempt] = []
    
    async def save_dom_state(self) -> DOMSnapshot:
        """
        Capture current form state.
        
        Returns:
            DOMSnapshot with all form values
        """
        snapshot = DOMSnapshot(html="")
        
        try:
            # Get all form elements
            html = await self.page.content()
            snapshot.html = html
            
            # Capture text inputs
            text_inputs = await self.page.locator('input[type="text"], input[type="email"], input[type="tel"], input:not([type])').all()
            for inp in text_inputs:
                try:
                    inp_id = await inp.get_attribute("id") or await inp.get_attribute("name") or ""
                    value = await inp.input_value()
                    if inp_id and value:
                        snapshot.values[inp_id] = value
                except Exception:
                    pass
            
            # Capture checkboxes
            checkboxes = await self.page.locator('input[type="checkbox"]').all()
            for cb in checkboxes:
                try:
                    cb_id = await cb.get_attribute("id") or await cb.get_attribute("name") or ""
                    checked = await cb.is_checked()
                    if cb_id:
                        snapshot.checkboxes[cb_id] = checked
                except Exception:
                    pass
            
            # Capture radios
            radios = await self.page.locator('input[type="radio"]').all()
            for radio in radios:
                try:
                    name = await radio.get_attribute("name") or ""
                    value = await radio.get_attribute("value") or ""
                    if await radio.is_checked() and name:
                        snapshot.radios[name] = value
                except Exception:
                    pass
            
            # Capture selects
            selects = await self.page.locator('select').all()
            for sel in selects:
                try:
                    sel_id = await sel.get_attribute("id") or await sel.get_attribute("name") or ""
                    value = await sel.input_value()
                    if sel_id and value:
                        snapshot.selects[sel_id] = value
                except Exception:
                    pass
            
            self._snapshots.append(snapshot)
            logger.debug(f"📸 Saved DOM state: {len(snapshot.values)} values, {len(snapshot.checkboxes)} checkboxes")
            
        except Exception as e:
            logger.error(f"❌ save_dom_state error: {e}")
        
        return snapshot
    
    async def restore_dom_state(self, snapshot: DOMSnapshot) -> bool:
        """
        Restore form to a previous state.
        
        Args:
            snapshot: Previously saved state
        
        Returns:
            True if restored successfully
        """
        try:
            # Restore text inputs
            for inp_id, value in snapshot.values.items():
                try:
                    locator = self.page.locator(f'#{inp_id}, [name="{inp_id}"]')
                    if await locator.count() > 0:
                        await locator.first.fill(value)
                except Exception:
                    pass
            
            # Restore checkboxes
            for cb_id, checked in snapshot.checkboxes.items():
                try:
                    locator = self.page.locator(f'#{cb_id}, [name="{cb_id}"]')
                    if await locator.count() > 0:
                        el = locator.first
                        current = await el.is_checked()
                        if current != checked:
                            await el.click()
                except Exception:
                    pass
            
            # Restore radios
            for name, value in snapshot.radios.items():
                try:
                    locator = self.page.locator(f'input[type="radio"][name="{name}"][value="{value}"]')
                    if await locator.count() > 0:
                        await locator.first.click()
                except Exception:
                    pass
            
            # Restore selects
            for sel_id, value in snapshot.selects.items():
                try:
                    locator = self.page.locator(f'#{sel_id}, [name="{sel_id}"]')
                    if await locator.count() > 0:
                        await locator.first.select_option(value=value)
                except Exception:
                    pass
            
            logger.debug(f"🔄 Restored DOM state")
            return True
            
        except Exception as e:
            logger.error(f"❌ restore_dom_state error: {e}")
            return False
    
    async def fill_step(
        self,
        step_name: str,
        questions: list[Question],
        fill_fn: Any = None,
    ) -> StepResult:
        """
        Fill a form step with backtracking support.
        
        Args:
            step_name: Name/identifier for this step
            questions: List of questions to fill
            fill_fn: Async function to fill the form (optional)
                     Signature: async fn(questions) -> bool
        
        Returns:
            StepResult indicating success or how to proceed
        """
        for attempt in range(self.MAX_ATTEMPTS_PER_STEP):
            # Check budget
            if self.budget_tracker and not self.budget_tracker.check():
                logger.warning(f"⚠️ Budget exhausted during step '{step_name}'")
                return StepResult.FAILURE_FINAL
            
            # Save state before attempting
            snapshot = await self.save_dom_state()
            
            # Get answer variants for this attempt
            answer_map = {}
            for q in questions:
                answer_map[q.field_id] = q.get_answer_variant(attempt)
            
            attempt_record = StepAttempt(
                step_name=step_name,
                attempt=attempt,
                answer_used=str(answer_map),
                snapshot_before=snapshot,
            )
            
            # Fill each question
            all_success = True
            for q in questions:
                answer = q.get_answer_variant(attempt)
                
                try:
                    success = await self._fill_question(q, answer)
                    if not success:
                        all_success = False
                        break
                except Exception as e:
                    logger.warning(f"⚠️ Error filling {q.field_id}: {e}")
                    all_success = False
                    break
            
            if all_success:
                # Try to advance (submit/next button)
                can_advance = await self._can_advance(step_name)
                if can_advance:
                    attempt_record.success = True
                    self._attempt_history.append(attempt_record)
                    logger.info(f"✅ Step '{step_name}' completed on attempt {attempt + 1}")
                    return StepResult.SUCCESS
                else:
                    logger.info(f"↩️ FSM: rollback step '{step_name}', attempt {attempt + 1} - can't advance")
            
            # Restore state and try again
            await self.restore_dom_state(snapshot)
            attempt_record.error = "Could not advance"
            self._attempt_history.append(attempt_record)
            
            # Backoff before next attempt
            await asyncio.sleep(self.BACKOFF_DELAY * (attempt + 1))
        
        # All attempts exhausted
        logger.warning(f"⚠️ Step '{step_name}' failed after {self.MAX_ATTEMPTS_PER_STEP} attempts")
        
        # Decide result type
        if self._has_variants(questions):
            return StepResult.FAILURE_ALTERNATE
        else:
            return StepResult.FAILURE_FINAL
    
    async def _fill_question(self, question: Question, answer: str) -> bool:
        """
        Fill a single question.
        
        Args:
            question: Question object with element
            answer: Answer to fill
        
        Returns:
            True if filled successfully
        """
        from applier.radio_filler import (
            fill_checkbox_verified,
            click_radio_option,
            fill_select,
            scroll_into_view_if_needed,
        )
        
        element = question.element
        if not element:
            # Try to find element
            element = await self._find_element(question)
            if not element:
                logger.warning(f"⚠️ Could not find element for {question.field_id}")
                return False
        
        # Scroll into view
        await scroll_into_view_if_needed(element)
        
        # Fill based on type
        field_type = question.field_type.lower()
        
        try:
            if field_type == "text" or field_type == "email" or field_type == "tel":
                await element.fill(answer)
                return True
            
            elif field_type == "checkbox":
                return await fill_checkbox_verified(element, answer)
            
            elif field_type == "radio":
                # Find radio group
                name = await element.get_attribute("name") or ""
                group = self.page.locator(f'input[type="radio"][name="{name}"]')
                return await click_radio_option(group, answer)
            
            elif field_type == "select":
                return await fill_select(element, answer, page=self.page)
            
            else:
                # Generic fill
                await element.fill(answer)
                return True
                
        except Exception as e:
            logger.error(f"❌ Error filling {question.field_id}: {e}")
            return False
    
    async def _find_element(self, question: Question) -> Any:
        """Find element for a question."""
        field_id = question.field_id
        
        # Try various selectors
        selectors = [
            f'#{field_id}',
            f'[name="{field_id}"]',
            f'[aria-label*="{question.label}" i]',
            f'label:has-text("{question.label}") >> input',
        ]
        
        for selector in selectors:
            try:
                locator = self.page.locator(selector).first
                if await locator.count() > 0:
                    return locator
            except Exception:
                pass
        
        return None
    
    async def _can_advance(self, step_name: str) -> bool:
        """
        Check if we can advance to the next step.
        
        This checks for:
        - No visible error messages
        - Submit/next button is clickable
        - No validation errors
        
        Returns:
            True if form appears ready to advance
        """
        try:
            # Check for error messages
            error_selectors = [
                '.error', '.alert-error', '[role="alert"]',
                '.validation-error', '.form-error',
            ]
            
            for selector in error_selectors:
                errors = self.page.locator(selector)
                if await errors.count() > 0:
                    # Check if visible
                    visible_error = errors.first
                    if await visible_error.is_visible():
                        error_text = await visible_error.text_content()
                        logger.debug(f"Found error: {error_text}")
                        return False
            
            # Check if next/submit button exists and is enabled
            next_buttons = [
                'button[type="submit"]',
                'button:has-text("Next")',
                'button:has-text("Continue")',
                'button:has-text("Submit")',
            ]
            
            for selector in next_buttons:
                btn = self.page.locator(selector).first
                if await btn.count() > 0:
                    if await btn.is_enabled():
                        return True
            
            # If no next button found, assume success
            return True
            
        except Exception as e:
            logger.debug(f"Error checking advance: {e}")
            return True
    
    def _has_variants(self, questions: list[Question]) -> bool:
        """Check if any questions have answer variants."""
        for q in questions:
            if q.answer_variants:
                return True
        return False
    
    def get_attempt_history(self) -> list[StepAttempt]:
        """Get history of all attempts."""
        return self._attempt_history.copy()


# Convenience function
async def fill_form_by_description(
    page: Any,
    questions: list[dict],
    llm_chain: Any = None,
    verify_fn: Any = None,
) -> bool:
    """
    Fill an entire form by description using AI.
    
    This is Phase 8 - Full Form AI Mode.
    Used when step-by-step filling has failed multiple times.
    
    Args:
        page: Playwright page
        questions: List of question dicts with 'label' and 'answer'
        llm_chain: LLMChain for AI calls
        verify_fn: Optional verification function
    
    Returns:
        True if form filled successfully
    """
    if not llm_chain:
        from ai.llm_chain import LLMChain
        llm_chain = LLMChain()
    
    try:
        # Take screenshot
        screenshot = await page.screenshot()
        
        # Format questions for LLM
        questions_text = "\n".join([
            f"- {q.get('label', 'Unknown')}: {q.get('answer', '')}"
            for q in questions
        ])
        
        prompt = f"""Fill out this form. Here are the questions and answers:

{questions_text}

For each field, provide:
1. The field label or description
2. The action to take (click, type, select)
3. The value if applicable

Respond in JSON format:
[
  {{"field": "field_description", "action": "type", "value": "answer"}},
  {{"field": "field_description", "action": "click", "value": "checked"}}
]

Only include fields that need to be filled.
"""
        
        result = await llm_chain.call(
            prompt=prompt,
            screenshot=screenshot,
            page=page,
            verify_fn=verify_fn,
        )
        
        if not result.success:
            logger.warning("⚠️ fill_form_by_description: LLM call failed")
            return False
        
        # Parse actions
        import json
        content = result.response.content.strip()
        
        # Try to extract JSON
        try:
            # Remove markdown code blocks if present
            if "```" in content:
                import re
                content = re.search(r'```(?:json)?\s*([\s\S]*?)\s*```', content)
                if content:
                    content = content.group(1)
            
            actions = json.loads(content)
        except json.JSONDecodeError as e:
            logger.warning(f"⚠️ Could not parse LLM response: {e}")
            return False
        
        # Execute actions
        for action in actions:
            field_desc = action.get("field", "")
            action_type = action.get("action", "type")
            value = action.get("value", "")
            
            # Try to find field by description
            # This is a simplified implementation
            try:
                if action_type == "type":
                    # Find text input by label
                    locator = page.locator(f'input:below(label:has-text("{field_desc}"))').first
                    if await locator.count() > 0:
                        await locator.fill(value)
                
                elif action_type == "click":
                    # Find checkbox/radio by label
                    locator = page.locator(f'input[type="checkbox"], input[type="radio"]').near(page.locator(f'label:has-text("{field_desc}")')).first
                    if await locator.count() > 0:
                        await locator.click()
                
                elif action_type == "select":
                    # Find select by label
                    locator = page.locator(f'select:below(label:has-text("{field_desc}"))').first
                    if await locator.count() > 0:
                        await locator.select_option(label=value)
                
                await asyncio.sleep(0.1)  # Small delay between actions
                
            except Exception as e:
                logger.warning(f"⚠️ Could not execute action {action}: {e}")
        
        # Verify if function provided
        if verify_fn:
            return await verify_fn(page)
        
        return True
        
    except Exception as e:
        logger.error(f"❌ fill_form_by_description error: {e}")
        return False
