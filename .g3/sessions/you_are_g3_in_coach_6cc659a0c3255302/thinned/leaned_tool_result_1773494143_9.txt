"""
JS-скрипты для спуфинга fingerprint API.

ВАЖНО: Не используй add_init_script() с patchright — ломает DNS.
Вместо этого инжекти через page.evaluate() СРАЗУ после goto().
Порядок: goto() → inject_fingerprint_protection() → дальнейшая работа.
"""
import random


def canvas_noise_script(noise_seed: int | None = None) -> str:
    """
    Добавляет ±1 noise к каждому пикселю canvas.getImageData().

    Args:
        noise_seed: Фиксированный seed для воспроизводимых тестов.
                    None = случайный (для реальных сессий).
    Returns:
        JS-строка для инжекции через page.evaluate()
    """
    seed_js = str(noise_seed) if noise_seed is not None else "Math.random() * 1000000 | 0"
    return f"""
(function() {{
    const SEED = {seed_js};
    // Простой PRNG: mulberry32
    function prng(seed) {{
        return function() {{
            seed |= 0; seed = seed + 0x6D2B79F5 | 0;
            var t = Math.imul(seed ^ seed >>> 15, 1 | seed);
            t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
            return ((t ^ t >>> 14) >>> 0) / 4294967296;
        }};
    }}
    const rand = prng(SEED);

    const origGetImageData = CanvasRenderingContext2D.prototype.getImageData;
    CanvasRenderingContext2D.prototype.getImageData = function(sx, sy, sw, sh, settings) {{
        const imageData = origGetImageData.call(this, sx, sy, sw, sh, settings);
        const data = imageData.data;
        // Добавляем ±1 к RGB каналам (не к alpha)
        for (let i = 0; i < data.length; i += 4) {{
            data[i]     = Math.max(0, Math.min(255, data[i]     + (rand() > 0.5 ? 1 : -1)));
            data[i + 1] = Math.max(0, Math.min(255, data[i + 1] + (rand() > 0.5 ? 1 : -1)));
            data[i + 2] = Math.max(0, Math.min(255, data[i + 2] + (rand() > 0.5 ? 1 : -1)));
        }}
        return imageData;
    }};

    // Также патчим toDataURL и toBlob
    const origToDataURL = HTMLCanvasElement.prototype.toDataURL;
    HTMLCanvasElement.prototype.toDataURL = function(type, quality) {{
        const ctx = this.getContext('2d');
        if (ctx) {{
            // Рисуем невидимый пиксель — меняет итоговый hash
            const x = rand() * this.width | 0;
            const y = rand() * this.height | 0;
            const orig = ctx.getImageData(x, y, 1, 1).data;
            ctx.fillStyle = `rgba(${{orig[0]}}, ${{orig[1]}}, ${{orig[2]}}, 0.004)`;
            ctx.fillRect(x, y, 1, 1);
        }}
        return origToDataURL.call(this, type, quality);
    }};
}})();
"""


def audio_noise_script(noise_seed: int | None = None) -> str:
    """
    Добавляет микро-noise к AudioContext frequency/time данным.
    Noise настолько мал (~1e-7), что аудио звучит нормально,
    но fingerprint хэш меняется.
    """
    seed_js = str(noise_seed) if noise_seed is not None else "Math.random() * 1000000 | 0"
    return f"""
(function() {{
    const SEED = {seed_js};
    function prng(seed) {{
        return function() {{
            seed |= 0; seed = seed + 0x6D2B79F5 | 0;
            var t = Math.imul(seed ^ seed >>> 15, 1 | seed);
            t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
            return ((t ^ t >>> 14) >>> 0) / 4294967296;
        }};
    }}
    const rand = prng(SEED);
    const NOISE_LEVEL = 0.0000001;  // ~1e-7: невидимый для человека

    const origGetChannelData = AudioBuffer.prototype.getChannelData;
    AudioBuffer.prototype.getChannelData = function(channel) {{
        const data = origGetChannelData.call(this, channel);
        for (let i = 0; i < data.length; i++) {{
            data[i] += (rand() - 0.5) * NOISE_LEVEL;
        }}
        return data;
    }};

    const origGetFloatFreqData = AnalyserNode.prototype.getFloatFrequencyData;
    AnalyserNode.prototype.getFloatFrequencyData = function(array) {{
        origGetFloatFreqData.call(this, array);
        for (let i = 0; i < array.length; i++) {{
            array[i] += (rand() - 0.5) * NOISE_LEVEL;
        }}
    }};

    const origGetByteFreqData = AnalyserNode.prototype.getByteFrequencyData;
    AnalyserNode.prototype.getByteFrequencyData = function(array) {{
        origGetByteFreqData.call(this, array);
        // Byte data: иногда меняем 1 значение на ±1
        const idx = Math.floor(rand() * array.length);
        array[idx] = Math.max(0, Math.min(255, array[idx] + (rand() > 0.5 ? 1 : -1)));
    }};
}})();
"""


def battery_spoof_script() -> str:
    """
    Спуфит navigator.getBattery() под типичный ноутбук.
    level: 0.65-0.95, charging: true/false случайно.
    """
    return """
(function() {
    const BATTERY_LEVEL = 0.65 + Math.random() * 0.30;  // 65-95%
    const IS_CHARGING = Math.random() > 0.4;              // 60% = charging

    const fakeBattery = {
        charging: IS_CHARGING,
        chargingTime: IS_CHARGING ? Math.floor(Math.random() * 3600) : Infinity,
        dischargingTime: IS_CHARGING ? Infinity : Math.floor(5400 + Math.random() * 3600),
        level: BATTERY_LEVEL,
        addEventListener: () => {},
        removeEventListener: () => {},
        dispatchEvent: () => true,
    };

    if (navigator.getBattery) {
        navigator.getBattery = async function() {
            return fakeBattery;
        };
    }
})();
"""
