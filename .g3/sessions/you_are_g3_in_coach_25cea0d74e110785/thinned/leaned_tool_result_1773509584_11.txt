"""
MatchRateCalculator — оценка соответствия JD и профиля кандидата.

Возвращает детальный отчёт с breakdown по категориям:
- Skills match
- Experience match
- Education match
- Keywords coverage

Используется для фильтрации вакансий перед применением.
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Optional

from loguru import logger

from src.ai.gap_analyser import GapAnalysis


@dataclass
class MatchReport:
    """Детальный отчёт о соответствии."""
    total_score: float              # 0-100
    skill_score: float              # 0-40
    experience_score: float         # 0-30
    keyword_score: float            # 0-20
    education_score: float          # 0-10
    matched_skills: list[str] = field(default_factory=list)
    missing_critical: list[str] = field(default_factory=list)
    recommendation: str = ""        # "apply" / "skip" / "stretch"

    @property
    def grade(self) -> str:
        if self.total_score >= 80:
            return "A (Excellent match)"
        elif self.total_score >= 65:
            return "B (Good match)"
        elif self.total_score >= 50:
            return "C (Partial match)"
        else:
            return "D (Weak match)"

    def to_log(self) -> str:
        return (
            f"Match: {self.total_score:.0f}/100 ({self.grade}) | "
            f"skills={self.skill_score:.0f} exp={self.experience_score:.0f} "
            f"kw={self.keyword_score:.0f} | {self.recommendation}"
        )


class MatchRateCalculator:
    """
    Оценка соответствия резюме и JD.

    Usage:
        calc = MatchRateCalculator()
        report = calc.calculate(gap_analysis, job, profile)
        if report.total_score < 40:
            skip()
    """

    # Минимальный score для подачи заявки
    APPLY_THRESHOLD = 40.0
    STRETCH_THRESHOLD = 55.0

    def calculate(
        self,
        gap_analysis: GapAnalysis,
        job: dict,
        profile: Optional[dict] = None,
        facts: Optional[dict] = None,
    ) -> MatchReport:
        """
        Рассчитать match rate.

        Args:
            gap_analysis: Результат GapAnalyser
            job: Словарь вакансии
            profile: profile.yaml
            facts: facts.json из PKB
        """
        profile = profile or {}
        facts = facts or {}

        skill_score = self._skill_score(gap_analysis)
        exp_score = self._experience_score(job, profile, facts)
        kw_score = self._keyword_score(gap_analysis)
        edu_score = self._education_score(job, profile)

        total = skill_score + exp_score + kw_score + edu_score

        # Определить рекомендацию
        if total >= self.STRETCH_THRESHOLD:
            recommendation = "apply"
        elif total >= self.APPLY_THRESHOLD:
            recommendation = "stretch"
        else:
            recommendation = "skip"

        # Critical missing skills (в gaps, но явно требуются)
        missing_critical = [g for g in gap_analysis.gaps
                           if any(kw in g.lower() for kw in
                                  ["required", "must", "essential", "mandatory"])]

        report = MatchReport(
            total_score=round(total, 1),
            skill_score=round(skill_score, 1),
            experience_score=round(exp_score, 1),
            keyword_score=round(kw_score, 1),
            education_score=round(edu_score, 1),
            matched_skills=gap_analysis.matched[:10],
            missing_critical=missing_critical,
            recommendation=recommendation,
        )
        logger.info(report.to_log())
        return report

    def _skill_score(self, gap: GapAnalysis) -> float:
        """0-40 баллов за покрытие навыков."""
        total = len(gap.matched) + len(gap.adjacent) + len(gap.gaps)
        if not total:
            return 20.0
        # matched=full, adjacent=half, gaps=0
        covered = len(gap.matched) + len(gap.adjacent) * 0.5
        return min(40.0, (covered / total) * 40.0)

    def _experience_score(self, job: dict, profile: dict, facts: dict) -> float:
        """0-30 баллов за опыт."""
        score = 0.0
        exp = profile.get("experience", {})
        total_years = int(exp.get("total_years", 0) or 0)

        # Базовый опыт
        if total_years >= 5:
            score += 30.0
        elif total_years >= 3:
            score += 22.0
        elif total_years >= 1:
            score += 15.0
        else:
            score += 8.0

        # Salary match (up to 5 bonus points)
        salary = profile.get("salary", {})
        expected = int(salary.get("expected_usd", 0) or 0)
        job_max = int(job.get("salary_max", 0) or 0)
        if expected and job_max and job_max >= expected:
            score = min(30.0, score + 5.0)

        return score

    def _keyword_score(self, gap: GapAnalysis) -> float:
        """0-20 баллов за keyword coverage."""
        total = len(gap.matched) + len(gap.adjacent) + len(gap.gaps)
        if not total:
            return 10.0
        coverage = len(gap.matched) / total
        return min(20.0, coverage * 20.0)

    def _education_score(self, job: dict, profile: dict) -> float:
        """0-10 баллов за образование."""
        certs = profile.get("certifications", {})
        score = 5.0  # базовый балл
        if certs.get("has_bachelors") or certs.get("has_associates"):
            score += 3.0
        if certs.get("has_masters"):
            score += 2.0
        return min(10.0, score)
