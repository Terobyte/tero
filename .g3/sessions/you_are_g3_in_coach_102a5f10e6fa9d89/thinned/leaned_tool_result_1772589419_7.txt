"""
Google Gemini Deep Research Automation Script

Usage:
    python deep_research.py --login    # First time - save session
    python deep_research.py "query"    # Run research
"""

import asyncio
import sys
from datetime import datetime
from pathlib import Path
from typing import Optional

import click
from loguru import logger

from browser import StealthBrowser
from extractor import ReportExtractor

# Configure logger
logger.remove()
logger.add(
    sys.stderr,
    format="<green>{time:HH:mm:ss}</green> | <level>{level: <8}</level> | <cyan>{message}</cyan>",
    level="INFO",
)


# Constants
GEMINI_URL = "https://gemini.google.com"
GEMINI_APP_URL = "https://gemini.google.com/app"
SESSION_FILE = Path(__file__).parent / "session.json"
OUTPUT_DIR = Path.home() / "Desktop" / "Personal Researchs"


class DeepResearchRunner:
    """Main class for running Deep Research automation."""

    def __init__(self):
        self.browser: Optional[StealthBrowser] = None
        self.extractor = ReportExtractor()

    async def login_and_save_session(self) -> None:
        """
        Open browser for manual login and save session.
        """
        logger.info("Starting browser for manual login...")
        self.browser = StealthBrowser(headless=False)
        await self.browser.start()

        try:
            # Navigate to Gemini
            logger.info(f"Navigating to {GEMINI_URL}")
            await self.browser.page.goto(GEMINI_URL, wait_until="networkidle")

            # Wait for user to login manually
            logger.info("Please login to your Google account in the browser")
            logger.info("Press Enter in this terminal when you have successfully logged in...")

            # Wait for user input in a separate thread
            loop = asyncio.get_event_loop()
            await loop.run_in_executor(None, input)

            # Save the session
            await self.browser.save_session(SESSION_FILE)
            logger.success(f"Session saved to {SESSION_FILE}")

        finally:
            await self.browser.close()

    async def run_research(self, query: str) -> Path:
        """
        Run Deep Research with the given query.
        
        Args:
            query: The research query
            
        Returns:
            Path to the saved report file
        """
        logger.info(f"Starting Deep Research for: {query}")

        # Ensure output directory exists
        OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

        # Start browser with saved session
        self.browser = StealthBrowser(headless=False)
        await self.browser.start()

        try:
            # Load session
            if not SESSION_FILE.exists():
                logger.error("No session file found! Please run with --login first.")
                raise FileNotFoundError("session.json not found. Run with --login to create it.")

            await self.browser.load_session(SESSION_FILE)
            logger.info("Session loaded successfully")

            # Navigate to Gemini app
            logger.info(f"Navigating to {GEMINI_APP_URL}")
            await self.browser.page.goto(GEMINI_APP_URL, wait_until="networkidle")
            await asyncio.sleep(2)  # Wait for page to fully load

            # Select Deep Research model
            await self._select_deep_research()

            # Enter query
            await self._enter_query(query)

            # Submit and confirm
            await self._submit_research()

            # Wait for research to complete
            report_html = await self._wait_for_completion()

            # Extract and save report
            output_path = await self._save_report(query, report_html)

            return output_path

        finally:
            await self.browser.close()

    async def _select_deep_research(self) -> None:
        """Select Deep Research from the model dropdown."""
        logger.info("Selecting Deep Research model...")

        # Try multiple selectors for the model toggle button
        model_selectors = [
            'button[aria-label*="model"]',
            'button[aria-label*="Model"]',
            'button:has-text("Fast")',
            'button:has-text("Thinking")',
            'button[aria-label*="Fast"]',
            'button[aria-label*="Thinking"]',
            '.model-selector-button',
            'button[data-test-id="model-selector"]',
            'button:has(mat-icon):near(:text("Fast"))',
            'div[role="button"]:has-text("Fast")',
            'div[role="button"]:has-text("Thinking")',
        ]

        model_button = None
        for selector in model_selectors:
            try:
                model_button = await self.browser.page.wait_for_selector(selector, timeout=3000)
                if model_button:
                    logger.debug(f"Found model button with selector: {selector}")
                    break
            except Exception:
                continue

        if not model_button:
            # Try a broader approach - look for any clickable element that might be the model selector
            logger.warning("Could not find model button with standard selectors, trying alternative approach...")
            try:
                # The model selector is usually in the top bar
                model_button = await self.browser.page.query_selector(
                    'button:has-text("Fast"), button:has-text("Thinking")'
                )
            except Exception:
                pass

        if model_button:
            await model_button.click()
            logger.info("Model dropdown opened")
            await asyncio.sleep(1)
        else:
            logger.warning("Could not find model toggle button, continuing anyway...")
            return

        # Click on Deep Research option
        deep_research_selectors = [
            'text="Deep Research"',
            'button:has-text("Deep Research")',
            'div:has-text("Deep Research"):not(:has(div))',
            '[data-value="deep-research"]',
            'li:has-text("Deep Research")',
            'mat-option:has-text("Deep Research")',
            'div[role="option"]:has-text("Deep Research")',
            'button[role="menuitem"]:has-text("Deep Research")',
        ]

        for selector in deep_research_selectors:
            try:
                deep_research_option = await self.browser.page.wait_for_selector(selector, timeout=3000)
                if deep_research_option:
                    await deep_research_option.click()
                    logger.success("Deep Research model selected")
                    await asyncio.sleep(1)
                    return
            except Exception:
                continue

        logger.warning("Could not click Deep Research option, it might already be selected")

    async def _enter_query(self, query: str) -> None:
        """Enter the research query into the textarea."""
        logger.info("Entering query...")

        # Multiple selectors for the input textarea
        textarea_selectors = [
            'textarea[aria-label*="Enter a prompt"]',
            'textarea[placeholder*="Enter"]',
            'textarea[placeholder*="prompt"]',
            'div[contenteditable="true"]',
            'textarea.ql-editor',
            'rich-textarea textarea',
            '.text-input-field textarea',
            'textarea[aria-label*="message"]',
            'textarea.input-area',
            'div[role="textbox"]',
            'div[contenteditable][role="textbox"]',
        ]

        textarea = None
        for selector in textarea_selectors:
            try:
                textarea = await self.browser.page.wait_for_selector(selector, timeout=5000)
                if textarea:
                    logger.debug(f"Found textarea with selector: {selector}")
                    break
            except Exception:
                continue

        if not textarea:
            logger.error("Could not find input textarea")
            raise RuntimeError("Could not find input textarea")

        # Clear and enter query
        await textarea.click()
        await textarea.fill("")  # Clear any existing content
        await asyncio.sleep(0.3)

        # Type the query
        await textarea.type(query, delay=10)
        logger.info(f"Query entered: {query[:50]}...")
        await asyncio.sleep(0.5)

    async def _submit_research(self) -> None:
        """Submit the research query and confirm the plan."""
        logger.info("Submitting research...")

        # Press Enter to submit
        await self.browser.page.keyboard.press("Enter")
        await asyncio.sleep(3)  # Wait for response

        # Look for "Start research" button (Deep Research confirmation)
        start_button_selectors = [
            'button:has-text("Start research")',
            'button:has-text("Start Research")',
            'button:has-text("Start")',
            'button:has-text("Begin")',
            'button:has-text("Confirm")',
            'button[aria-label*="Start research"]',
            'button[data-test-id="start-research"]',
            'div[role="button"]:has-text("Start research")',
        ]

        for selector in start_button_selectors:
            try:
                start_button = await self.browser.page.wait_for_selector(selector, timeout=5000)
                if start_button:
                    await start_button.click()
                    logger.success("Research started")
                    await asyncio.sleep(2)
                    return
            except Exception:
                continue

        # If no button found, research might start automatically
        logger.info("Research may have started automatically (no confirmation button found)")

    async def _wait_for_completion(self) -> str:
        """Wait for research to complete and return the report HTML."""
        logger.info("Waiting for Deep Research to complete...")
        logger.info("This may take several minutes...")

        poll_interval = 30  # seconds
        max_wait_time = 60 * 60  # 1 hour max
        elapsed = 0

        while elapsed < max_wait_time:
            elapsed += poll_interval

            # Check for completion indicators
            # 1. Export button appearing
            # 2. Spinner disappearing
            # 3. "Research complete" text

            try:
                # Check for export button (appears when research is complete)
                export_selectors = [
                    'button[aria-label*="Export"]',
                    'button:has-text("Export")',
                    'button[aria-label*="Share"]',
                    'button:has-text("Share")',
                    'button[aria-label*="Copy"]',
                    'button:has-text("Copy")',
                    '[data-test-id="export-button"]',
                ]

                for selector in export_selectors:
                    try:
                        export_button = await self.browser.page.query_selector(selector)
                        if export_button and await export_button.is_visible():
                            logger.success("Research completed! Export button found.")
                            return await self._extract_report_html()
                    except Exception:
                        continue

                # Check for spinner (if present, research is still running)
                spinner_selectors = [
                    '.loading-spinner',
                    'mat-spinner',
                    '.spinner',
                    'div[role="progressbar"]',
                    'circle[stroke]',
                ]

                spinner_found = False
                for selector in spinner_selectors:
                    try:
                        spinner = await self.browser.page.query_selector(selector)
                        if spinner and await spinner.is_visible():
                            spinner_found = True
                            break
                    except Exception:
                        continue

                # Check for "Research complete" or similar text
                completion_text = await self.browser.page.query_selector(
                    'text="Research complete", text="Research completed", text="Done"'
                )

                if completion_text:
                    logger.success("Research completed! Completion text found.")
                    return await self._extract_report_html()

                # If no spinner and no export button, check if we're at a completion state
                if not spinner_found:
                    # Might be complete - try to extract content
                    page_content = await self.browser.page.content()
                    if "export" in page_content.lower() or len(page_content) > 10000:
                        # Heuristic: lots of content suggests completion
                        logger.success("Research appears complete (no spinner, content detected)")
                        return await self._extract_report_html()

                logger.info(f"Research in progress... ({elapsed}s elapsed, next check in {poll_interval}s)")

            except Exception as e:
                logger.debug(f"Error checking completion status: {e}")

            await asyncio.sleep(poll_interval)

        raise TimeoutError("Deep Research did not complete within the maximum wait time")

    async def _extract_report_html(self) -> str:
        """Extract the HTML content of the completed report."""
        logger.info("Extracting report content...")

        # Try multiple selectors for the response content
        report_selectors = [
            '.response-content',
            '.message-content',
            'model-response',
            '.model-response',
            '.output-content',
            'response-container',
            '.conversation-turn:last-child',
            '.user-query + .model-response',
            'div[data-test-id="response"]',
            '.markdown-content',
            'div.prose',
            '.response-container',
        ]

        for selector in report_selectors:
            try:
                element = await self.browser.page.query_selector(selector)
                if element:
                    html = await element.inner_html()
                    if len(html) > 100:  # Ensure we got meaningful content
                        logger.success(f"Extracted {len(html)} characters of HTML")
                        return html
            except Exception:
                continue

        # Fallback: get the entire page content
        logger.warning("Could not find specific report container, extracting full page")
        html = await self.browser.page.content()
        return html

    async def _save_report(self, query: str, html: str) -> Path:
        """Save the report as a Markdown file."""
        logger.info("Converting report to Markdown...")

        # Extract Markdown from HTML
        markdown_content = self.extractor.extract(html, query)

        # Generate filename
        date_str = datetime.now().strftime("%Y-%m-%d")
        slug = self.extractor.slugify(query)
        filename = f"{date_str}_{slug}.md"
        output_path = OUTPUT_DIR / filename

        # Save the file
        self.extractor.save(markdown_content, output_path)

        logger.success(f"Report saved to: {output_path}")
        return output_path


# CLI Interface
@click.command()
@click.argument("query", required=False)
@click.option("--login", is_flag=True, help="Login and save session")
@click.option("-v", "--verbose", is_flag=True, help="Enable verbose logging")
def main(query: Optional[str], login: bool, verbose: bool):
    """
    Google Gemini Deep Research Automation

    Examples:

        python deep_research.py --login

        python deep_research.py "How does quantum entanglement work?"
    """
    if verbose:
        logger.remove()
        logger.add(
            sys.stderr,
            format="<green>{time:HH:mm:ss}</green> | <level>{level: <8}</level> | <cyan>{function}</cyan>:<cyan>{line}</cyan> | <cyan>{message}</cyan>",
            level="DEBUG",
        )

    if login:
        # Login mode
        asyncio.run(DeepResearchRunner().login_and_save_session())
    elif query:
        # Research mode
        runner = DeepResearchRunner()
        output_path = asyncio.run(runner.run_research(query))
        click.echo(f"\n✅ Report saved to: {output_path}")
    else:
        click.echo("Error: Please provide a query or use --login")
        click.echo("\nUsage:")
        click.echo("  python deep_research.py --login           # First time login")
        click.echo('  python deep_research.py "your query"      # Run research')
        sys.exit(1)


if __name__ == "__main__":
    main()
